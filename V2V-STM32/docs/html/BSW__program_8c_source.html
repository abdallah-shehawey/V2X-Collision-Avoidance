67/*
68 * @brief Begin a new BSW processing cycle
69 *
70 * - Latch the four side distances.
71 * - Sender role: build our bsw_flag from the front-side sensors. Each side is
72 * an independent bit, so a vehicle on both sides sets BSW_FLAG_BOTH.
73 * - Reset the receiver-side alert accumulators.
74 */
75void BSW_voidBeginCycle(float front_left, float front_right, float rear_left, float rear_right)
76{
77 BSW_FrontLeft = front_left;
78 BSW_FrontRight = front_right;
79 BSW_RearLeft = rear_left;
80 BSW_RearRight = rear_right;
81
82 /* ---- Sender: front-side detection, one bit per side (no priority) ---- */
83 BSW_SenderFlag = BSW_FLAG_NONE;
84
85 if (front_left > 0.0f && front_left < BSW_SIDE_THRESHOLD)
86 {
87 BSW_SenderFlag |= BSW_FLAG_LEFT;
88 }
89 if (front_right > 0.0f && front_right < BSW_SIDE_THRESHOLD)
90 {
91 BSW_SenderFlag |= BSW_FLAG_RIGHT;
92 }
93
94 /* ---- Receiver: reset per-side alert accumulators ---- */
95 BSW_AlertLeft = 0;
96 BSW_AlertRight = 0;
97}

99/*
100 * @brief Process one DSRC neighbor for BSW (receiver role)
101 *
102 * A neighbor broadcasts the front side(s) on which it sees a car. That car sits
103 * behind the neighbor on the mirrored side, so each side bit maps to our
104 * opposite rear sensor:
105 * sender LEFT (its front-left) -> check our REAR-RIGHT
106 * sender RIGHT (its front-right) -> check our REAR-LEFT
107 * The bits are tested independently, so BSW_FLAG_BOTH runs both checks.
108 */
109void BSW_voidProcessNeighbor(const Neighbor *n)
110{
111 /* Severity scales with how close the car behind us is on the mirrored side.
112 * Keep the worst across neighbors that map to the same rear sensor. */
113 if (n->bsw_flag & BSW_FLAG_LEFT)
114 {
115 uint8_t sev = BSW_u8DistToSeverity(BSW_RearRight);
116 if (sev > BSW_AlertRight) BSW_AlertRight = sev;
117 }
118 if (n->bsw_flag & BSW_FLAG_RIGHT)
119 {
120 uint8_t sev = BSW_u8DistToSeverity(BSW_RearLeft);
121 if (sev > BSW_AlertLeft) BSW_AlertLeft = sev;
122 }
123}

136/*
137 * @brief Get the receiver-side blind-spot result for THIS car. The LED/buzzer
138 * is driven elsewhere by the caller — this module only computes which
139 * side(s) have a vehicle in the blind spot.
140 * @return bit0 = LEFT, bit1 = RIGHT
141 * (0=none, 1=LEFT, 2=RIGHT, 3=both)
142 */
143uint8_t BSW_u8GetBlindSpot(void)
144{
145 return (uint8_t)((BSW_AlertLeft ? 0x01U : 0x00U) | (BSW_AlertRight ? 0x02U : 0x00U));
146}
147
148/*
149 * @brief Get the receiver-side blind-spot SEVERITY for THIS car — the worst of
150 * the two sides. Distance-graded: closer car behind → higher severity.
151 * @return 0 = safe, 1 = warning (< BSW_SIDE_THRESHOLD), 2 = critical (< BSW_SIDE_CRITICAL)
152 */
153uint8_t BSW_u8GetSeverity(void)
154{
155 return (BSW_AlertLeft > BSW_AlertRight) ? BSW_AlertLeft : BSW_AlertRight;
156}
157
158/*
159 * @brief Get the receiver-side blind-spot severity PER SIDE, packed into one
160 * byte so the Raspberry Pi can distinguish LEFT from RIGHT (the worst-of
161 * getter above collapses both sides into a single number).
162 * @return (BSW_AlertLeft & 0x3) | ((BSW_AlertRight & 0x3) << 2)
163 * bits 1:0 = LEFT severity (0/1/2), bits 3:2 = RIGHT severity (0/1/2)
164 */
165uint8_t BSW_u8GetSidesSeverity(void)
166{
167 return (uint8_t)((BSW_AlertLeft & 0x3U) | ((BSW_AlertRight & 0x3U) << 2));
168}
