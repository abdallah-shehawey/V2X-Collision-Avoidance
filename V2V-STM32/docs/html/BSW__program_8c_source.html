37/* Map a measured rear-side distance to a BSW severity level. */
38static uint8_t BSW_u8DistToSeverity(float dist)
39{
40 if (dist <= 0.0f || dist >= BSW_SIDE_THRESHOLD) return 0; /* clear / out of band */
41 if (dist < BSW_SIDE_CRITICAL) return 2; /* critical */
42 return 1; /* warning */
43}

65void BSW_voidBeginCycle(float front_left, float front_right, float rear_left, float rear_right)
66{
67 BSW_FrontLeft = front_left;
68 BSW_FrontRight = front_right;
69 BSW_RearLeft = rear_left;
70 BSW_RearRight = rear_right;
71
72 /* ---- Sender: front-side detection, one bit per side (no priority) ---- */
73 BSW_SenderFlag = BSW_FLAG_NONE;
74
75 if (front_left > 0.0f && front_left < BSW_SIDE_THRESHOLD)
76 {
77 BSW_SenderFlag |= BSW_FLAG_LEFT;
78 }
79 if (front_right > 0.0f && front_right < BSW_SIDE_THRESHOLD)
80 {
81 BSW_SenderFlag |= BSW_FLAG_RIGHT;
82 }
83
84 /* ---- Receiver: reset per-side alert accumulators ---- */
85 BSW_AlertLeft = 0;
86 BSW_AlertRight = 0;
87}

99void BSW_voidProcessNeighbor(const Neighbor *n)
100{
101 /* Severity scales with how close the car behind us is on the mirrored side.
102 * Keep the worst across neighbors that map to the same rear sensor. */
103 if (n->bsw_flag & BSW_FLAG_LEFT)
104 {
105 uint8_t sev = BSW_u8DistToSeverity(BSW_RearRight);
106 if (sev > BSW_AlertRight) BSW_AlertRight = sev;
107 }
108 if (n->bsw_flag & BSW_FLAG_RIGHT)
109 {
110 uint8_t sev = BSW_u8DistToSeverity(BSW_RearLeft);
111 if (sev > BSW_AlertLeft) BSW_AlertLeft = sev;
112 }
113}
