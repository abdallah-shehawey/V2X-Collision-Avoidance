58/* Read one AK8963 register through the MPU9250 internal I2C master.
59 * The fetched byte lands in EXT_SENS_DATA_00 once the master completes. */
60static uint8_t MPU9250_u8ReadMagReg(uint8_t reg) {
61 MPU9250_vWriteReg(I2C_SLV0_ADDR, 0x80 | AK8963_ADDR); /* AK8963 addr, read */
62 MPU9250_vWriteReg(I2C_SLV0_REG, reg);
63 MPU9250_vWriteReg(I2C_SLV0_CTRL, 0x81); /* enable, 1 byte */
64 TIM_vDelayMs(MPU9250_DELAY_TIMER, 10);
65 return MPU9250_u8ReadReg(EXT_SENS_DATA_00);

68static void MPU9250_vInitMag(void) {
69 /* SPI-only (I2C_IF_DIS) + enable internal I2C master @400kHz, no bypass */
70 MPU9250_vWriteReg(0x37, 0x00); TIM_vDelayMs(MPU9250_DELAY_TIMER, 10);
71 MPU9250_vWriteReg(0x6A, 0x30); MPU9250_vWriteReg(0x24, 0x0D);
72
73 /* ── Read factory sensitivity adjustment (ASA) from the AK8963 fuse ROM ──
74 * Sequence per datasheet: power-down -> fuse-ROM access -> read ASA ->
75 * power-down. Applied per-axis as a scale in MPU9250_enumReadData. */
76 MPU9250_vWriteReg(I2C_SLV0_ADDR, AK8963_ADDR); /* AK8963 addr, write */
77 MPU9250_vWriteReg(I2C_SLV0_REG, AK8963_CNTL); /* CNTL1 */
78 MPU9250_vWriteReg(I2C_SLV0_DO, 0x00); /* power-down */
79 MPU9250_vWriteReg(I2C_SLV0_CTRL, 0x81);
80 TIM_vDelayMs(MPU9250_DELAY_TIMER, 10);
81
82 MPU9250_vWriteReg(I2C_SLV0_DO, 0x0F); /* fuse-ROM access mode */
83 MPU9250_vWriteReg(I2C_SLV0_CTRL, 0x81);
84 TIM_vDelayMs(MPU9250_DELAY_TIMER, 10);
85
86 uint8_t asa_x = MPU9250_u8ReadMagReg(AK8963_ASAX);
87 uint8_t asa_y = MPU9250_u8ReadMagReg(AK8963_ASAY);
88 uint8_t asa_z = MPU9250_u8ReadMagReg(AK8963_ASAZ);
89 Mag_ASA[0] = ((float)asa_x - 128.0f) / 256.0f + 1.0f;
90 Mag_ASA[1] = ((float)asa_y - 128.0f) / 256.0f + 1.0f;
91 Mag_ASA[2] = ((float)asa_z - 128.0f) / 256.0f + 1.0f;
92
93 MPU9250_vWriteReg(I2C_SLV0_ADDR, AK8963_ADDR); /* back to write mode */
94 MPU9250_vWriteReg(I2C_SLV0_REG, AK8963_CNTL);
95 MPU9250_vWriteReg(I2C_SLV0_DO, 0x00); /* power-down */
96 MPU9250_vWriteReg(I2C_SLV0_CTRL, 0x81);
97 TIM_vDelayMs(MPU9250_DELAY_TIMER, 10);
98
99 /* Put the AK8963 straight into continuous mode 2 (100 Hz), 16-bit output
100 * (CNTL1 = 0x16) via the internal I2C master. */
101 MPU9250_vWriteReg(I2C_SLV0_ADDR, 0x0C); /* AK8963 I2C addr, write */
102 MPU9250_vWriteReg(I2C_SLV0_REG, 0x0A); /* CNTL1 */
103 MPU9250_vWriteReg(I2C_SLV0_DO, 0x16);
104 MPU9250_vWriteReg(I2C_SLV0_CTRL, 0x81);
105 TIM_vDelayMs(MPU9250_DELAY_TIMER, 10);
106
107 /* Stream 7 bytes (HXL..HZH + ST2) from AK8963 reg 0x03 every sample.
108 * ST2 MUST be in the read or the AK8963 latches and stops updating. */
109 MPU9250_vWriteReg(I2C_SLV0_ADDR, 0x8C); /* AK8963 I2C addr, read */
110 MPU9250_vWriteReg(I2C_SLV0_REG, 0x03); /* HXL */
111 MPU9250_vWriteReg(I2C_SLV0_CTRL, 0x87); /* enable, 7 bytes */
112}

114ErrorState_t MPU9250_enumInit(void) {
115 GPIO_PinConfig_t CS = {.Port=MPU9250_CS_PORT, .PinNum=MPU9250_CS_PIN, .Mode=GPIO_OUTPUT, .Speed=GPIO_VERY_HIGH_SPEED};
116 GPIO_enumPinInit(&CS); MPU9250_vDeselect();
117 SPI_enumInit(&MPU9250_SPI);
118
119 MPU9250_vWriteReg(0x6B, 0x80); TIM_vDelayMs(MPU9250_DELAY_TIMER, 150);
120 MPU9250_vWriteReg(0x6B, 0x01); TIM_vDelayMs(MPU9250_DELAY_TIMER, 50);
121 MPU9250_vWriteReg(0x1A, 0x03);
122
123 MPU9250_ID = MPU9250_u8ReadReg(0x75);
124 MPU9250_vInitMag();
125
126 float sum[6] = {0};
127 for(int i=0; i<400; i++) {
128 uint8_t b[14]; uint16_t v;
129 MPU9250_vSelect(); SPI_enumTrancieve(&MPU9250_SPI, 0xBB, &v);
130 for(int j=0; j<14; j++) { SPI_enumTrancieve(&MPU9250_SPI, 0, &v); b[j]=(uint8_t)v; }
131 MPU9250_vDeselect();
132 sum[0] += (int16_t)((b[0]<<8)|b[1]); sum[1] += (int16_t)((b[2]<<8)|b[3]); sum[2] += (int16_t)((b[4]<<8)|b[5]);
133 sum[3] += (int16_t)((b[8]<<8)|b[9]); sum[4] += (int16_t)((b[10]<<8)|b[11]); sum[5] += (int16_t)((b[12]<<8)|b[13]);
134 TIM_vDelayMs(MPU9250_DELAY_TIMER, 2);
135 }
136 for(int i=0; i<2; i++) Accel_Offset[i] = sum[i]/400.0f;
137 Accel_Offset[2] = (sum[2]/400.0f) - 16384.0f;
138 for(int i=0; i<3; i++) Gyro_Offset[i] = sum[i+3]/400.0f;
139
140 MPU9250_SPI.BaudRatePrescaler = SPI_BAUDRATEPRESCALER_4;
141 SPI_enumInit(&MPU9250_SPI);
142 return OK;
143}

145ErrorState_t MPU9250_enumReadData(MPU9250_Data_t *D) {
146 if (!D) return NULL_POINTER;
147 uint8_t buffer[21]; uint16_t val;
148 MPU9250_vSelect(); SPI_enumTrancieve(&MPU9250_SPI, 0xBB, &val);
149 for(int i=0; i<21; i++) { SPI_enumTrancieve(&MPU9250_SPI, 0, &val); buffer[i] = (uint8_t)val; }
150 MPU9250_vDeselect();
151 D->AccelX = ((float)((int16_t)((buffer[0]<<8)|buffer[1])) - Accel_Offset[0]) / 16384.0f;
152 D->AccelY = ((float)((int16_t)((buffer[2]<<8)|buffer[3])) - Accel_Offset[1]) / 16384.0f;
153 D->AccelZ = ((float)((int16_t)((buffer[4]<<8)|buffer[5])) - Accel_Offset[2]) / 16384.0f;
154 D->GyroX = ((float)((int16_t)((buffer[8]<<8)|buffer[9])) - Gyro_Offset[0]) / 131.0f;
155 D->GyroY = ((float)((int16_t)((buffer[10]<<8)|buffer[11])) - Gyro_Offset[1]) / 131.0f;
156 D->GyroZ = ((float)((int16_t)((buffer[12]<<8)|buffer[13])) - Gyro_Offset[2]) / 131.0f;
157 D->MagX = (float)((int16_t)((buffer[15]<<8)|buffer[14])) * Mag_ASA[0];
158 D->MagY = (float)((int16_t)((buffer[17]<<8)|buffer[16])) * Mag_ASA[1];
159 D->MagZ = (float)((int16_t)((buffer[19]<<8)|buffer[18])) * Mag_ASA[2];
160 return OK;
161}

163ErrorState_t MPU9250_enumGetAttitude(MPU9250_Data_t *D, float dt, float *P, float *R) {
164 if (!D || !P || !R) return NULL_POINTER;
165 float accP = atan2f(-D->AccelX, sqrtf(D->AccelY*D->AccelY + D->AccelZ*D->AccelZ)) * (180.0f/M_PI);
166 float accR = atan2f(D->AccelY, D->AccelZ) * (180.0f/M_PI);
167
168 /* Fast Response Filter (0.95) for better Z tracking */
169 Filtered_Pitch = 0.95f * (Filtered_Pitch + D->GyroY * dt) + 0.05f * accP;
170 Filtered_Roll = 0.95f * (Filtered_Roll + D->GyroX * dt) + 0.05f * accR;
171
172 *P = Filtered_Pitch; *R = Filtered_Roll;
173 return OK;
174}

176/* ============ Boot-time hard-iron calibration ============
177 * Collect the magnetometer field extremes while the vehicle is rotated ~360°,
178 * then freeze the hard-iron bias = midpoint of each axis' swing. Runs once at
179 * startup so the heading is repeatable afterwards (no live "moving target"). */
180ErrorState_t MPU9250_enumCalibrateMag(void)
181{
182 float minX = 1.0e9f, maxX = -1.0e9f;
183 float minY = 1.0e9f, maxY = -1.0e9f;
184 MPU9250_Data_t d;
185
186 uint16_t samples = (uint16_t)(MPU9250_MAGCAL_DURATION_MS / MPU9250_MAGCAL_SAMPLE_MS);
187 for (uint16_t i = 0; i < samples; i++)
188 {
189 if (MPU9250_enumReadData(&d) == OK)
190 {
191 if (d.MagX < minX) minX = d.MagX;
192 if (d.MagX > maxX) maxX = d.MagX;
193 if (d.MagY < minY) minY = d.MagY;
194 if (d.MagY > maxY) maxY = d.MagY;
195 }
196 TIM_vDelayMs(MPU9250_DELAY_TIMER, MPU9250_MAGCAL_SAMPLE_MS);
197 }
198
199 /* Accept only if the car actually turned (wide span on BOTH axes). */
200 if ((maxX - minX) > MPU9250_MAG_MIN_SPAN && (maxY - minY) > MPU9250_MAG_MIN_SPAN)
201 {
202 Mag_OffX = 0.5f * (maxX + minX);
203 Mag_OffY = 0.5f * (maxY + minY);
204 Heading_Init = 0; /* re-seed the fused heading from the new reference */
205 return OK;
206 }
207 return NOK; /* not enough rotation → offsets left unchanged */
208}

210ErrorState_t MPU9250_enumGetHeading(MPU9250_Data_t *D, float dt, float *h) {
211 if (!D || !h) return NULL_POINTER;
212
213 /* Absolute heading from the magnetometer with the frozen hard-iron bias
214 * removed (sensor assumed ~level — flat-arena use). */
215 float mx = D->MagX - Mag_OffX;
216 float my = D->MagY - Mag_OffY;
217 float magH = atan2f(my, mx) * (180.0f/M_PI);
218 if (magH < 0.0f) magH += 360.0f;
219
220 /* Seed on the first call so we lock onto the absolute heading immediately
221 * instead of slowly ramping in from 0. */
222 if (!Heading_Init)
223 {
224 Filtered_Heading = magH;
225 Heading_Init = 1;
226 *h = magH;
227 return OK;
228 }
229
230 /* Complementary filter: GyroZ gives the fast, motor-noise-immune yaw change;
231 * the magnetometer slowly corrects long-term gyro drift. Wrap-aware so the
232 * 360°→0° seam is handled correctly. */
233 float gyro_pred = Filtered_Heading + (MPU9250_GYROZ_SIGN * D->GyroZ * dt);
234
235 float diff = magH - gyro_pred;
236 if (diff > 180.0f) diff -= 360.0f;
237 if (diff < -180.0f) diff += 360.0f;
238
239 Filtered_Heading = gyro_pred + (1.0f - MPU9250_HEADING_ALPHA) * diff;
240
241 if (Filtered_Heading >= 360.0f) Filtered_Heading -= 360.0f;
242 if (Filtered_Heading < 0.0f) Filtered_Heading += 360.0f;
243
244 *h = Filtered_Heading;
245 return OK;
246}

248ErrorState_t MPU9250_enumGetSpeed(MPU9250_Data_t *D, float dt, float *s) {
249 if (!D || !s) return NULL_POINTER;
250
251 float total_acc = sqrtf(D->AccelX*D->AccelX + D->AccelY*D->AccelY + D->AccelZ*D->AccelZ);
252 /* Convert G to m/s^2 (1g = 9.80665 m/s^2) */
253 float linear_acc_m_s2 = (total_acc - 1.0f) * 9.80665f;
254
255 /* 1. Enhanced ZUPT Logic (Stillness Detection) */
256 /* If Gyro noise and Accel variation are below thresholds, force deceleration */
257 if (fabs(D->GyroX) < 1.5f && fabs(D->GyroY) < 1.5f && fabs(D->GyroZ) < 1.5f) {
258 if (fabs(linear_acc_m_s2) < 0.35f) {
259 /* Apply strong damping to bleed off drift speed */
260 *s *= 0.70f;
261 if (fabs(*s) < 0.05f) *s = 0.0f;
262 linear_acc_m_s2 = 0.0f;
263 }
264 }
265
266 /* 2. Deadzone to ignore sensor floor noise */
267 if (fabs(linear_acc_m_s2) < 0.20f) {
268 linear_acc_m_s2 = 0.0f;
269 }
270
271 /* 3. Trapezoidal Integration */
272 *s += linear_acc_m_s2 * dt;
273
274 /* 4. Physical Limit: Speed cannot be negative in this integration model */
275 if (*s < 0) *s = 0;
276 if (*s < 0.015f) *s = 0; /* 1.5 cm/s clamp in meters */
277
278 return OK;
279}

281ErrorState_t MPU9250_enumGetPosition(MPU9250_Data_t *D, float s, float h, float pit, float dt, MPU9250_Position_t *pos) {
282 if (!D || !pos) return NULL_POINTER;
283
284 float radP = Filtered_Pitch * (M_PI/180.0f);
285 float radR = Filtered_Roll * (M_PI/180.0f);
286
287 /* Gravity-Compensated World Acceleration in m/s^2 */
288 float az_world = (D->AccelZ * cosf(radP) * cosf(radR)) - (D->AccelX * sinf(radP)) + (D->AccelY * sinf(radR) * cosf(radP)) - 1.0f;
289 float az_m_s2 = az_world * 9.80665f;
290
291 /* Smart Vertical Stillness Detection */
292 float total_gyro = fabs(D->GyroX) + fabs(D->GyroY) + fabs(D->GyroZ);
293 if (total_gyro < 1.5f && fabs(az_m_s2) < 0.35f) {
294 az_m_s2 = 0;
295 Vert_Velocity_m_s *= 0.75f; /* Snap speed to zero when stationary */
296 }
297
298 /* Leaky Integration to fix the Negative Drifting Issue */
299 if (fabs(az_m_s2) > 0.20f) {
300 Vert_Velocity_m_s += az_m_s2 * dt;
301 }
302 Vert_Velocity_m_s *= 0.97f; /* Continuous bleed to stabilize at rest */
303
304 /* Update Distance Z in Meters */
305 pos->Z += Vert_Velocity_m_s * dt;
306
307 /* When stationary: bleed Z back toward 0 (reference point)
308 * Avoids drift accumulation — at 50ms cycles: ~5s to converge back to 0 */
309 if (total_gyro < 1.5f && fabs(Vert_Velocity_m_s) < 0.01f) {
310 pos->Z *= 0.98f;
311 if (fabs(pos->Z) < 0.005f) pos->Z = 0.0f; /* snap at 5mm */
312 }
313
314 pos->X = 0; pos->Y = 0;
315 return OK;
316}
