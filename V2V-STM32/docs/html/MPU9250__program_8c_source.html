86/* Read one AK8963 register through the MPU9250 internal I2C master.
87 * The fetched byte lands in EXT_SENS_DATA_00 once the master completes. */
96static uint8_t MPU9250_u8ReadMagReg(uint8_t reg) {
97 MPU9250_vWriteReg(I2C_SLV0_ADDR, 0x80 | AK8963_ADDR); /* AK8963 addr, read */
98 MPU9250_vWriteReg(I2C_SLV0_REG, reg);
99 MPU9250_vWriteReg(I2C_SLV0_CTRL, 0x81); /* enable, 1 byte */
100 TIM_vDelayMs(MPU9250_DELAY_TIMER, 10);
101 return MPU9250_u8ReadReg(EXT_SENS_DATA_00);

110static void MPU9250_vInitMag(void) {
111 /* SPI-only (I2C_IF_DIS) + enable internal I2C master @400kHz, no bypass */
112 MPU9250_vWriteReg(0x37, 0x00); TIM_vDelayMs(MPU9250_DELAY_TIMER, 10);
113 MPU9250_vWriteReg(0x6A, 0x30); MPU9250_vWriteReg(0x24, 0x0D);
114
115 /* ── Read factory sensitivity adjustment (ASA) from the AK8963 fuse ROM ──
116 * Sequence per datasheet: power-down -> fuse-ROM access -> read ASA ->
117 * power-down. Applied per-axis as a scale in MPU9250_enumReadData. */
118 MPU9250_vWriteReg(I2C_SLV0_ADDR, AK8963_ADDR); /* AK8963 addr, write */
119 MPU9250_vWriteReg(I2C_SLV0_REG, AK8963_CNTL); /* CNTL1 */
120 MPU9250_vWriteReg(I2C_SLV0_DO, 0x00); /* power-down */
121 MPU9250_vWriteReg(I2C_SLV0_CTRL, 0x81);
122 TIM_vDelayMs(MPU9250_DELAY_TIMER, 10);
123
124 MPU9250_vWriteReg(I2C_SLV0_DO, 0x0F); /* fuse-ROM access mode */
125 MPU9250_vWriteReg(I2C_SLV0_CTRL, 0x81);
126 TIM_vDelayMs(MPU9250_DELAY_TIMER, 10);
127
128 uint8_t asa_x = MPU9250_u8ReadMagReg(AK8963_ASAX);
129 uint8_t asa_y = MPU9250_u8ReadMagReg(AK8963_ASAY);
130 uint8_t asa_z = MPU9250_u8ReadMagReg(AK8963_ASAZ);
131 Mag_ASA[0] = ((float)asa_x - 128.0f) / 256.0f + 1.0f;
132 Mag_ASA[1] = ((float)asa_y - 128.0f) / 256.0f + 1.0f;
133 Mag_ASA[2] = ((float)asa_z - 128.0f) / 256.0f + 1.0f;
134
135 MPU9250_vWriteReg(I2C_SLV0_ADDR, AK8963_ADDR); /* back to write mode */
136 MPU9250_vWriteReg(I2C_SLV0_REG, AK8963_CNTL);
137 MPU9250_vWriteReg(I2C_SLV0_DO, 0x00); /* power-down */
138 MPU9250_vWriteReg(I2C_SLV0_CTRL, 0x81);
139 TIM_vDelayMs(MPU9250_DELAY_TIMER, 10);
140
141 /* Put the AK8963 straight into continuous mode 2 (100 Hz), 16-bit output
142 * (CNTL1 = 0x16) via the internal I2C master. */
143 MPU9250_vWriteReg(I2C_SLV0_ADDR, 0x0C); /* AK8963 I2C addr, write */
144 MPU9250_vWriteReg(I2C_SLV0_REG, 0x0A); /* CNTL1 */
145 MPU9250_vWriteReg(I2C_SLV0_DO, 0x16);
146 MPU9250_vWriteReg(I2C_SLV0_CTRL, 0x81);
147 TIM_vDelayMs(MPU9250_DELAY_TIMER, 10);
148
149 /* Stream 7 bytes (HXL..HZH + ST2) from AK8963 reg 0x03 every sample.
150 * ST2 MUST be in the read or the AK8963 latches and stops updating. */
151 MPU9250_vWriteReg(I2C_SLV0_ADDR, 0x8C); /* AK8963 I2C addr, read */
152 MPU9250_vWriteReg(I2C_SLV0_REG, 0x03); /* HXL */
153 MPU9250_vWriteReg(I2C_SLV0_CTRL, 0x87); /* enable, 7 bytes */
154}

156ErrorState_t MPU9250_enumInit(void) {
157 GPIO_PinConfig_t CS = {.Port=MPU9250_CS_PORT, .PinNum=MPU9250_CS_PIN, .Mode=GPIO_OUTPUT, .Speed=GPIO_VERY_HIGH_SPEED};
158 GPIO_enumPinInit(&CS); MPU9250_vDeselect();
159 SPI_enumInit(&MPU9250_SPI);
160
161 MPU9250_vWriteReg(0x6B, 0x80); TIM_vDelayMs(MPU9250_DELAY_TIMER, 150);
162 MPU9250_vWriteReg(0x6B, 0x01); TIM_vDelayMs(MPU9250_DELAY_TIMER, 50);
163 MPU9250_vWriteReg(0x1A, 0x03);
164
165 MPU9250_ID = MPU9250_u8ReadReg(0x75);
166 MPU9250_vInitMag();
167
168 float sum[6] = {0};
169 for(int i=0; i<400; i++) {
170 uint8_t b[14]; uint16_t v;
171 MPU9250_vSelect(); SPI_enumTrancieve(&MPU9250_SPI, 0xBB, &v);
172 for(int j=0; j<14; j++) { SPI_enumTrancieve(&MPU9250_SPI, 0, &v); b[j]=(uint8_t)v; }
173 MPU9250_vDeselect();
174 sum[0] += (int16_t)((b[0]<<8)|b[1]); sum[1] += (int16_t)((b[2]<<8)|b[3]); sum[2] += (int16_t)((b[4]<<8)|b[5]);
175 sum[3] += (int16_t)((b[8]<<8)|b[9]); sum[4] += (int16_t)((b[10]<<8)|b[11]); sum[5] += (int16_t)((b[12]<<8)|b[13]);
176 TIM_vDelayMs(MPU9250_DELAY_TIMER, 2);
177 }
178 for(int i=0; i<2; i++) Accel_Offset[i] = sum[i]/400.0f;
179 Accel_Offset[2] = (sum[2]/400.0f) - 16384.0f;
180 for(int i=0; i<3; i++) Gyro_Offset[i] = sum[i+3]/400.0f;
181
182 MPU9250_SPI.BaudRatePrescaler = SPI_BAUDRATEPRESCALER_4;
183 SPI_enumInit(&MPU9250_SPI);
184 return OK;
185}

187ErrorState_t MPU9250_enumReadData(MPU9250_Data_t *D) {
188 if (!D) return NULL_POINTER;
189 uint8_t buffer[21]; uint16_t val;
190 MPU9250_vSelect(); SPI_enumTrancieve(&MPU9250_SPI, 0xBB, &val);
191 for(int i=0; i<21; i++) { SPI_enumTrancieve(&MPU9250_SPI, 0, &val); buffer[i] = (uint8_t)val; }
192 MPU9250_vDeselect();
193 D->AccelX = ((float)((int16_t)((buffer[0]<<8)|buffer[1])) - Accel_Offset[0]) / 16384.0f;
194 D->AccelY = ((float)((int16_t)((buffer[2]<<8)|buffer[3])) - Accel_Offset[1]) / 16384.0f;
195 D->AccelZ = ((float)((int16_t)((buffer[4]<<8)|buffer[5])) - Accel_Offset[2]) / 16384.0f;
196 D->GyroX = ((float)((int16_t)((buffer[8]<<8)|buffer[9])) - Gyro_Offset[0]) / 131.0f;
197 D->GyroY = ((float)((int16_t)((buffer[10]<<8)|buffer[11])) - Gyro_Offset[1]) / 131.0f;
198 D->GyroZ = ((float)((int16_t)((buffer[12]<<8)|buffer[13])) - Gyro_Offset[2]) / 131.0f;
199 D->MagX = (float)((int16_t)((buffer[15]<<8)|buffer[14])) * Mag_ASA[0];
200 D->MagY = (float)((int16_t)((buffer[17]<<8)|buffer[16])) * Mag_ASA[1];
201 D->MagZ = (float)((int16_t)((buffer[19]<<8)|buffer[18])) * Mag_ASA[2];
202 return OK;
203}

205ErrorState_t MPU9250_enumGetAttitude(MPU9250_Data_t *D, float dt, float *P, float *R) {
206 if (!D || !P || !R) return NULL_POINTER;
207 float accP = atan2f(-D->AccelX, sqrtf(D->AccelY*D->AccelY + D->AccelZ*D->AccelZ)) * (180.0f/M_PI);
208 float accR = atan2f(D->AccelY, D->AccelZ) * (180.0f/M_PI);
209
210 /* Fast Response Filter (0.95) for better Z tracking */
211 Filtered_Pitch = 0.95f * (Filtered_Pitch + D->GyroY * dt) + 0.05f * accP;
212 Filtered_Roll = 0.95f * (Filtered_Roll + D->GyroX * dt) + 0.05f * accR;
213
214 *P = Filtered_Pitch; *R = Filtered_Roll;
215 return OK;
216}

218/* ============ Boot-time hard-iron calibration ============
219 * Collect the magnetometer field extremes while the vehicle is rotated ~360°,
220 * then freeze the hard-iron bias = midpoint of each axis' swing. Runs once at
221 * startup so the heading is repeatable afterwards (no live "moving target"). */
222ErrorState_t MPU9250_enumCalibrateMag(void)
223{
224 float minX = 1.0e9f, maxX = -1.0e9f;
225 float minY = 1.0e9f, maxY = -1.0e9f;
226 MPU9250_Data_t d;
227
228 uint16_t samples = (uint16_t)(MPU9250_MAGCAL_DURATION_MS / MPU9250_MAGCAL_SAMPLE_MS);
229 for (uint16_t i = 0; i < samples; i++)
230 {
231 if (MPU9250_enumReadData(&d) == OK)
232 {
233 if (d.MagX < minX) minX = d.MagX;
234 if (d.MagX > maxX) maxX = d.MagX;
235 if (d.MagY < minY) minY = d.MagY;
236 if (d.MagY > maxY) maxY = d.MagY;
237 }
238 TIM_vDelayMs(MPU9250_DELAY_TIMER, MPU9250_MAGCAL_SAMPLE_MS);
239 }
240
241 /* Accept only if the car actually turned (wide span on BOTH axes). */
242 if ((maxX - minX) > MPU9250_MAG_MIN_SPAN && (maxY - minY) > MPU9250_MAG_MIN_SPAN)
243 {
244 Mag_OffX = 0.5f * (maxX + minX);
245 Mag_OffY = 0.5f * (maxY + minY);
246 Heading_Init = 0; /* re-seed the fused heading from the new reference */
247 return OK;
248 }
249 return NOK; /* not enough rotation → offsets left unchanged */
250}

252ErrorState_t MPU9250_enumGetHeading(MPU9250_Data_t *D, float dt, float *h) {
253 if (!D || !h) return NULL_POINTER;
254
255 /* Absolute heading from the magnetometer with the frozen hard-iron bias
256 * removed (sensor assumed ~level — flat-arena use). */
257 float mx = D->MagX - Mag_OffX;
258 float my = D->MagY - Mag_OffY;
259 float magH = atan2f(my, mx) * (180.0f/M_PI);
260 if (magH < 0.0f) magH += 360.0f;
261
262 /* Seed on the first call so we lock onto the absolute heading immediately
263 * instead of slowly ramping in from 0. */
264 if (!Heading_Init)
265 {
266 Filtered_Heading = magH;
267 Heading_Init = 1;
268 *h = magH;
269 return OK;
270 }
271
272 /* Complementary filter: GyroZ gives the fast, motor-noise-immune yaw change;
273 * the magnetometer slowly corrects long-term gyro drift. Wrap-aware so the
274 * 360°→0° seam is handled correctly. */
275 float gyro_pred = Filtered_Heading + (MPU9250_GYROZ_SIGN * D->GyroZ * dt);
276
277 float diff = magH - gyro_pred;
278 if (diff > 180.0f) diff -= 360.0f;
279 if (diff < -180.0f) diff += 360.0f;
280
281 Filtered_Heading = gyro_pred + (1.0f - MPU9250_HEADING_ALPHA) * diff;
282
283 if (Filtered_Heading >= 360.0f) Filtered_Heading -= 360.0f;
284 if (Filtered_Heading < 0.0f) Filtered_Heading += 360.0f;
285
286 *h = Filtered_Heading;
287 return OK;
288}

290ErrorState_t MPU9250_enumGetSpeed(MPU9250_Data_t *D, float dt, float *s) {
291 if (!D || !s) return NULL_POINTER;
292
293 float total_acc = sqrtf(D->AccelX*D->AccelX + D->AccelY*D->AccelY + D->AccelZ*D->AccelZ);
294 /* Convert G to m/s^2 (1g = 9.80665 m/s^2) */
295 float linear_acc_m_s2 = (total_acc - 1.0f) * 9.80665f;
296
297 /* 1. Enhanced ZUPT Logic (Stillness Detection) */
298 /* If Gyro noise and Accel variation are below thresholds, force deceleration */
299 if (fabs(D->GyroX) < 1.5f && fabs(D->GyroY) < 1.5f && fabs(D->GyroZ) < 1.5f) {
300 if (fabs(linear_acc_m_s2) < 0.35f) {
301 /* Apply strong damping to bleed off drift speed */
302 *s *= 0.70f;
303 if (fabs(*s) < 0.05f) *s = 0.0f;
304 linear_acc_m_s2 = 0.0f;
305 }
306 }
307
308 /* 2. Deadzone to ignore sensor floor noise */
309 if (fabs(linear_acc_m_s2) < 0.20f) {
310 linear_acc_m_s2 = 0.0f;
311 }
312
313 /* 3. Trapezoidal Integration */
314 *s += linear_acc_m_s2 * dt;
315
316 /* 4. Physical Limit: Speed cannot be negative in this integration model */
317 if (*s < 0) *s = 0;
318 if (*s < 0.015f) *s = 0; /* 1.5 cm/s clamp in meters */
319
320 return OK;
321}

323ErrorState_t MPU9250_enumGetPosition(MPU9250_Data_t *D, float s, float h, float pit, float dt, MPU9250_Position_t *pos) {
324 if (!D || !pos) return NULL_POINTER;
325
326 float radP = Filtered_Pitch * (M_PI/180.0f);
327 float radR = Filtered_Roll * (M_PI/180.0f);
328
329 /* Gravity-Compensated World Acceleration in m/s^2 */
330 float az_world = (D->AccelZ * cosf(radP) * cosf(radR)) - (D->AccelX * sinf(radP)) + (D->AccelY * sinf(radR) * cosf(radP)) - 1.0f;
331 float az_m_s2 = az_world * 9.80665f;
332
333 /* Smart Vertical Stillness Detection */
334 float total_gyro = fabs(D->GyroX) + fabs(D->GyroY) + fabs(D->GyroZ);
335 if (total_gyro < 1.5f && fabs(az_m_s2) < 0.35f) {
336 az_m_s2 = 0;
337 Vert_Velocity_m_s *= 0.75f; /* Snap speed to zero when stationary */
338 }
339
340 /* Leaky Integration to fix the Negative Drifting Issue */
341 if (fabs(az_m_s2) > 0.20f) {
342 Vert_Velocity_m_s += az_m_s2 * dt;
343 }
344 Vert_Velocity_m_s *= 0.97f; /* Continuous bleed to stabilize at rest */
345
346 /* Update Distance Z in Meters */
347 pos->Z += Vert_Velocity_m_s * dt;
348
349 /* When stationary: bleed Z back toward 0 (reference point)
350 * Avoids drift accumulation — at 50ms cycles: ~5s to converge back to 0 */
351 if (total_gyro < 1.5f && fabs(Vert_Velocity_m_s) < 0.01f) {
352 pos->Z *= 0.98f;
353 if (fabs(pos->Z) < 0.005f) pos->Z = 0.0f; /* snap at 5mm */
354 }
355
356 pos->X = 0; pos->Y = 0;
357 return OK;
358}
