47void ITM_SendChar(uint8_t ch)
48{
49
50 //Enable TRCENA
51 DEMCR |= ( 1 << 24);
52
53 //enable stimulus port 0
54 ITM_TRACE_EN |= ( 1 << 0);
55
56 // read FIFO status in bit [0]:
57 while(!(ITM_STIMULUS_PORT0 & 1));
58
59 //Write to ITM stimulus port0
60 ITM_STIMULUS_PORT0 = ch;
61}
