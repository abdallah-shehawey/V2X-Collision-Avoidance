72void ITM_SendChar(uint8_t ch)
73{
74
75 //Enable TRCENA
76 DEMCR |= ( 1 << 24);
77
78 //enable stimulus port 0
79 ITM_TRACE_EN |= ( 1 << 0);
80
81 // read FIFO status in bit [0]:
82 while(!(ITM_STIMULUS_PORT0 & 1));
83
84 //Write to ITM stimulus port0
85 ITM_STIMULUS_PORT0 = ch;
86}
