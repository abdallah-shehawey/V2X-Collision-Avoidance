56static void queue_push(Neighbor *n)
57{
58 uint8_t next = (queue_tail + 1) % QUEUE_SIZE;

60 {
61 rx_queue[queue_tail] = *n;

66static uint8_t queue_pop(Neighbor *out)
67{
68 if (queue_head == queue_tail)
69 {
70 // reset to zero when empty
71 queue_head = 0;
72 queue_tail = 0;
73 return 0; // empty
74 }
75 *out = rx_queue[queue_head];
76 queue_head = (queue_head + 1) % QUEUE_SIZE;
77 return 1;
78}

83static void update_neighbor(const Neighbor *msg)
84{
85 /* Stamp with LOCAL FreeRTOS tick so DSRC_RemoveStale compares apples to apples.
86 * The sender's last_update is from their own clock and is meaningless here. */
87 uint32_t local_time = (uint32_t)xTaskGetTickCount();
88
89 // if exists → update only
90 for (uint8_t i = 0; i < neighbor_count; i++)
91 {
92 if (neighbor_table[i].vehicle_id == msg->vehicle_id)
93 {
94 neighbor_table[i] = *msg;
95 neighbor_table[i].last_update = local_time;
96 return;
97 }
98 }
99
100 // not found → add
101 if (neighbor_count < MAX_NEIGHBORS)
102 {
103 neighbor_table[neighbor_count] = *msg;
104 neighbor_table[neighbor_count].last_update = local_time;
105 neighbor_count++;
106 return;
107 }
108
109 // table full → replace oldest
110 uint8_t oldest_i = 0;
111 uint32_t oldest_time = neighbor_table[0].last_update;
112 for (uint8_t i = 1; i < neighbor_count; i++)
113 {
114 if (neighbor_table[i].last_update < oldest_time)
115 {
116 oldest_time = neighbor_table[i].last_update;
117 oldest_i = i;
118 }
119 }
120 neighbor_table[oldest_i] = *msg;
121 neighbor_table[oldest_i].last_update = local_time;
122}

136void DSRC_SendNeighbor(Neighbor *n)
137{

205// Feed one received byte into the parser state machine.
206// NOTE: call from task context (vTask_ESP_Comm), NOT from the ISR —
207// queue_push() below mutates the non-atomic rx_queue.
208void DSRC_RxCallback(uint8_t byte)
209{
210 switch (parse_state)
211 {
212 case WAIT_START:
213 if (byte == START_BYTE)
214 {
215 rx_idx = 0;
216 parse_state = READ_DATA;
217 }
218 break;
219
220 case READ_DATA:
221 rx_buf[rx_idx++] = byte;
222 if (rx_idx >= sizeof(Neighbor))
223 {
224 parse_state = READ_CHECKSUM;
225 }
226 break;
227
228 case READ_CHECKSUM:
229 rx_checksum = byte;
230 parse_state = READ_END;
231 break;
232
233 case READ_END:
234 if (byte == END_BYTE)
235 {
236 uint8_t expected = calc_checksum(rx_buf, sizeof(Neighbor));
237 if (expected == rx_checksum)
238 {
239 Neighbor n;
240 memcpy(&n, rx_buf, sizeof(Neighbor));
241 // or you can make casting if you prefer for example : queue_push((Neighbor*)rx_buf);
242 queue_push(&n);
243 }
244 }
245 parse_state = WAIT_START;
246 break;
247 }
248}
