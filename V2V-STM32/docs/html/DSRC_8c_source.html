104static void queue_push(Neighbor *n)
105{
106 uint8_t next = (queue_tail + 1) % QUEUE_SIZE;
107 if (next != queue_head) // not full
108 {
109 rx_queue[queue_tail] = *n;
110 queue_tail = next;
111 }
112}

120static uint8_t queue_pop(Neighbor *out)
121{
122 if (queue_head == queue_tail)
123 {
124 // reset to zero when empty
125 queue_head = 0;
126 queue_tail = 0;
127 return 0; // empty
128 }
129 *out = rx_queue[queue_head];
130 queue_head = (queue_head + 1) % QUEUE_SIZE;
131 return 1;
132}

151static void update_neighbor(const Neighbor *msg)
152{
153 /* Stamp with LOCAL FreeRTOS tick so DSRC_RemoveStale compares apples to apples.
154 * The sender's last_update is from their own clock and is meaningless here. */
155 uint32_t local_time = (uint32_t)xTaskGetTickCount();
156
157 // if exists → update only
158 for (uint8_t i = 0; i < neighbor_count; i++)
159 {
160 if (neighbor_table[i].vehicle_id == msg->vehicle_id)
161 {
162 neighbor_table[i] = *msg;
163 neighbor_table[i].last_update = local_time;
164 return;
165 }
166 }
167
168 // not found → add
169 if (neighbor_count < MAX_NEIGHBORS)
170 {
171 neighbor_table[neighbor_count] = *msg;
172 neighbor_table[neighbor_count].last_update = local_time;
173 neighbor_count++;
174 return;
175 }
176
177 // table full → replace oldest
178 uint8_t oldest_i = 0;
179 uint32_t oldest_time = neighbor_table[0].last_update;
180 for (uint8_t i = 1; i < neighbor_count; i++)
181 {
182 if (neighbor_table[i].last_update < oldest_time)
183 {
184 oldest_time = neighbor_table[i].last_update;
185 oldest_i = i;
186 }
187 }
188 neighbor_table[oldest_i] = *msg;
189 neighbor_table[oldest_i].last_update = local_time;
190}

204void DSRC_SendNeighbor(Neighbor *n)
205{
206 uint8_t raw[sizeof(Neighbor)];
207 memcpy(raw, n, sizeof(Neighbor));
208 uint8_t chk = calc_checksum(raw, sizeof(Neighbor));
209
210 USART_enumTransmit((USART_Config_t*)&USART_1, START_BYTE);
211 for (uint8_t i = 0; i < sizeof(Neighbor); i++)
212 {
213 USART_enumTransmit((USART_Config_t*)&USART_1, raw[i]);
214 }
215 USART_enumTransmit((USART_Config_t*)&USART_1, chk);
216 USART_enumTransmit((USART_Config_t*)&USART_1, END_BYTE);
217}

230{

273// Feed one received byte into the parser state machine.
274// NOTE: call from task context (vTask_ESP_Comm), NOT from the ISR —
275// queue_push() below mutates the non-atomic rx_queue.
276void DSRC_RxCallback(uint8_t byte)
277{
278 switch (parse_state)
279 {
280 case WAIT_START:
281 if (byte == START_BYTE)
282 {
283 rx_idx = 0;
284 parse_state = READ_DATA;
285 }
286 break;
287
288 case READ_DATA:
289 rx_buf[rx_idx++] = byte;
290 if (rx_idx >= sizeof(Neighbor))
291 {
292 parse_state = READ_CHECKSUM;
293 }
294 break;
295
296 case READ_CHECKSUM:
297 rx_checksum = byte;
298 parse_state = READ_END;
299 break;
300
301 case READ_END:
302 if (byte == END_BYTE)
303 {
304 uint8_t expected = calc_checksum(rx_buf, sizeof(Neighbor));
305 if (expected == rx_checksum)
306 {
307 Neighbor n;
308 memcpy(&n, rx_buf, sizeof(Neighbor));
309 // or you can make casting if you prefer for example : queue_push((Neighbor*)rx_buf);
310 queue_push(&n);
311 }
312 }
313 parse_state = WAIT_START;
314 break;
315 }
316}
