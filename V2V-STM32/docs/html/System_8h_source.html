62/*
63 * ========================================================================================
64 * 🧠 RTOS TASKS ARCHITECTURE & PRIORITIES (configMAX_PRIORITIES = 5)
65 * ========================================================================================
66 *
67 * | Task Name | Priority | Stack (+base) | Freq | Description |
68 * |------------------------|----------|---------------|--------------------|-----------------------------------------------|
69 * | [1] vTask_SafetyEngine | 4 (MAX) | +256 | ~50 ms | Single-pass ADAS brain (FCW/EEBL/BSW/DNPW/IMA)|
70 * | | | | | + feedback aggregation → command channel |
71 * | [1] vTask_ESP_Comm | 4 (MAX) | +128 | ~10ms RX/100ms TX | ESP-NOW V2X communication |
72 * | [2] vTask_Sensors | 3 | +256 | ~25-82 ms adaptive | All 6 US (interrupt, 2m cap) + MPU9250 |
73 * | [3] vTask_Feedback | 2 | +128 | ~25 ms | Alert manager (LEDs + Buzzer; motors on RPi) |
74 * | [4] vTask_RPi_Comm | 1 (LOW) | +100 | ~100 ms | Raspberry Pi telemetry TX (status + sensors) |
75 *
76 * NOTE: Priority 4 is the highest user priority. Priority 0 is the FreeRTOS Idle task.
77 * configMAX_SYSCALL_INTERRUPT_PRIORITY = 5 → NVIC_USART1 must be set to ≥ 6.
78 * vTask_Sensors reads all 6 US per cycle (interrupt-driven: the task SLEEPS
79 * during each echo, CPU free) + MPU, then a 10ms gap. Scan is adaptive:
80 * near objects → ~25ms, all out-of-range → ~72ms (2m cap). Reads are
81 * sequential → no acoustic cross-talk. Priority 3 (producer) sits below the
82 * priority-4 ADAS/comm tasks but above the priority-2 Feedback consumer.
83 *
84 * ADAS architecture = SINGLE-PASS, with a clean Brain/Muscle split:
85 * • vTask_SafetyEngine (Brain, detection-only): runs all modules over the
86 * neighbor table in ONE pass; each module reports its risk level. Then it
87 * publishes G_u16SystemFlags (2 bits/module: 00 safe / 01 warning / 10 crit).
88 * It makes NO actuator decision. Holds both mutexes (NeighborTable → Data).
89 * • vTask_Feedback (Muscle): reads G_u16SystemFlags; if 0 → everything off.
90 * Any alert → interior LED + buzzer ON. Additionally: FCW CRITICAL → front
91 * LEDs, EEBL CRITICAL → rear LEDs. All other states add no external LED.
92 * NO motor control — the Raspberry Pi drives the motors from the telemetry.
93 * Lock usage: ESP_Comm takes the two mutexes separately, Sensors & Feedback take
94 * Data only → deadlock-free.
95 * ========================================================================================
96 */

117/* ════════════════════════════════════════════════════════════════════════
118 * G_u16SystemFlags — system status word (2 bits per ADAS module)
119 * ════════════════════════════════════════════════════════════════════════
120 * SafetyEngine writes it; vTask_Feedback and vTask_RPi_Comm read it.
121 * Each module occupies 2 bits:
122 * 0b00 = SAFE (no hazard)
123 * 0b01 = WARNING (caution — slow down)
124 * 0b10 = CRITICAL (danger — stop)
125 *
126 * Bit layout (uint16_t):
127 * bits 1:0 → FCW (Forward Collision Warning)
128 * bits 3:2 → EEBL (Emergency Electronic Brake Light)
129 * bits 5:4 → BSW (Blind Spot Warning) [WARNING only]
130 * bits 7:6 → DNPW (Do Not Pass Warning)
131 * bits 9:8 → IMA (Intersection Movement Assist)
132 * bits 15:10 → reserved (0)
133 *
134 * 0x0000 = everything safe.
135 * Extract one module: status = (G_u16SystemFlags >> SYS_xxx_POS) & SYS_MASK
136 *
137 * Worked examples (binary grouped as IMA|DNPW|BSW|EEBL|FCW):
138 * 0x0001 00 00 00 00 01 → FCW WARNING
139 * 0x0002 00 00 00 00 10 → FCW CRITICAL
140 * 0x0004 00 00 00 01 00 → EEBL WARNING
141 * 0x0010 00 00 01 00 00 → BSW WARNING
142 * 0x0040 00 01 00 00 00 → DNPW WARNING
143 * 0x0100 01 00 00 00 00 → IMA WARNING
144 * 0x0006 00 00 00 01 10 → FCW CRITICAL + EEBL WARNING
145 * 0x0101 01 00 00 00 01 → FCW WARNING + IMA WARNING
146 * ════════════════════════════════════════════════════════════════════════ */
147#define SYS_SAFE 0x0u
148#define SYS_WARNING 0x1u
149#define SYS_CRITICAL 0x2u
150#define SYS_MASK 0x3u /* 2-bit field mask */
151
152#define SYS_FCW_POS 0u
153#define SYS_EEBL_POS 2u
154#define SYS_BSW_POS 4u
155#define SYS_DNPW_POS 6u
156#define SYS_IMA_POS 8u
157
158/* status (0/1/2) of one module from the packed word */
159#define SYS_GET(flags, pos) (((flags) >> (pos)) & SYS_MASK)

161/* ── RPi telemetry ──
162 * Sent every 100ms by vTask_RPi_Comm as an ASCII, '\n'-delimited CSV line (the
163 * old binary 0xAA..0x55 packet was dropped because its zero-bytes were lost on
164 * the UART link). The RPi parser (server.py PACKET_FIELDS) must match this
165 * column order EXACTLY — change one side, change the other.
166 *
167 * T,speed,heading,pitch,roll,FL,FC,FR,BL,BC,BR,flags,bsw_sides\n
168 *
169 * speed cm/s
170 * heading degrees 0-360
171 * pitch, roll degrees
172 * FL..BR 6 ultrasonic distances [cm]
173 * flags G_u16SystemFlags: 2 bits/module (00 safe/01 warn/10 crit)
174 * bsw_sides per-side BSW severity: bits 1:0 LEFT, bits 3:2 RIGHT
175 * (lets the RPi report left / right / both — the
176 * aggregated BSW severity is already inside `flags`)
177 *
178 * The struct below is the legacy binary layout, kept for reference only. */
179typedef struct __attribute__((packed))
180{
181 uint8_t start; /* 0xAA */
182 uint16_t sys_flags; /* G_u16SystemFlags: 2 bits/module (00/01/10) */
183 float speed; /* cm/s */
184 float heading; /* degrees 0-360 */
185 float front_left; /* ultrasonic distance [cm] */
186 float front_center;
187 float front_right;
188 float back_left;
189 float back_center;
190 float back_right;
191 uint8_t end; /* 0x55 */
192} RPi_Packet_t;
