45/* Exactly ONE measurement is active at a time (sequential reads), so a single
46 * shared context + one binary semaphore is race-free. All fields are touched by
47 * both the reading task and the ISR → volatile. */
48static volatile struct
49{
50 TIM_Num_t timer;
51 uint8_t channel; /* 0..3 */
52 uint32_t maxval; /* timer wrap value (0xFFFF or 0xFFFFFFFF) */
53 uint32_t t1; /* rising-edge timestamp */
54 US_Phase_t phase;
55 uint8_t valid; /* 0 = result is out-of-range/garbage, 1 = a real echo was timed */
56 uint16_t dist_cm; /* result, valid when phase == US_PHASE_DONE && valid */
57} US_Active;

69static void US_vSendTrigger(const US_Config_t *pxSensor)
70{
71 /* Settle LOW for a clean edge, then a 10us HIGH pulse (HC-SR04 datasheet). */
72 GPIO_enumWritePinVal(pxSensor->TrigPort, pxSensor->TrigPin, GPIO_PIN_LOW);
73 TIM_vDelayUs(TIM_TIMER6, US_TRIG_SETTLE_US);
74
75 GPIO_enumWritePinVal(pxSensor->TrigPort, pxSensor->TrigPin, GPIO_PIN_HIGH);
76 TIM_vDelayUs(TIM_TIMER6, US_TRIG_PULSE_US);
77
78 GPIO_enumWritePinVal(pxSensor->TrigPort, pxSensor->TrigPin, GPIO_PIN_LOW);
79}

99/* Input-Capture ISR callback (timer IRQ context). Runs the 2-edge state machine
100 * for the single active measurement, then wakes the reading task. */
101static void US_CC_Handler(TIM_Num_t Copy_eTimer, uint8_t Copy_u8Channel, uint32_t Copy_u32Capture)
102{
103 /* Ignore captures that don't belong to the active measurement */
104 if (Copy_eTimer != US_Active.timer || Copy_u8Channel != US_Active.channel)
105 {
106 return;
107 }
108
109 if (US_Active.phase == US_PHASE_RISING)
110 {
111 US_Active.t1 = Copy_u32Capture;
112 US_Active.phase = US_PHASE_FALLING;
113 /* Now look for the falling edge on the same channel */
114 TIM_vSetICPolarity(Copy_eTimer, (TIM_Channel_t)Copy_u8Channel, TIM_POLARITY_LOW);
115 }
116 else if (US_Active.phase == US_PHASE_FALLING)
117 {
118 uint32_t high = (Copy_u32Capture >= US_Active.t1)
119 ? (Copy_u32Capture - US_Active.t1)
120 : ((US_Active.maxval - US_Active.t1) + Copy_u32Capture + 1u);
121
122 /* Decode µs → cm. An echo longer than the sensor's useful range means
123 * "no object within range": report it as OUT-OF-RANGE (valid = 0) instead
124 * of silently clamping a bogus long/wrapped pulse to a fake 400cm reading
125 * — clamping is what made a lost echo masquerade as a solid 400cm fix and
126 * made the reading flap between a real distance and 400. */
127 uint32_t dist = high / US_SOUND_SPEED_FACTOR;
128 if (dist == 0u || dist > US_MAX_RANGE_CM)
129 {
130 US_Active.valid = 0u; /* out of range / spurious */
131 US_Active.dist_cm = US_MAX_RANGE_CM;
132 }
133 else
134 {
135 US_Active.valid = 1u; /* a real, in-range echo */
136 US_Active.dist_cm = (uint16_t)dist;
137 }
138 US_Active.phase = US_PHASE_DONE;
139
140 /* Done — silence this channel and wake the task */
141 TIM_vDisableCCInterrupt(Copy_eTimer, (TIM_Channel_t)Copy_u8Channel);
142
143 BaseType_t xHPW = pdFALSE;
144 xSemaphoreGiveFromISR(US_xDoneSem, &xHPW);
145 portYIELD_FROM_ISR(xHPW);
146 }
147}

152ErrorState_t US_vInit(const US_Config_t *pxSensor)
153{
154 if (pxSensor == NULL)
155 return NULL_POINTER;
156 if (pxSensor->Timer >= TIM_TIMER6)
157 return NOK; /* basic timers have no IC */
158
159 /* 1. Initialize GPIO Pins */
160 GPIO_PinConfig_t TrigCfg = {
161 .Port = pxSensor->TrigPort, .PinNum = pxSensor->TrigPin, .Mode = GPIO_OUTPUT, .Otype = GPIO_PUSH_PULL, .Speed = GPIO_MEDIUM_SPEED, .PullType = GPIO_NO_PULL};
162 GPIO_enumPinInit(&TrigCfg);
163
164 GPIO_PinConfig_t EchoCfg = {
165 .Port = pxSensor->EchoPort, .PinNum = pxSensor->EchoPin, .Mode = GPIO_ALTFN, .Otype = GPIO_PUSH_PULL, .Speed = GPIO_VERY_HIGH_SPEED, .PullType = GPIO_NO_PULL};
166
167 if (pxSensor->Timer == TIM_TIMER1 || pxSensor->Timer == TIM_TIMER2)
168 EchoCfg.AlternateFunction = GPIO_AF1;
169 else if (pxSensor->Timer >= TIM_TIMER3 && pxSensor->Timer <= TIM_TIMER5)
170 EchoCfg.AlternateFunction = GPIO_AF2;
171 else if (pxSensor->Timer == TIM_TIMER8)
172 EchoCfg.AlternateFunction = GPIO_AF3;
173 else
174 return NOK;
175 GPIO_enumPinInit(&EchoCfg);
176
177 /* 2. Configure Timer Prescaler for 1us resolution (only if not already running) */
178 TIM_TypeDef *TIMx = US_TIM_Array[pxSensor->Timer];
179
180 if (!(TIMx->CR1 & TIM_CR1_CEN))
181 {
182 uint32_t SystemBusClock = US_SYS_CLK_HZ;
183 uint16_t Local_u16PSC = (uint16_t)((SystemBusClock / 1000000U) - 1U);
184
185 TIMx->CR1 = 0;
186 TIMx->PSC = Local_u16PSC;
187 TIMx->ARR = (pxSensor->Timer == TIM_TIMER2 || pxSensor->Timer == TIM_TIMER5) ? 0xFFFFFFFF : 0xFFFF;
188
189 if (pxSensor->Timer == TIM_TIMER1 || pxSensor->Timer == TIM_TIMER8)
190 {
191 SET_BIT(TIMx->BDTR, 15); /* MOE: Main Output Enable */
192 }
193
194 SET_BIT(TIMx->EGR, 0); /* Force update */
195 TIMx->SR = 0;
196 }
197
198 /* 3. Configure ICU Channel (rising edge, capture enabled; CC interrupt stays OFF) */
199 TIM_ICConfig_t IC_Cfg = {
200 .Timer = pxSensor->Timer, .Channel = pxSensor->Channel, .Selection = TIM_IC_SELECTION_DIRECT_TI, .Prescaler = TIM_IC_PSC_DIV1, .Polarity = TIM_POLARITY_HIGH, .Filter = 0xF /* max digital filter: reject glitches that fake an early rising edge → bogus 400cm */
201 };
202 TIM_vIC_Init(&IC_Cfg);
203
204 /* 4. Start Timer only if not running */
205 if (!(TIMx->CR1 & TIM_CR1_CEN))
206 {
207 TIM_vStart(pxSensor->Timer);
208 }
209
210 /* 5. Interrupt infrastructure (idempotent across sensors) */
211 if (US_xDoneSem == NULL)
212 {
213 US_xDoneSem = xSemaphoreCreateBinary(); /* safe to create before scheduler */
214 }
215 TIM_vSetCCCallback(pxSensor->Timer, US_CC_Handler);
216
217 uint8_t Local_u8Irq = US_u8NvicIrqForTimer(pxSensor->Timer);
218 if (Local_u8Irq != 0xFFu)
219 {
220 NVIC_vSetPriority(Local_u8Irq, 6); /* FreeRTOS-safe (>= configMAX_SYSCALL_INTERRUPT_PRIORITY) */
221 NVIC_vEnableIRQ(Local_u8Irq);
222 }
223
224 return OK;
225}

234ErrorState_t US_u16ReadDistance_cm(const US_Config_t *pxSensor, uint16_t *pu16Dist_cm)
235{
236 if (pxSensor == NULL || pu16Dist_cm == NULL)
237 return NULL_POINTER;
238 if (US_xDoneSem == NULL)
239 return NOK; /* US_vInit not done */
240
241 TIM_TypeDef *TIMx = US_TIM_Array[pxSensor->Timer];
242 uint8_t ch = (uint8_t)pxSensor->Channel;
243
244 /* Drain any stale completion signal left by a previous (late) measurement */
245 (void)xSemaphoreTake(US_xDoneSem, 0);
246
247 /* Prepare the active-measurement context */
248 US_Active.timer = pxSensor->Timer;
249 US_Active.channel = ch;
250 US_Active.maxval = (pxSensor->Timer == TIM_TIMER2 || pxSensor->Timer == TIM_TIMER5) ? 0xFFFFFFFFu : 0xFFFFu;
251 US_Active.phase = US_PHASE_RISING;
252 US_Active.valid = 0u;
253
254 /* Arm rising edge → clear ONLY this channel's stale capture flag → enable CC IRQ.
255 * TIM SR flags are "rc_w0": writing 0 clears, writing 1 leaves untouched. So to
256 * clear just CCxIF we write all-ones EXCEPT that bit (~mask). The old code wrote
257 * ~(1<<(ch+1)) which is the same intent, but built the bit from a raw shift; use
258 * the named CCxIF position and a clear comment so it can't be misread as "write
259 * the whole register". */
260 TIM_vSetICPolarity(pxSensor->Timer, pxSensor->Channel, TIM_POLARITY_HIGH);
261 TIMx->SR = ~(1UL << (TIM_SR_CC1IF + ch)); /* clear this channel's CCxIF only */
262 TIM_vEnableCCInterrupt(pxSensor->Timer, pxSensor->Channel);
263
264 /* Fire the trigger pulse */
265 US_vSendTrigger(pxSensor);
266
267 /* Sleep until the ISR delivers both edges, or until the echo window expires */
268 if (xSemaphoreTake(US_xDoneSem, pdMS_TO_TICKS(US_TASK_TIMEOUT_MS)) == pdTRUE && US_Active.phase == US_PHASE_DONE)
269 {
270 /* An out-of-range / spurious echo (valid==0) is reported like "no object"
271 * so the caller's default (400 / clear) kicks in, instead of returning a
272 * fake solid 400cm reading that flaps against the real distance. */
273 if (!US_Active.valid)
274 return TIMEOUT_STATE;
275
276 *pu16Dist_cm = US_Active.dist_cm;
277 return OK;
278 }
279
280 /* Timeout: no echo (out of range / no object) — disarm and report */
281 TIM_vDisableCCInterrupt(pxSensor->Timer, pxSensor->Channel);
282 return TIMEOUT_STATE;
283}
