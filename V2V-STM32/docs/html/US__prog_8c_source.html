96static void US_vSendTrigger(const US_Config_t *pxSensor)
97{
98 /* Settle LOW for a clean edge, then a 10us HIGH pulse (HC-SR04 datasheet). */
99 GPIO_enumWritePinVal(pxSensor->TrigPort, pxSensor->TrigPin, GPIO_PIN_LOW);
100 TIM_vDelayUs(TIM_TIMER6, US_TRIG_SETTLE_US);
101
102 GPIO_enumWritePinVal(pxSensor->TrigPort, pxSensor->TrigPin, GPIO_PIN_HIGH);
103 TIM_vDelayUs(TIM_TIMER6, US_TRIG_PULSE_US);
104
105 GPIO_enumWritePinVal(pxSensor->TrigPort, pxSensor->TrigPin, GPIO_PIN_LOW);
106}

145static void US_CC_Handler(TIM_Num_t Copy_eTimer, uint8_t Copy_u8Channel, uint32_t Copy_u32Capture)
146{
147 /* Ignore captures that don't belong to the active measurement */
148 if (Copy_eTimer != US_Active.timer || Copy_u8Channel != US_Active.channel)
149 {
150 return;
151 }
152
153 if (US_Active.phase == US_PHASE_RISING)
154 {
155 US_Active.t1 = Copy_u32Capture;
156 US_Active.phase = US_PHASE_FALLING;
157 /* Now look for the falling edge on the same channel */
158 TIM_vSetICPolarity(Copy_eTimer, (TIM_Channel_t)Copy_u8Channel, TIM_POLARITY_LOW);
159 }
160 else if (US_Active.phase == US_PHASE_FALLING)
161 {
162 uint32_t high = (Copy_u32Capture >= US_Active.t1)
163 ? (Copy_u32Capture - US_Active.t1)
164 : ((US_Active.maxval - US_Active.t1) + Copy_u32Capture + 1u);
165
166 /* Decode µs → cm. An echo longer than the sensor's useful range means
167 * "no object within range": report it as OUT-OF-RANGE (valid = 0) instead
168 * of silently clamping a bogus long/wrapped pulse to a fake 400cm reading
169 * — clamping is what made a lost echo masquerade as a solid 400cm fix and
170 * made the reading flap between a real distance and 400. */
171 uint32_t dist = high / US_SOUND_SPEED_FACTOR;
172 if (dist == 0u || dist > US_MAX_RANGE_CM)
173 {
174 US_Active.valid = 0u; /* out of range / spurious */
175 US_Active.dist_cm = US_MAX_RANGE_CM;
176 }
177 else
178 {
179 US_Active.valid = 1u; /* a real, in-range echo */
180 US_Active.dist_cm = (uint16_t)dist;
181 }
182 US_Active.phase = US_PHASE_DONE;
183
184 /* Done — silence this channel and wake the task */
185 TIM_vDisableCCInterrupt(Copy_eTimer, (TIM_Channel_t)Copy_u8Channel);
186
187 BaseType_t xHPW = pdFALSE;
188 xSemaphoreGiveFromISR(US_xDoneSem, &xHPW);
189 portYIELD_FROM_ISR(xHPW);
190 }

196ErrorState_t US_vInit(const US_Config_t *pxSensor)
197{
198 if (pxSensor == NULL)
199 return NULL_POINTER;
200 if (pxSensor->Timer >= TIM_TIMER6)
201 return NOK; /* basic timers have no IC */
202
203 /* 1. Initialize GPIO Pins */
204 GPIO_PinConfig_t TrigCfg = {
205 .Port = pxSensor->TrigPort, .PinNum = pxSensor->TrigPin, .Mode = GPIO_OUTPUT, .Otype = GPIO_PUSH_PULL, .Speed = GPIO_MEDIUM_SPEED, .PullType = GPIO_NO_PULL};
206 GPIO_enumPinInit(&TrigCfg);
207
208 GPIO_PinConfig_t EchoCfg = {
209 .Port = pxSensor->EchoPort, .PinNum = pxSensor->EchoPin, .Mode = GPIO_ALTFN, .Otype = GPIO_PUSH_PULL, .Speed = GPIO_VERY_HIGH_SPEED, .PullType = GPIO_NO_PULL};
210
211 if (pxSensor->Timer == TIM_TIMER1 || pxSensor->Timer == TIM_TIMER2)
212 EchoCfg.AlternateFunction = GPIO_AF1;
213 else if (pxSensor->Timer >= TIM_TIMER3 && pxSensor->Timer <= TIM_TIMER5)
214 EchoCfg.AlternateFunction = GPIO_AF2;
215 else if (pxSensor->Timer == TIM_TIMER8)
216 EchoCfg.AlternateFunction = GPIO_AF3;
217 else
218 return NOK;
219 GPIO_enumPinInit(&EchoCfg);
220
221 /* 2. Configure Timer Prescaler for 1us resolution (only if not already running) */
222 TIM_TypeDef *TIMx = US_TIM_Array[pxSensor->Timer];
223
224 if (!(TIMx->CR1 & TIM_CR1_CEN))
225 {
226 uint32_t SystemBusClock = US_SYS_CLK_HZ;
227 uint16_t Local_u16PSC = (uint16_t)((SystemBusClock / 1000000U) - 1U);
228
229 TIMx->CR1 = 0;
230 TIMx->PSC = Local_u16PSC;
231 TIMx->ARR = (pxSensor->Timer == TIM_TIMER2 || pxSensor->Timer == TIM_TIMER5) ? 0xFFFFFFFF : 0xFFFF;
232
233 if (pxSensor->Timer == TIM_TIMER1 || pxSensor->Timer == TIM_TIMER8)
234 {
235 SET_BIT(TIMx->BDTR, 15); /* MOE: Main Output Enable */
236 }
237
238 SET_BIT(TIMx->EGR, 0); /* Force update */
239 TIMx->SR = 0;
240 }
241
242 /* 3. Configure ICU Channel (rising edge, capture enabled; CC interrupt stays OFF) */
243 TIM_ICConfig_t IC_Cfg = {
244 .Timer = pxSensor->Timer, .Channel = pxSensor->Channel, .Selection = TIM_IC_SELECTION_DIRECT_TI, .Prescaler = TIM_IC_PSC_DIV1, .Polarity = TIM_POLARITY_HIGH, .Filter = 0xF /* max digital filter: reject glitches that fake an early rising edge → bogus 400cm */
245 };
246 TIM_vIC_Init(&IC_Cfg);
247
248 /* 4. Start Timer only if not running */
249 if (!(TIMx->CR1 & TIM_CR1_CEN))
250 {
251 TIM_vStart(pxSensor->Timer);
252 }
253
254 /* 5. Interrupt infrastructure (idempotent across sensors) */
255 if (US_xDoneSem == NULL)
256 {
257 US_xDoneSem = xSemaphoreCreateBinary(); /* safe to create before scheduler */
258 }
259 TIM_vSetCCCallback(pxSensor->Timer, US_CC_Handler);
260
261 uint8_t Local_u8Irq = US_u8NvicIrqForTimer(pxSensor->Timer);
262 if (Local_u8Irq != 0xFFu)
263 {
264 NVIC_vSetPriority(Local_u8Irq, 6); /* FreeRTOS-safe (>= configMAX_SYSCALL_INTERRUPT_PRIORITY) */
265 NVIC_vEnableIRQ(Local_u8Irq);
266 }
267
268 return OK;
269}

271/*
272 * @brief Trigger sensor and measure distance (cm). The calling task SLEEPS on a
273 * semaphore while the echo is in flight (CPU free), then the IC ISR wakes
274 * it with the result. Returns TIMEOUT_STATE if no echo within the window.
275 * @note MUST be called from a task context (uses a FreeRTOS semaphore).
276 * Not reentrant: one measurement at a time (sequential by design).
277 */
278ErrorState_t US_u16ReadDistance_cm(const US_Config_t *pxSensor, uint16_t *pu16Dist_cm)
279{
280 if (pxSensor == NULL || pu16Dist_cm == NULL)
281 return NULL_POINTER;
282 if (US_xDoneSem == NULL)
283 return NOK; /* US_vInit not done */
284
285 TIM_TypeDef *TIMx = US_TIM_Array[pxSensor->Timer];
286 uint8_t ch = (uint8_t)pxSensor->Channel;
287
288 /* Drain any stale completion signal left by a previous (late) measurement */
289 (void)xSemaphoreTake(US_xDoneSem, 0);
290
291 /* Prepare the active-measurement context */
292 US_Active.timer = pxSensor->Timer;
293 US_Active.channel = ch;
294 US_Active.maxval = (pxSensor->Timer == TIM_TIMER2 || pxSensor->Timer == TIM_TIMER5) ? 0xFFFFFFFFu : 0xFFFFu;
295 US_Active.phase = US_PHASE_RISING;
296 US_Active.valid = 0u;
297
298 /* Arm rising edge → clear ONLY this channel's stale capture flag → enable CC IRQ.
299 * TIM SR flags are "rc_w0": writing 0 clears, writing 1 leaves untouched. So to
300 * clear just CCxIF we write all-ones EXCEPT that bit (~mask). The old code wrote
301 * ~(1<<(ch+1)) which is the same intent, but built the bit from a raw shift; use
302 * the named CCxIF position and a clear comment so it can't be misread as "write
303 * the whole register". */
304 TIM_vSetICPolarity(pxSensor->Timer, pxSensor->Channel, TIM_POLARITY_HIGH);
305 TIMx->SR = ~(1UL << (TIM_SR_CC1IF + ch)); /* clear this channel's CCxIF only */
306 TIM_vEnableCCInterrupt(pxSensor->Timer, pxSensor->Channel);
307
308 /* Fire the trigger pulse */
309 US_vSendTrigger(pxSensor);
310
311 /* Sleep until the ISR delivers both edges, or until the echo window expires */
312 if (xSemaphoreTake(US_xDoneSem, pdMS_TO_TICKS(US_TASK_TIMEOUT_MS)) == pdTRUE && US_Active.phase == US_PHASE_DONE)
313 {
314 /* An out-of-range / spurious echo (valid==0) is reported like "no object"
315 * so the caller's default (400 / clear) kicks in, instead of returning a
316 * fake solid 400cm reading that flaps against the real distance. */
317 if (!US_Active.valid)
318 return TIMEOUT_STATE;
319
320 *pu16Dist_cm = US_Active.dist_cm;
321 return OK;
322 }
323
324 /* Timeout: no echo (out of range / no object) — disarm and report */
325 TIM_vDisableCCInterrupt(pxSensor->Timer, pxSensor->Channel);
326 return TIMEOUT_STATE;
327}
