171 xTaskCreate(vTask_RPi_Comm, "RPi_Comm_Task", configMINIMAL_STACK_SIZE + 256, NULL, 1, NULL);

179 /* 3. OS Initialization and start running tasks */

247void vTask_Sensors(void *pvParameters)
248{
249 static float local_speed_ms = 0.0f;
250 static MPU9250_Position_t local_pos = {0};
251 MPU9250_Data_t mpu_data = {0};
252
253 /* Interleaved order: each pair is geographically distant (front↔back same side)
254 * → no two adjacent reads share an acoustic path → cross-talk eliminated. */
255 const US_Config_t *sensors[6] = {
256 &FrontUS[0], &BackUS[0], /* Left: front then back */
257 &FrontUS[1], &BackUS[1], /* Center: front then back */
258 &FrontUS[2], &BackUS[2] /* Right: front then back */
259 };
260
261 /* Timestamp of the PREVIOUS MPU sample. dt is measured right around the MPU
262 * read (not at loop top), so it is the true interval between two consecutive
263 * IMU samples — independent of the variable-length ultrasonic scan that runs
264 * in between. This keeps the attitude/heading/speed/position integration exact. */
265 TickType_t xPrevMpuTick = xTaskGetTickCount();
266
267 for(;;)
268 {
269 G_au32Heartbeat[HB_SENSORS]++;
270
271 /* ── Read ALL 6 ultrasonics into locals (task sleeps during each echo) ── */
272 float us[6];
273 uint16_t raw;
274 for (uint8_t i = 0; i < 6; i++)
275 {
276 us[i] = 400.0f; /* default = out of range / clear (no echo within 4m) */
277 if (US_u16ReadDistance_cm(sensors[i], &raw) == OK)
278 us[i] = (float)raw;
279 }
280
281 /* ── dt = interval since the previous MPU sample (measured at the read) ── */
282 TickType_t xNow = xTaskGetTickCount();
283 float dt = (float)(xNow - xPrevMpuTick) * 0.001f; /* ms → seconds */
284 if (dt <= 0.0f) dt = 0.010f; /* guard: first run */
285 xPrevMpuTick = xNow;
286
287 /* ── MPU9250: read and process into locals ── */
288 MPU9250_enumReadData(&mpu_data);
289
290 float pitch = 0.0f, roll = 0.0f, heading = 0.0f;
291 MPU9250_enumGetAttitude(&mpu_data, dt, &pitch, &roll);
292 MPU9250_enumGetHeading(&mpu_data, dt, &heading);
293 MPU9250_enumGetSpeed(&mpu_data, dt, &local_speed_ms);
294 MPU9250_enumGetPosition(&mpu_data, local_speed_ms, heading, pitch, dt, &local_pos);
295
296 /* ── Publish everything in ONE short mutex section ── */
297 xSemaphoreTake(G_xDataMutex, portMAX_DELAY);
298 G_stHostVehicleState.FrontLeftUS = us[0];
299 G_stHostVehicleState.BackLeftUS = us[1];
300 G_stHostVehicleState.FrontCenterUS = us[2];
301 G_stHostVehicleState.BackCenterUS = us[3];
302 G_stHostVehicleState.FrontRightUS = us[4];
303 G_stHostVehicleState.BackRightUS = us[5];
304 G_stHostVehicleState.Speed = local_speed_ms * 100.0f; /* m/s → cm/s */
305 G_stHostVehicleState.Heading = heading;
306 G_stHostVehicleState.Pitch = pitch;
307 G_stHostVehicleState.Roll = roll;
308 G_stHostVehicleState.PosX = local_pos.X;
309 G_stHostVehicleState.PosY = local_pos.Y;
310 G_stHostVehicleState.PosZ = local_pos.Z;
311 xSemaphoreGive(G_xDataMutex);
312
313 /* ── Small adaptive gap (other tasks get CPU; sets min scan spacing) ── */
314 vTaskDelay(pdMS_TO_TICKS(SENSORS_CYCLE_GAP_MS));
315 }

334void vTask_Feedback(void *pvParameters)
335{
336 for(;;)
337 {
338 G_au32Heartbeat[HB_FEEDBACK]++;
339
340 /* Single atomic read of the volatile status word (uint16 read is atomic on M4). */
341 uint16_t flags = G_u16SystemFlags;
342
343 if (flags == 0)
344 {
345 /* ── ALL SAFE: no hazard from any module → silence everything ── */
346 LED_TurnOff(&FrontR_LED);
347 LED_TurnOff(&FrontL_LED);
348 LED_TurnOff(&BackR_LED);
349 LED_TurnOff(&BackL_LED);
350 LED_TurnOff(&Interior_LED);
351 BUZ_Off(&V2X_Buzzer);
352 }
353 else
354 {
355 /* ── ANY alert (any module, warning OR critical) → warn the driver ── */
356 LED_TurnOn(&Interior_LED);
357 BUZ_On(&V2X_Buzzer);
358
359 /* FCW CRITICAL → front LEDs (imminent forward collision). */
360 if (SYS_GET(flags, SYS_FCW_POS) == SYS_CRITICAL)
361 { LED_TurnOn(&FrontR_LED); LED_TurnOn(&FrontL_LED); }
362 else
363 { LED_TurnOff(&FrontR_LED); LED_TurnOff(&FrontL_LED); }
364
365 /* EEBL CRITICAL → rear LEDs (hard braking → warn the car behind). */
366 if (SYS_GET(flags, SYS_EEBL_POS) == SYS_CRITICAL)
367 { LED_TurnOn(&BackR_LED); LED_TurnOn(&BackL_LED); }
368 else
369 { LED_TurnOff(&BackR_LED); LED_TurnOff(&BackL_LED); }
370 }
371
372 vTaskDelay(pdMS_TO_TICKS(25));
373 }

387void vTask_ESP_Comm(void *pvParameters)
388{
389 TickType_t xLastTXTime = xTaskGetTickCount();
390
391 for(;;)
392 {
393 G_au32Heartbeat[HB_ESP]++;
394
395 /* ── RX: event-driven byte processing ───────────────────────── */
396 uint8_t byte; // Buffer to hold received byte from ESP32
397 if (xQueueReceive(G_xESP_RX_Queue, &byte, pdMS_TO_TICKS(10)) == pdTRUE) // Waits for 10ms for a byte to be received
398 {
399 DSRC_RxCallback(byte); // Calls the callback function to process the received byte
400
401 /* Drain every remaining byte without blocking */
402 while (xQueueReceive(G_xESP_RX_Queue, &byte, 0) == pdTRUE)
403 {
404 DSRC_RxCallback(byte);
405 }
406 }
407
408 /* ── Neighbor table maintenance — runs EVERY iteration (~10ms) ──
409 * DSRC_Update flushes any newly-assembled packets (no-op if none).
410 * DSRC_RemoveStale MUST run even when no bytes arrived, otherwise a
411 * vehicle that goes silent (out of range / powered off) would never
412 * be purged and would haunt the table forever. */
413 xSemaphoreTake(G_xNeighborTableMutex, portMAX_DELAY);
414 DSRC_Update();
415 DSRC_RemoveStale((uint32_t)xTaskGetTickCount());
416 xSemaphoreGive(G_xNeighborTableMutex);
417
418 /* ── TX: periodic broadcast every 100 ms ────────────────────── */
419 if ((xTaskGetTickCount() - xLastTXTime) >= pdMS_TO_TICKS(100))
420 {
421 xLastTXTime = xTaskGetTickCount();
422
423 Neighbor my_data = {0};
424 my_data.vehicle_id = VEHICLE_ID;
425 /* last_update is intentionally left 0: the receiver overwrites it with its
426 * OWN local FreeRTOS tick in update_neighbor() (a sender's clock is
427 * meaningless to us), so transmitting it would only waste bytes. */
428
429 /* Read host state AND the cooperative ADAS flags under the SAME mutex, so
430 * the broadcast packet is one consistent snapshot of the last SafetyEngine
431 * cycle. vTask_SafetyEngine writes these module statics while holding
432 * G_xDataMutex, so taking it here closes the race — notably IMA_u8GetFlag(),
433 * which reads three statics and is NOT atomic.
434 * fcw_headon_flag : head-on candidate (0/1) for the oncoming car to confirm
435 * bsw_flag : my own front side(s) seeing a car (bit0=L, bit1=R)
436 * ima_flag : intersection movement assist (0/1/2)
437 * distance_to_intersection : for the neighbors' IMA geometry */
438 xSemaphoreTake(G_xDataMutex, portMAX_DELAY);
439 my_data.speed = G_stHostVehicleState.Speed;
440 my_data.heading = G_stHostVehicleState.Heading;
441 my_data.fcw_headon_flag = FCW_GetHeadonFlag();
442 my_data.bsw_flag = BSW_u8GetFlag();
443 my_data.ima_flag = IMA_u8GetFlag();
444 my_data.distance_to_intersection = Host_DistToIntersection;
445 xSemaphoreGive(G_xDataMutex);
446
447 DSRC_SendNeighbor(&my_data);
448 }
449 }
450}

466void vTask_RPi_Comm(void *pvParameters)
467{
468 TickType_t xLastWakeTime = xTaskGetTickCount();
469
470 char line[112];
471
472 for (;;)
473 {
474 G_au32Heartbeat[HB_RPI]++;
475
476 float spd, hdg, pit, rol, fl, fc, fr, bl, bc, br;
477 uint16_t flags;
478 uint8_t bsw_sides;
479
480 /* Read shared state under mutex. The BSW per-side severity is computed from
481 * statics that vTask_SafetyEngine writes while holding G_xDataMutex, so read
482 * it here under the SAME lock to get a consistent snapshot. */
483 xSemaphoreTake(G_xDataMutex, portMAX_DELAY);
484 spd = G_stHostVehicleState.Speed;
485 hdg = G_stHostVehicleState.Heading;
486 pit = G_stHostVehicleState.Pitch;
487 rol = G_stHostVehicleState.Roll;
488 fl = G_stHostVehicleState.FrontLeftUS;
489 fc = G_stHostVehicleState.FrontCenterUS;
490 fr = G_stHostVehicleState.FrontRightUS;
491 bl = G_stHostVehicleState.BackLeftUS;
492 bc = G_stHostVehicleState.BackCenterUS;
493 br = G_stHostVehicleState.BackRightUS;
494 bsw_sides = BSW_u8GetSidesSeverity();
495 xSemaphoreGive(G_xDataMutex);
496
497 /* 16-bit ADAS status word (2 bits/module) — volatile read is atomic, no mutex.
498 * The RPi decodes each module via (flags >> SYS_xxx_POS) & 0x3. */
499 flags = G_u16SystemFlags;
500
501 /* ASCII CSV line — contains NO 0x00 bytes, so it survives the UART link that
502 * was losing the binary packet's zero-runs. '\n'-delimited, trivial to parse.
503 * Format: T,speed,heading,pitch,roll,FL,FC,FR,BL,BC,BR,flags,bsw_sides\n
504 * bsw_sides: bits 1:0 = LEFT blind-spot severity, bits 3:2 = RIGHT (0/1/2).
505 * The aggregated BSW severity already lives inside `flags`; this extra column
506 * only carries WHICH side(s) so the RPi can say left / right / both. */
507 snprintf(line, sizeof(line),
508 "T,%.1f,%.1f,%.1f,%.1f,%.0f,%.0f,%.0f,%.0f,%.0f,%.0f,%u,%u\n",
509 spd, hdg, pit, rol, fl, fc, fr, bl, bc, br,
510 (unsigned)flags, (unsigned)bsw_sides);
511
512 USART_enumTransmitString(&RPi_UART, (uint8_t *)line);
513
514 vTaskDelayUntil(&xLastWakeTime, pdMS_TO_TICKS(100));
515 }
516}

534void vTask_Watchdog(void *pvParameters)
535{
536 uint32_t snapshot[HB_COUNT];
537 for (uint8_t i = 0; i < HB_COUNT; i++)
538 snapshot[i] = G_au32Heartbeat[i];
539
540 IWDG_voidRefresh(); /* initial kick so the window starts clean */
541
542 TickType_t xLastWake = xTaskGetTickCount();
543 for (;;)
544 {
545 vTaskDelayUntil(&xLastWake, pdMS_TO_TICKS(WDG_CHECK_PERIOD_MS));
546
547 uint8_t all_alive = 1;
548 for (uint8_t i = 0; i < HB_COUNT; i++)
549 {
550 if (G_au32Heartbeat[i] == snapshot[i]) all_alive = 0; /* this task stalled */
551 snapshot[i] = G_au32Heartbeat[i];
552 }
553
554 /* Kick the dog only while EVERY monitored task is still advancing.
555 * If any stalled, deliberately skip the refresh → IWDG resets the MCU. */
556 if (all_alive)
557 IWDG_voidRefresh();
558 }
559}
