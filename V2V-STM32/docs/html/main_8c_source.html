47/* ================== Watchdog heartbeats ==================
48 * Each monitored task bumps its own slot every loop. vTask_Watchdog refreshes
49 * the IWDG ONLY while EVERY slot keeps advancing; if any task stalls (hard
50 * fault, stack-overflow halt, infinite loop, ISR storm, or starvation) its slot
51 * freezes → the refresh stops → the IWDG hardware-resets the MCU. */
52enum { HB_SAFETY = 0, HB_SENSORS, HB_ESP, HB_FEEDBACK, HB_RPI, HB_COUNT };
53volatile uint32_t G_au32Heartbeat[HB_COUNT] = {0};

146 * and sets the minimum spacing between scans. The scan itself is adaptive —
147 * its length follows echo flight time (near objects → faster refresh). */

163void vTask_Sensors(void *pvParameters)
164{
165 static float local_speed_ms = 0.0f;
166 static MPU9250_Position_t local_pos = {0};
167 MPU9250_Data_t mpu_data = {0};
168
169 /* Interleaved order: each pair is geographically distant (front↔back same side)
170 * → no two adjacent reads share an acoustic path → cross-talk eliminated. */
171 const US_Config_t *sensors[6] = {
172 &FrontUS[0], &BackUS[0], /* Left: front then back */
173 &FrontUS[1], &BackUS[1], /* Center: front then back */
174 &FrontUS[2], &BackUS[2] /* Right: front then back */
175 };
176
177 /* Timestamp of the PREVIOUS MPU sample. dt is measured right around the MPU
178 * read (not at loop top), so it is the true interval between two consecutive
179 * IMU samples — independent of the variable-length ultrasonic scan that runs
180 * in between. This keeps the attitude/heading/speed/position integration exact. */
181 TickType_t xPrevMpuTick = xTaskGetTickCount();
182
183 for(;;)
184 {
185 G_au32Heartbeat[HB_SENSORS]++;
186
187 /* ── Read ALL 6 ultrasonics into locals (task sleeps during each echo) ── */
188 float us[6];
189 uint16_t raw;
190 for (uint8_t i = 0; i < 6; i++)
191 {
192 us[i] = 400.0f; /* default = out of range / clear (no echo within 4m) */
193 if (US_u16ReadDistance_cm(sensors[i], &raw) == OK)
194 us[i] = (float)raw;
195 }
196
197 /* ── dt = interval since the previous MPU sample (measured at the read) ── */
198 TickType_t xNow = xTaskGetTickCount();
199 float dt = (float)(xNow - xPrevMpuTick) * 0.001f; /* ms → seconds */
200 if (dt <= 0.0f) dt = 0.010f; /* guard: first run */
201 xPrevMpuTick = xNow;
202
203 /* ── MPU9250: read and process into locals ── */
204 MPU9250_enumReadData(&mpu_data);
205
206 float pitch = 0.0f, roll = 0.0f, heading = 0.0f;
207 MPU9250_enumGetAttitude(&mpu_data, dt, &pitch, &roll);
208 MPU9250_enumGetHeading(&mpu_data, dt, &heading);
209 MPU9250_enumGetSpeed(&mpu_data, dt, &local_speed_ms);
210 MPU9250_enumGetPosition(&mpu_data, local_speed_ms, heading, pitch, dt, &local_pos);
211
212 /* ── Publish everything in ONE short mutex section ── */
213 xSemaphoreTake(G_xDataMutex, portMAX_DELAY);
214 G_stHostVehicleState.FrontLeftUS = us[0];
215 G_stHostVehicleState.BackLeftUS = us[1];
216 G_stHostVehicleState.FrontCenterUS = us[2];
217 G_stHostVehicleState.BackCenterUS = us[3];
218 G_stHostVehicleState.FrontRightUS = us[4];
219 G_stHostVehicleState.BackRightUS = us[5];
220 G_stHostVehicleState.Speed = local_speed_ms * 100.0f; /* m/s → cm/s */
221 G_stHostVehicleState.Heading = heading;
222 G_stHostVehicleState.Pitch = pitch;
223 G_stHostVehicleState.Roll = roll;
224 G_stHostVehicleState.PosX = local_pos.X;
225 G_stHostVehicleState.PosY = local_pos.Y;
226 G_stHostVehicleState.PosZ = local_pos.Z;
227 xSemaphoreGive(G_xDataMutex);
228
229 /* ── Small adaptive gap (other tasks get CPU; sets min scan spacing) ── */
230 vTaskDelay(pdMS_TO_TICKS(SENSORS_CYCLE_GAP_MS));
231 }

248void vTask_Feedback(void *pvParameters)
249{
250 for(;;)
251 {
252 G_au32Heartbeat[HB_FEEDBACK]++;
253
254 /* Single atomic read of the volatile status word (uint16 read is atomic on M4). */
255 uint16_t flags = G_u16SystemFlags;
256
257 if (flags == 0)
258 {
259 /* ── ALL SAFE: no hazard from any module → silence everything ── */
260 LED_TurnOff(&FrontR_LED);
261 LED_TurnOff(&FrontL_LED);
262 LED_TurnOff(&BackR_LED);
263 LED_TurnOff(&BackL_LED);
264 LED_TurnOff(&Interior_LED);
265 BUZ_Off(&V2X_Buzzer);
266 }
267 else
268 {
269 /* ── ANY alert (any module, warning OR critical) → warn the driver ── */
270 LED_TurnOn(&Interior_LED);
271 BUZ_On(&V2X_Buzzer);
272
273 /* FCW CRITICAL → front LEDs (imminent forward collision). */
274 if (SYS_GET(flags, SYS_FCW_POS) == SYS_CRITICAL)
275 { LED_TurnOn(&FrontR_LED); LED_TurnOn(&FrontL_LED); }
276 else
277 { LED_TurnOff(&FrontR_LED); LED_TurnOff(&FrontL_LED); }
278
279 /* EEBL CRITICAL → rear LEDs (hard braking → warn the car behind). */
280 if (SYS_GET(flags, SYS_EEBL_POS) == SYS_CRITICAL)
281 { LED_TurnOn(&BackR_LED); LED_TurnOn(&BackL_LED); }
282 else
283 { LED_TurnOff(&BackR_LED); LED_TurnOff(&BackL_LED); }
284 }
285
286 vTaskDelay(pdMS_TO_TICKS(25));
287 }

299void vTask_ESP_Comm(void *pvParameters)
300{
301 TickType_t xLastTXTime = xTaskGetTickCount();
302
303 for(;;)
304 {
305 G_au32Heartbeat[HB_ESP]++;
306
307 /* ── RX: event-driven byte processing ───────────────────────── */
308 uint8_t byte; // Buffer to hold received byte from ESP32
309 if (xQueueReceive(G_xESP_RX_Queue, &byte, pdMS_TO_TICKS(10)) == pdTRUE) // Waits for 10ms for a byte to be received
310 {
311 DSRC_RxCallback(byte); // Calls the callback function to process the received byte
312
313 /* Drain every remaining byte without blocking */
314 while (xQueueReceive(G_xESP_RX_Queue, &byte, 0) == pdTRUE)
315 {
316 DSRC_RxCallback(byte);
317 }
318 }
319
320 /* ── Neighbor table maintenance — runs EVERY iteration (~10ms) ──
321 * DSRC_Update flushes any newly-assembled packets (no-op if none).
322 * DSRC_RemoveStale MUST run even when no bytes arrived, otherwise a
323 * vehicle that goes silent (out of range / powered off) would never
324 * be purged and would haunt the table forever. */
325 xSemaphoreTake(G_xNeighborTableMutex, portMAX_DELAY);
326 DSRC_Update();
327 DSRC_RemoveStale((uint32_t)xTaskGetTickCount());
328 xSemaphoreGive(G_xNeighborTableMutex);
329
330 /* ── TX: periodic broadcast every 100 ms ────────────────────── */
331 if ((xTaskGetTickCount() - xLastTXTime) >= pdMS_TO_TICKS(100))
332 {
333 xLastTXTime = xTaskGetTickCount();
334
335 Neighbor my_data = {0};
336 my_data.vehicle_id = VEHICLE_ID;
337 /* last_update is intentionally left 0: the receiver overwrites it with its
338 * OWN local FreeRTOS tick in update_neighbor() (a sender's clock is
339 * meaningless to us), so transmitting it would only waste bytes. */
340
341 /* Read host state AND the cooperative ADAS flags under the SAME mutex, so
342 * the broadcast packet is one consistent snapshot of the last SafetyEngine
343 * cycle. vTask_SafetyEngine writes these module statics while holding
344 * G_xDataMutex, so taking it here closes the race — notably IMA_u8GetFlag(),
345 * which reads three statics and is NOT atomic.
346 * fcw_headon_flag : head-on candidate (0/1) for the oncoming car to confirm
347 * bsw_flag : my own front side(s) seeing a car (bit0=L, bit1=R)
348 * ima_flag : intersection movement assist (0/1/2)
349 * distance_to_intersection : for the neighbors' IMA geometry */
350 xSemaphoreTake(G_xDataMutex, portMAX_DELAY);
351 my_data.speed = G_stHostVehicleState.Speed;
352 my_data.heading = G_stHostVehicleState.Heading;
353 my_data.fcw_headon_flag = FCW_GetHeadonFlag();
354 my_data.bsw_flag = BSW_u8GetFlag();
355 my_data.ima_flag = IMA_u8GetFlag();
356 my_data.distance_to_intersection = Host_DistToIntersection;
357 xSemaphoreGive(G_xDataMutex);
358
359 DSRC_SendNeighbor(&my_data);
360 }
361 }

365void vTask_RPi_Comm(void *pvParameters)
366{
367 TickType_t xLastWakeTime = xTaskGetTickCount();
368
369 char line[112];
370
371 for (;;)
372 {
373 G_au32Heartbeat[HB_RPI]++;
374
375 float spd, hdg, pit, rol, fl, fc, fr, bl, bc, br;
376 uint16_t flags;
377 uint8_t bsw_sides;
378
379 /* Read shared state under mutex. The BSW per-side severity is computed from
380 * statics that vTask_SafetyEngine writes while holding G_xDataMutex, so read
381 * it here under the SAME lock to get a consistent snapshot. */
382 xSemaphoreTake(G_xDataMutex, portMAX_DELAY);
383 spd = G_stHostVehicleState.Speed;
384 hdg = G_stHostVehicleState.Heading;
385 pit = G_stHostVehicleState.Pitch;
386 rol = G_stHostVehicleState.Roll;
387 fl = G_stHostVehicleState.FrontLeftUS;
388 fc = G_stHostVehicleState.FrontCenterUS;
389 fr = G_stHostVehicleState.FrontRightUS;
390 bl = G_stHostVehicleState.BackLeftUS;
391 bc = G_stHostVehicleState.BackCenterUS;
392 br = G_stHostVehicleState.BackRightUS;
393 bsw_sides = BSW_u8GetSidesSeverity();
394 xSemaphoreGive(G_xDataMutex);
395
396 /* 16-bit ADAS status word (2 bits/module) — volatile read is atomic, no mutex.
397 * The RPi decodes each module via (flags >> SYS_xxx_POS) & 0x3. */
398 flags = G_u16SystemFlags;
399
400 /* ASCII CSV line — contains NO 0x00 bytes, so it survives the UART link that
401 * was losing the binary packet's zero-runs. '\n'-delimited, trivial to parse.
402 * Format: T,speed,heading,pitch,roll,FL,FC,FR,BL,BC,BR,flags,bsw_sides\n
403 * bsw_sides: bits 1:0 = LEFT blind-spot severity, bits 3:2 = RIGHT (0/1/2).
404 * The aggregated BSW severity already lives inside `flags`; this extra column
405 * only carries WHICH side(s) so the RPi can say left / right / both. */
406 snprintf(line, sizeof(line),
407 "T,%.1f,%.1f,%.1f,%.1f,%.0f,%.0f,%.0f,%.0f,%.0f,%.0f,%u,%u\n",
408 spd, hdg, pit, rol, fl, fc, fr, bl, bc, br,
409 (unsigned)flags, (unsigned)bsw_sides);
410
411 USART_enumTransmitString(&RPi_UART, (uint8_t *)line);
412
413 vTaskDelayUntil(&xLastWakeTime, pdMS_TO_TICKS(100));
414 }

431void vTask_Watchdog(void *pvParameters)
432{
433 uint32_t snapshot[HB_COUNT];
434 for (uint8_t i = 0; i < HB_COUNT; i++)
435 snapshot[i] = G_au32Heartbeat[i];
436
437 IWDG_voidRefresh(); /* initial kick so the window starts clean */
438
439 TickType_t xLastWake = xTaskGetTickCount();
440 for (;;)
441 {
442 vTaskDelayUntil(&xLastWake, pdMS_TO_TICKS(WDG_CHECK_PERIOD_MS));
443
444 uint8_t all_alive = 1;
445 for (uint8_t i = 0; i < HB_COUNT; i++)
446 {
447 if (G_au32Heartbeat[i] == snapshot[i]) all_alive = 0; /* this task stalled */
448 snapshot[i] = G_au32Heartbeat[i];
449 }
450
451 /* Kick the dog only while EVERY monitored task is still advancing.
452 * If any stalled, deliberately skip the refresh → IWDG resets the MCU. */
453 if (all_alive)
454 IWDG_voidRefresh();
455 }
456}
