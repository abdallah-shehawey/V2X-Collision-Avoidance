17/*_______________________________________________________________________________*/
18/*
19 * System Clock Frequency (Hz)
20 * Used to calculate timer prescaler for 1us tick.
21 * Default: 16MHz (HSI)
22 * Change this if you are using a different clock (e.g. 180000000UL for 180MHz PLL)
23 */
24#define US_SYS_CLK_HZ 16000000UL
25
26/*_______________________________________________________________________________*/
27/*
28 * Settle time before the trigger (microseconds)
29 * TRIG is driven LOW for this long before the HIGH pulse to guarantee a clean
30 * edge (no glitch from a previous measurement).
31 */
32#define US_TRIG_SETTLE_US 2U
33
34/*_______________________________________________________________________________*/
35/*
36 * Trigger Pulse Duration (microseconds)
37 * HC-SR04 requires a minimum 10us HIGH pulse on TRIG pin (datasheet).
38 * Do NOT lower this below 10us.
39 */
40#define US_TRIG_PULSE_US 10U
41
42/*_______________________________________________________________________________*/
43/*
44 * Maximum range (centimeters).
45 * HC-SR04 useful range is ~2..400 cm. Any echo that decodes to a larger
46 * distance is treated as out-of-range (object lost) and reported as a timeout.
47 * This is the SINGLE source of truth for "max range":
48 * - it clamps the decoded distance in the IC ISR, and
49 * - it derives the task-level echo timeout below.
50 */
51#define US_MAX_RANGE_CM 400U
52
53/*_______________________________________________________________________________*/
54/*
55 * Sound Speed Factor
56 * Distance (cm) = Echo_pulse_us / 58
57 * (Speed of sound ~343 m/s => ~58 us/cm round-trip)
58 */
59#define US_SOUND_SPEED_FACTOR 58U
60
61/*_______________________________________________________________________________*/
62/*
63 * Task-level Echo Timeout (milliseconds) — interrupt-driven driver.
64 * The reading task SLEEPS (vTask/semaphore) up to this long waiting for the
65 * IC interrupt to deliver both echo edges. It is DERIVED from US_MAX_RANGE_CM so
66 * the timeout and the range clamp can never drift apart:
67 * timeout_ms = ceil(MAX_RANGE_CM * 58us/cm / 1000) + 1ms slack
68 * = ceil(400 * 58 / 1000) + 1 = 24 + 1 = 25ms → full 4m range.
69 * Worst case (all 6 sensors out of range) ≈ 6 * 25 = 150ms per scan.
70 */
71#define US_TASK_TIMEOUT_MS \
72 ((((US_MAX_RANGE_CM * US_SOUND_SPEED_FACTOR) + 999U) / 1000U) + 1U)
