53ErrorState_t RCC_enumSetClkSts(uint8_t Copy_u8CLK, uint8_t Copy_u8Status)
54{
55 ErrorState_t Local_u8ErrorState = OK;
56 uint32_t Local_u32TimeoutCounter = 0;
57
58 if (RCC_u8State == IDLE)
59 {
60 RCC_u8State = BUSY;
61
62 if (Copy_u8CLK == RCC_HSI_CLK)
63 {
64 /* HSI Clock Control */
65 if (Copy_u8Status == RCC_CLK_ON)
66 {
67 /* Enable HSI and wait for it to be ready */
68 SET_BIT(MRCC->CR, 0);
69 while (READ_BIT(MRCC->CR, 1) == 0 && Local_u32TimeoutCounter != RCC_u32TIMEOUT)
70 {
71 Local_u32TimeoutCounter++;
72 }
73
74 if (Local_u32TimeoutCounter == RCC_u32TIMEOUT)
75 {
76 Local_u8ErrorState = TIMEOUT_STATE;
77 }
78 }
79 else if (Copy_u8Status == RCC_CLK_OFF)
80 {
81
82 CLR_BIT(MRCC->CR, 0);
83 while (READ_BIT(MRCC->CR, 1) == 1 && Local_u32TimeoutCounter != RCC_u32TIMEOUT)
84 {
85 Local_u32TimeoutCounter++;
86 }
87
88 if (Local_u32TimeoutCounter == RCC_u32TIMEOUT)
89 {
90 Local_u8ErrorState = TIMEOUT_STATE;
91 }
92 }
93 else
94 {
95 }
96 }
97 else if (Copy_u8CLK == RCC_HSE_CLK)
98 {
99 /* HSE Clock Control */
100 if (Copy_u8Status == RCC_CLK_ON)
101 {
102 /* Enable HSE bypass and HSE clock, then wait for it to be ready */
103 SET_BIT(MRCC->CR, 18); /* Enable HSE bypass */
104 SET_BIT(MRCC->CR, 16); /* Enable HSE clock */
105 while (READ_BIT(MRCC->CR, 17) == 0 && Local_u32TimeoutCounter != RCC_u32TIMEOUT)
106 {
107 Local_u32TimeoutCounter++;
108 }
109
110 if (Local_u32TimeoutCounter == RCC_u32TIMEOUT)
111 {
112 Local_u8ErrorState = TIMEOUT_STATE;
113 }
114 }
115
116 else if (Copy_u8Status == RCC_CLK_OFF)
117 {
118 /* Disable HSE clock and wait for it to stop */
119 CLR_BIT(MRCC->CR, 16);
120 while (READ_BIT(MRCC->CR, 17) == 1 && Local_u32TimeoutCounter != RCC_u32TIMEOUT)
121 {
122 Local_u32TimeoutCounter++;
123 }
124
125 if (Local_u32TimeoutCounter == RCC_u32TIMEOUT)
126 {
127 Local_u8ErrorState = TIMEOUT_STATE;
128 }
129 }
130 }
131 else if (Copy_u8CLK == RCC_PLL_CLK)
132 {
133 /* PLL Clock Control */
134 if (Copy_u8Status == RCC_CLK_ON)
135 {
136 /* Enable PLL and wait for it to be ready */
137 SET_BIT(MRCC->CR, 24);
138 while (READ_BIT(MRCC->CR, 25) == 0 && Local_u32TimeoutCounter != RCC_u32TIMEOUT)
139 {
140 Local_u32TimeoutCounter++;
141 }
142
143 if (Local_u32TimeoutCounter == RCC_u32TIMEOUT)
144 {
145 Local_u8ErrorState = TIMEOUT_STATE;
146 }
147 }
148
149 else if (Copy_u8Status == RCC_CLK_OFF)
150 {
151 /* Disable PLL and wait for it to stop */
152 CLR_BIT(MRCC->CR, 24);
153 while (READ_BIT(MRCC->CR, 25) == 1 && Local_u32TimeoutCounter != RCC_u32TIMEOUT)
154 {
155 Local_u32TimeoutCounter++;
156 }
157
158 if (Local_u32TimeoutCounter == RCC_u32TIMEOUT)
159 {
160 Local_u8ErrorState = TIMEOUT_STATE;
161 }
162 }
163 }
164 else
165 {
166 }
167
168 RCC_u8State = IDLE;
169 }
170 else
171 {
172 Local_u8ErrorState = BUSY_STATE;
173 }

191ErrorState_t RCC_enumSetSysClk(uint8_t Copy_u8CLK)
192{
193 ErrorState_t Local_u8ErrorState = OK;
194
195 if (RCC_u8State == IDLE)
196 {
197 RCC_u8State = BUSY;
198
199 switch (Copy_u8CLK)
200 {
201 case RCC_HSI_CLK:
202 /* Select HSI as system clock */
203 MRCC->CFGR &= ~(RCC_SYS_CLK_MASK);
204 MRCC->CFGR |= Copy_u8CLK;
205 break;
206
207 case RCC_HSE_CLK:
208 /* Select HSE as system clock */
209 MRCC->CFGR &= ~(RCC_SYS_CLK_MASK);
210 MRCC->CFGR |= Copy_u8CLK;
211 break;
212
213 case RCC_PLLP_CLK:
214 /* Select PLL P output as system clock */
215 MRCC->CFGR &= ~(RCC_SYS_CLK_MASK);
216 MRCC->CFGR |= Copy_u8CLK;
217 break;
218
219 case RCC_PLLR_CLK:
220 /* Select PLL R output as system clock */
221 MRCC->CFGR &= ~(RCC_SYS_CLK_MASK);
222 MRCC->CFGR |= Copy_u8CLK;
223 break;

225 break;
226 }

228 RCC_u8State = IDLE;
229 }

257ErrorState_t RCC_enumPLLConfig(const RCC_PLLConfig_t *Copy_PLLConfig)
258{
259 ErrorState_t Local_u8ErrorState = OK;
260
261 if (RCC_u8State == IDLE)
262 {
263 RCC_u8State = BUSY;
264
265 /* Validate configuration parameters */
266 if ((Copy_PLLConfig != NULL) &&
267 (Copy_PLLConfig->PLLM_Div >= RCC_MPLL_DIV_2 && Copy_PLLConfig->PLLM_Div <= RCC_MPLL_DIV_63) &&
268 (Copy_PLLConfig->PLLN_Mult >= RCC_NPLL_CLK_MULT_50 && Copy_PLLConfig->PLLN_Mult <= RCC_NPLL_CLK_MULT_432) &&
269 (Copy_PLLConfig->PLLSource == RCC_PLL_HSI || Copy_PLLConfig->PLLSource == RCC_PLL_HSE) &&
270 (Copy_PLLConfig->PLLP_Div >= RCC_PPLL_DIV_2 && Copy_PLLConfig->PLLP_Div <= RCC_PPLL_DIV_8))
271 {
272 /* Configure PLL source */
273 if (Copy_PLLConfig->PLLSource == RCC_PLL_HSI)
274 {
275 CLR_BIT(MRCC->PLLCFGR, 22);
276 }
277 else if (Copy_PLLConfig->PLLSource == RCC_PLL_HSE)
278 {
279 SET_BIT(MRCC->PLLCFGR, 22);
280 }
281
282 /* Configure PLL M division factor */
283 MRCC->PLLCFGR &= (~(RCC_MPLL_DIV_MASK));
284 MRCC->PLLCFGR |= (Copy_PLLConfig->PLLM_Div);
285
286 /* Configure PLL N multiplication factor */
287 MRCC->PLLCFGR &= (~(RCC_NPLL_MULT_MASK << 6));
288 MRCC->PLLCFGR |= (Copy_PLLConfig->PLLN_Mult << 6);
289
290 /* Configure PLL P division factor */
291 MRCC->PLLCFGR &= (~(RCC_PPLL_DIV_MASK << 16));
292 MRCC->PLLCFGR |= ((Copy_PLLConfig->PLLP_Div * 2 + 2) << 16);
293 }
294 else
295 {
296 Local_u8ErrorState = NOK;
297 }
298
299 RCC_u8State = IDLE;
300 }
301 else
302 {
303 Local_u8ErrorState = BUSY_STATE;
304 }
305
306 return Local_u8ErrorState;
