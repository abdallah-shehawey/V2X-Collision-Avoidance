34/*
35 * @brief This function controls the state (ON/OFF) of the main clock sources
36 * It handles enabling/disabling of HSI, HSE, and PLL clocks with timeout protection
37 * @param Copy_u8CLK Clock source to control (HSI_CLK, HSE_CLK, PLL_CLK)
38 * @param Copy_u8Status Desired clock state (CLK_ON, CLK_OFF)
39 * @return ErrorState_t:
40 * - OK: Operation completed successfully
41 * - TIMEOUT_STATE: Clock failed to stabilize within timeout period
42 * - BUSY_STATE: RCC driver is busy with another operation
43 * @details
44 * - For HSI: Controls bit 0 in CR register and checks HSIRDY (bit 1)
45 * - For HSE: Controls bit 16 in CR register and checks HSERDY (bit 17)
46 * - For PLL: Controls bit 24 in CR register and checks PLLRDY (bit 25)
47 */
48ErrorState_t RCC_enumSetClkSts(uint8_t Copy_u8CLK, uint8_t Copy_u8Status)
49{
50 ErrorState_t Local_u8ErrorState = OK;
51 uint32_t Local_u32TimeoutCounter = 0;
52
53 if (RCC_u8State == IDLE)
54 {
55 RCC_u8State = BUSY;
56
57 if (Copy_u8CLK == RCC_HSI_CLK)
58 {
59 /* HSI Clock Control */
60 if (Copy_u8Status == RCC_CLK_ON)
61 {
62 /* Enable HSI and wait for it to be ready */
63 SET_BIT(MRCC->CR, 0);
64 while (READ_BIT(MRCC->CR, 1) == 0 && Local_u32TimeoutCounter != RCC_u32TIMEOUT)
65 {
66 Local_u32TimeoutCounter++;
67 }
68
69 if (Local_u32TimeoutCounter == RCC_u32TIMEOUT)
70 {
71 Local_u8ErrorState = TIMEOUT_STATE;
72 }
73 }
74 else if (Copy_u8Status == RCC_CLK_OFF)
75 {
76
77 CLR_BIT(MRCC->CR, 0);
78 while (READ_BIT(MRCC->CR, 1) == 1 && Local_u32TimeoutCounter != RCC_u32TIMEOUT)
79 {
80 Local_u32TimeoutCounter++;
81 }
82
83 if (Local_u32TimeoutCounter == RCC_u32TIMEOUT)
84 {
85 Local_u8ErrorState = TIMEOUT_STATE;
86 }
87 }
88 else
89 {
90 }
91 }
92 else if (Copy_u8CLK == RCC_HSE_CLK)
93 {
94 /* HSE Clock Control */
95 if (Copy_u8Status == RCC_CLK_ON)
96 {
97 /* Enable HSE bypass and HSE clock, then wait for it to be ready */
98 SET_BIT(MRCC->CR, 18); /* Enable HSE bypass */
99 SET_BIT(MRCC->CR, 16); /* Enable HSE clock */
100 while (READ_BIT(MRCC->CR, 17) == 0 && Local_u32TimeoutCounter != RCC_u32TIMEOUT)
101 {
102 Local_u32TimeoutCounter++;
103 }
104
105 if (Local_u32TimeoutCounter == RCC_u32TIMEOUT)
106 {
107 Local_u8ErrorState = TIMEOUT_STATE;
108 }
109 }
110
111 else if (Copy_u8Status == RCC_CLK_OFF)
112 {
113 /* Disable HSE clock and wait for it to stop */
114 CLR_BIT(MRCC->CR, 16);
115 while (READ_BIT(MRCC->CR, 17) == 1 && Local_u32TimeoutCounter != RCC_u32TIMEOUT)
116 {
117 Local_u32TimeoutCounter++;
118 }
119
120 if (Local_u32TimeoutCounter == RCC_u32TIMEOUT)
121 {
122 Local_u8ErrorState = TIMEOUT_STATE;
123 }
124 }
125 }
126 else if (Copy_u8CLK == RCC_PLL_CLK)
127 {
128 /* PLL Clock Control */
129 if (Copy_u8Status == RCC_CLK_ON)
130 {
131 /* Enable PLL and wait for it to be ready */
132 SET_BIT(MRCC->CR, 24);
133 while (READ_BIT(MRCC->CR, 25) == 0 && Local_u32TimeoutCounter != RCC_u32TIMEOUT)
134 {
135 Local_u32TimeoutCounter++;
136 }
137
138 if (Local_u32TimeoutCounter == RCC_u32TIMEOUT)
139 {
140 Local_u8ErrorState = TIMEOUT_STATE;
141 }
142 }
143
144 else if (Copy_u8Status == RCC_CLK_OFF)
145 {
146 /* Disable PLL and wait for it to stop */
147 CLR_BIT(MRCC->CR, 24);
148 while (READ_BIT(MRCC->CR, 25) == 1 && Local_u32TimeoutCounter != RCC_u32TIMEOUT)
149 {
150 Local_u32TimeoutCounter++;
151 }
152
153 if (Local_u32TimeoutCounter == RCC_u32TIMEOUT)
154 {
155 Local_u8ErrorState = TIMEOUT_STATE;
156 }
157 }
158 }
159 else
160 {
161 }
162
163 RCC_u8State = IDLE;
164 }
165 else
166 {
167 Local_u8ErrorState = BUSY_STATE;
168 }
169
170 return Local_u8ErrorState;
171}

173/*
174 * @brief This function configures the system clock source
175 * It allows switching between different clock sources as the main system clock
176 * @param Copy_u8CLK Clock source to be used as system clock
177 * (HSI_CLK, HSE_CLK, PLLP_CLK, PLLR_CLK)
178 * @return RCC_ErrorState:
179 * - RCC_OK: System clock switched successfully
180 * - BUSY_STATE: RCC driver is busy with another operation
181 * @details
182 * - Modifies SW bits in CFGR register to select system clock source
183 * - Ensures proper clock switching by clearing and setting appropriate bits
184 */
185ErrorState_t RCC_enumSetSysClk(uint8_t Copy_u8CLK)
186{
187 ErrorState_t Local_u8ErrorState = OK;
188
189 if (RCC_u8State == IDLE)
190 {
191 RCC_u8State = BUSY;
192
193 switch (Copy_u8CLK)
194 {
195 case RCC_HSI_CLK:
196 /* Select HSI as system clock */
197 MRCC->CFGR &= ~(RCC_SYS_CLK_MASK);
198 MRCC->CFGR |= Copy_u8CLK;
199 break;
200
201 case RCC_HSE_CLK:
202 /* Select HSE as system clock */
203 MRCC->CFGR &= ~(RCC_SYS_CLK_MASK);
204 MRCC->CFGR |= Copy_u8CLK;
205 break;
206
207 case RCC_PLLP_CLK:
208 /* Select PLL P output as system clock */
209 MRCC->CFGR &= ~(RCC_SYS_CLK_MASK);
210 MRCC->CFGR |= Copy_u8CLK;
211 break;
212
213 case RCC_PLLR_CLK:
214 /* Select PLL R output as system clock */
215 MRCC->CFGR &= ~(RCC_SYS_CLK_MASK);
216 MRCC->CFGR |= Copy_u8CLK;
217 break;
218 default:
219 break;
220 }
221
222 RCC_u8State = IDLE;
223 }
224 else
225 {
226 Local_u8ErrorState = BUSY_STATE;
227 }

233/*
234 * @brief This function configures the PLL clock parameters
235 * It sets up PLL multiplication and division factors for desired frequency
236 * @param Copy_PLLConfig Pointer to PLL configuration structure containing:
237 * - PLLSource: Clock source for PLL (HSI or HSE)
238 * - PLLM_Div: Division factor for PLL input (2-63)
239 * - PLLN_Mult: Multiplication factor (50-432)
240 * - PLLP_Div: Division factor for main PLL output (2,4,6,8)
241 * @return RCC_ErrorState:
242 * - RCC_OK: PLL configured successfully
243 * - NOK: Invalid configuration parameters
244 * - BUSY_STATE: RCC driver is busy
245 * @details
246 * - Configures PLLCFGR register for PLL clock setup
247 * - Validates all configuration parameters before applying
248 * - Ensures proper frequency ranges for stable operation
249 */
250ErrorState_t RCC_enumPLLConfig(const RCC_PLLConfig_t *Copy_PLLConfig)
251{
252 ErrorState_t Local_u8ErrorState = OK;
253
254 if (RCC_u8State == IDLE)
255 {
256 RCC_u8State = BUSY;
257
258 /* Validate configuration parameters */
259 if ((Copy_PLLConfig != NULL) &&
260 (Copy_PLLConfig->PLLM_Div >= RCC_MPLL_DIV_2 && Copy_PLLConfig->PLLM_Div <= RCC_MPLL_DIV_63) &&
261 (Copy_PLLConfig->PLLN_Mult >= RCC_NPLL_CLK_MULT_50 && Copy_PLLConfig->PLLN_Mult <= RCC_NPLL_CLK_MULT_432) &&
262 (Copy_PLLConfig->PLLSource == RCC_PLL_HSI || Copy_PLLConfig->PLLSource == RCC_PLL_HSE) &&
263 (Copy_PLLConfig->PLLP_Div >= RCC_PPLL_DIV_2 && Copy_PLLConfig->PLLP_Div <= RCC_PPLL_DIV_8))
264 {
265 /* Configure PLL source */
266 if (Copy_PLLConfig->PLLSource == RCC_PLL_HSI)
267 {
268 CLR_BIT(MRCC->PLLCFGR, 22);
269 }
270 else if (Copy_PLLConfig->PLLSource == RCC_PLL_HSE)
271 {
272 SET_BIT(MRCC->PLLCFGR, 22);
273 }
274
275 /* Configure PLL M division factor */
276 MRCC->PLLCFGR &= (~(RCC_MPLL_DIV_MASK));
277 MRCC->PLLCFGR |= (Copy_PLLConfig->PLLM_Div);
278
279 /* Configure PLL N multiplication factor */
280 MRCC->PLLCFGR &= (~(RCC_NPLL_MULT_MASK << 6));
281 MRCC->PLLCFGR |= (Copy_PLLConfig->PLLN_Mult << 6);
282
283 /* Configure PLL P division factor */
284 MRCC->PLLCFGR &= (~(RCC_PPLL_DIV_MASK << 16));
285 MRCC->PLLCFGR |= ((Copy_PLLConfig->PLLP_Div * 2 + 2) << 16);
286 }
287 else
288 {
289 Local_u8ErrorState = NOK;
290 }
291
292 RCC_u8State = IDLE;
293 }
294 else
295 {
296 Local_u8ErrorState = BUSY_STATE;
297 }
298
299 return Local_u8ErrorState;
