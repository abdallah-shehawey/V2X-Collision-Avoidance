112void System_setup(void)
113{
114
115 // 1. Initialize System Clock (HSI 16MHz)
116 RCC_enumSetSysClk(RCC_HSI_CLK);
117
118 // 2. Enable Peripheral Clocks
119 RCC_enumAHPPerSts(RCC_AHB1, RCC_GPIOAEN, RCC_PER_ON);
120 RCC_enumAHPPerSts(RCC_AHB1, RCC_GPIOBEN, RCC_PER_ON);
121 RCC_enumAHPPerSts(RCC_AHB1, RCC_GPIOCEN, RCC_PER_ON);
122
123 // 3.Enable Timer6 for delay and Timer5 for Background Timestamping
124 RCC_enumABPPerSts(RCC_APB1, RCC_TIM6EN, RCC_PER_ON);
125 RCC_enumABPPerSts(RCC_APB1, RCC_TIM5EN, RCC_PER_ON);
126
127 /* Start TIM5 as a continuous 32-bit background counter (1ms tick) */
128 TIM_Config_t TIM5_Config = {
129 .Timer = TIM_TIMER5,
130 .Prescaler = 16000 - 1,
131 .AutoReloadValue = 0xFFFFFFFF,
132 .Mode = TIM_COUNTERMODE_UP};
133 TIM_vInit(&TIM5_Config);
134 TIM_vStart(TIM_TIMER5);
135
136 /* *
137 * MPU9250 SPI configuration *
138 * */
139 RCC_enumABPPerSts(RCC_APB2, RCC_SPI1EN, RCC_PER_ON);
140 GPIO_PinConfig_t SPI_Pins = {
141 .Port = GPIO_PORTA,
142 .Mode = GPIO_ALTFN,
143 .Otype = GPIO_PUSH_PULL,
144 .Speed = GPIO_VERY_HIGH_SPEED,
145 .PullType = GPIO_NO_PULL,
146 .AlternateFunction = GPIO_AF5};
147 SPI_Pins.PinNum = GPIO_PIN5;
148 GPIO_enumPinInit(&SPI_Pins);
149 SPI_Pins.PinNum = GPIO_PIN6;
150 GPIO_enumPinInit(&SPI_Pins);
151 SPI_Pins.PinNum = GPIO_PIN7;
152 GPIO_enumPinInit(&SPI_Pins);
153
154 MPU9250_enumInit();
155
156 /* *
157 * Ultrasonics configuration *
158 * */
159
160 RCC_enumABPPerSts(RCC_APB1, RCC_TIM2EN, RCC_PER_ON);
161 RCC_enumABPPerSts(RCC_APB1, RCC_TIM3EN, RCC_PER_ON);
162 /* FRONT SENSORS MAPPING:
163 * S1: TIM2 CH1 -> PA15 Echo, PB0 Trig (Left)
164 * S2: TIM2 CH2 -> PB3 Echo, PB1 Trig (Center)
165 * S3: TIM3 CH1 -> PB4 Echo, PB2 Trig (Right)
166 */
167 FrontUS[0] = (US_Config_t){TIM_TIMER2, TIM_CHANNEL1, GPIO_PORTB, GPIO_PIN0, GPIO_PORTA, GPIO_PIN15};
168 US_vInit(&FrontUS[0]);
169
170 FrontUS[1] = (US_Config_t){TIM_TIMER2, TIM_CHANNEL2, GPIO_PORTB, GPIO_PIN1, GPIO_PORTB, GPIO_PIN3};
171 US_vInit(&FrontUS[1]);
172
173 FrontUS[2] = (US_Config_t){TIM_TIMER3, TIM_CHANNEL1, GPIO_PORTB, GPIO_PIN2, GPIO_PORTB, GPIO_PIN4};
174 US_vInit(&FrontUS[2]);
175
176 /* BACK SENSORS MAPPING:
177 * S4: TIM3 CH2 -> PB5 Echo, PB12 Trig (Left)
178 * S5: TIM3 CH3 -> PC8 Echo, PB13 Trig (Center)
179 * S6: TIM3 CH4 -> PC9 Echo, PB14 Trig (Right)
180 */
181 BackUS[0] = (US_Config_t){TIM_TIMER3, TIM_CHANNEL2, GPIO_PORTB, GPIO_PIN12, GPIO_PORTB, GPIO_PIN5};
182 US_vInit(&BackUS[0]);
183
184 BackUS[1] = (US_Config_t){TIM_TIMER3, TIM_CHANNEL3, GPIO_PORTB, GPIO_PIN13, GPIO_PORTC, GPIO_PIN8};
185 US_vInit(&BackUS[1]);
186
187 BackUS[2] = (US_Config_t){TIM_TIMER3, TIM_CHANNEL4, GPIO_PORTB, GPIO_PIN14, GPIO_PORTC, GPIO_PIN9};
188 US_vInit(&BackUS[2]);
189
190 /* *
191 * ESP-NOW / DSRC (V2X) on USART1 (PA9/PA10) *
192 * Interrupt-driven RX (vESP_UART_RX_Callback) + polled TX. *
193 * */
194 RCC_enumABPPerSts(RCC_APB2, RCC_USART1, RCC_PER_ON);
195 GPIO_PinConfig_t U1_Pins = {
196 .Port = GPIO_PORTA,
197 .Mode = GPIO_ALTFN,
198 .Otype = GPIO_PUSH_PULL,
199 .Speed = GPIO_VERY_HIGH_SPEED,
200 .PullType = GPIO_NO_PULL,
201 .AlternateFunction = GPIO_AF7};
202 U1_Pins.PinNum = GPIO_PIN9;
203 GPIO_enumPinInit(&U1_Pins); /* USART1 TX (PA9) */
204 U1_Pins.PinNum = GPIO_PIN10;
205 GPIO_enumPinInit(&U1_Pins); /* USART1 RX (PA10) */
206
207 USART_InitIT(&USART_1); /* USART_1.Channel == CHANNEL1 */
208 NVIC_vSetPriority(NVIC_USART1, 6); /* Safe for FreeRTOS (configMAX_SYSCALL_INTERRUPT_PRIORITY is 5) */
209 NVIC_vEnableIRQ(NVIC_USART1);
210
211 /* *
212 * Raspberry Pi telemetry on UART4 (PA0/PA1) *
213 * Polled TX only (ASCII telemetry line every 100ms). *
214 * */
215 RCC_enumABPPerSts(RCC_APB1, RCC_USART4EN, RCC_PER_ON);
216 GPIO_PinConfig_t U4_Pins = {
217 .Port = GPIO_PORTA,
218 .Mode = GPIO_ALTFN,
219 .Otype = GPIO_PUSH_PULL,
220 .Speed = GPIO_VERY_HIGH_SPEED,
221 .PullType = GPIO_NO_PULL,
222 .AlternateFunction = GPIO_AF8};
223 U4_Pins.PinNum = GPIO_PIN0;
224 GPIO_enumPinInit(&U4_Pins); /* UART4 TX (PA0) */
225 U4_Pins.PinNum = GPIO_PIN1;
226 GPIO_enumPinInit(&U4_Pins); /* UART4 RX (PA1) */
227 USART_Init(&RPi_UART); /* RPi_UART.Channel == CHANNEL4 */
228
229 /* *
230 * init actuato *
231 * */
232 /* Initialize LEDs */
233 LED_Init(&FrontR_LED);
234 LED_Init(&FrontL_LED);
235 LED_Init(&BackR_LED);
236 LED_Init(&BackL_LED);
237 LED_Init(&Interior_LED);
238 /* Initialize Buzzer */
239 BUZ_Init(&V2X_Buzzer);
240
241 /* ── Magnetometer hard-iron calibration (boot-time rotation) ──
242 * One short beep + interior LED ON = "rotate the car ~360° NOW".
243 * Keep turning until the LED goes off (~12s). Then:
244 * two short beeps = calibration OK
245 * one long beep = not enough rotation → power-cycle and retry.
246 * NOTE: runs BEFORE the IWDG is started (in main), so the long window
247 * cannot trip the watchdog. */
248 BUZ_On(&V2X_Buzzer);
249 TIM_vDelayMs(TIM_TIMER6, 400);
250 BUZ_Off(&V2X_Buzzer);
251 LED_TurnOn(&Interior_LED);
252
253 if (MPU9250_enumCalibrateMag() == OK)
254 {
255 LED_TurnOff(&Interior_LED);
256 BUZ_On(&V2X_Buzzer);
257 TIM_vDelayMs(TIM_TIMER6, 150);
258 BUZ_Off(&V2X_Buzzer);
259 TIM_vDelayMs(TIM_TIMER6, 120);
260 BUZ_On(&V2X_Buzzer);
261 TIM_vDelayMs(TIM_TIMER6, 150);
262 BUZ_Off(&V2X_Buzzer);
263 }
264 else
265 {
266 LED_TurnOff(&Interior_LED);
267 BUZ_On(&V2X_Buzzer);
268 TIM_vDelayMs(TIM_TIMER6, 1000);
269 BUZ_Off(&V2X_Buzzer);
270 }
271
272 // DSRC init
273 DSRC_Init();
274}
