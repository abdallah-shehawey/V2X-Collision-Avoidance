131 * System Initialization *

133void System_setup(void)
134{
135
136 // 1. Initialize System Clock (HSI 16MHz)
137 RCC_enumSetSysClk(RCC_HSI_CLK);
138
139 // 2. Enable Peripheral Clocks
140 RCC_enumAHPPerSts(RCC_AHB1, RCC_GPIOAEN, RCC_PER_ON);
141 RCC_enumAHPPerSts(RCC_AHB1, RCC_GPIOBEN, RCC_PER_ON);
142 RCC_enumAHPPerSts(RCC_AHB1, RCC_GPIOCEN, RCC_PER_ON);
143
144 // 3.Enable Timer6 for delay and Timer5 for Background Timestamping
145 RCC_enumABPPerSts(RCC_APB1, RCC_TIM6EN, RCC_PER_ON);
146 RCC_enumABPPerSts(RCC_APB1, RCC_TIM5EN, RCC_PER_ON);
147
148 /* Start TIM5 as a continuous 32-bit background counter (1ms tick) */
149 TIM_Config_t TIM5_Config = {
150 .Timer = TIM_TIMER5,
151 .Prescaler = 16000 - 1,
152 .AutoReloadValue = 0xFFFFFFFF,
153 .Mode = TIM_COUNTERMODE_UP};
154 TIM_vInit(&TIM5_Config);
155 TIM_vStart(TIM_TIMER5);
156
157 /* *
158 * MPU9250 SPI configuration *
159 * */
160 RCC_enumABPPerSts(RCC_APB2, RCC_SPI1EN, RCC_PER_ON);
161 GPIO_PinConfig_t SPI_Pins = {
162 .Port = GPIO_PORTA,
163 .Mode = GPIO_ALTFN,
164 .Otype = GPIO_PUSH_PULL,
165 .Speed = GPIO_VERY_HIGH_SPEED,
166 .PullType = GPIO_NO_PULL,
167 .AlternateFunction = GPIO_AF5};
168 SPI_Pins.PinNum = GPIO_PIN5;
169 GPIO_enumPinInit(&SPI_Pins);
170 SPI_Pins.PinNum = GPIO_PIN6;
171 GPIO_enumPinInit(&SPI_Pins);
172 SPI_Pins.PinNum = GPIO_PIN7;
173 GPIO_enumPinInit(&SPI_Pins);
174
175 MPU9250_enumInit();
176
177 /* *
178 * Ultrasonics configuration *
179 * */
180
181 RCC_enumABPPerSts(RCC_APB1, RCC_TIM2EN, RCC_PER_ON);
182 RCC_enumABPPerSts(RCC_APB1, RCC_TIM3EN, RCC_PER_ON);
183 /* FRONT SENSORS MAPPING:
184 * S1: TIM2 CH1 -> PA15 Echo, PB0 Trig (Left)
185 * S2: TIM2 CH2 -> PB3 Echo, PB1 Trig (Center)
186 * S3: TIM3 CH1 -> PB4 Echo, PB2 Trig (Right)
187 */
188 FrontUS[0] = (US_Config_t){TIM_TIMER2, TIM_CHANNEL1, GPIO_PORTB, GPIO_PIN0, GPIO_PORTA, GPIO_PIN15};
189 US_vInit(&FrontUS[0]);
190
191 FrontUS[1] = (US_Config_t){TIM_TIMER2, TIM_CHANNEL2, GPIO_PORTB, GPIO_PIN1, GPIO_PORTB, GPIO_PIN3};
192 US_vInit(&FrontUS[1]);
193
194 FrontUS[2] = (US_Config_t){TIM_TIMER3, TIM_CHANNEL1, GPIO_PORTB, GPIO_PIN2, GPIO_PORTB, GPIO_PIN4};
195 US_vInit(&FrontUS[2]);
196
197 /* BACK SENSORS MAPPING:
198 * S4: TIM3 CH2 -> PB5 Echo, PB12 Trig (Left)
199 * S5: TIM3 CH3 -> PC8 Echo, PB13 Trig (Center)
200 * S6: TIM3 CH4 -> PC9 Echo, PB14 Trig (Right)
201 */
202 BackUS[0] = (US_Config_t){TIM_TIMER3, TIM_CHANNEL2, GPIO_PORTB, GPIO_PIN12, GPIO_PORTB, GPIO_PIN5};
203 US_vInit(&BackUS[0]);
204
205 BackUS[1] = (US_Config_t){TIM_TIMER3, TIM_CHANNEL3, GPIO_PORTB, GPIO_PIN13, GPIO_PORTC, GPIO_PIN8};
206 US_vInit(&BackUS[1]);
207
208 BackUS[2] = (US_Config_t){TIM_TIMER3, TIM_CHANNEL4, GPIO_PORTB, GPIO_PIN14, GPIO_PORTC, GPIO_PIN9};
209 US_vInit(&BackUS[2]);
210
211 /* *
212 * ESP-NOW / DSRC (V2X) on USART1 (PA9/PA10) *
213 * Interrupt-driven RX (vESP_UART_RX_Callback) + polled TX. *
214 * */
215 RCC_enumABPPerSts(RCC_APB2, RCC_USART1, RCC_PER_ON);
216 GPIO_PinConfig_t U1_Pins = {
217 .Port = GPIO_PORTA,
218 .Mode = GPIO_ALTFN,
219 .Otype = GPIO_PUSH_PULL,
220 .Speed = GPIO_VERY_HIGH_SPEED,
221 .PullType = GPIO_NO_PULL,
222 .AlternateFunction = GPIO_AF7};
223 U1_Pins.PinNum = GPIO_PIN9;
224 GPIO_enumPinInit(&U1_Pins); /* USART1 TX (PA9) */
225 U1_Pins.PinNum = GPIO_PIN10;
226 GPIO_enumPinInit(&U1_Pins); /* USART1 RX (PA10) */
227
228 USART_InitIT(&USART_1); /* USART_1.Channel == CHANNEL1 */
229 NVIC_vSetPriority(NVIC_USART1, 6); /* Safe for FreeRTOS (configMAX_SYSCALL_INTERRUPT_PRIORITY is 5) */
230 NVIC_vEnableIRQ(NVIC_USART1);
231
232 /* *
233 * Raspberry Pi telemetry on UART4 (PA0/PA1) *
234 * Polled TX only (ASCII telemetry line every 100ms). *
235 * */
236 RCC_enumABPPerSts(RCC_APB1, RCC_USART4EN, RCC_PER_ON);
237 GPIO_PinConfig_t U4_Pins = {
238 .Port = GPIO_PORTA,
239 .Mode = GPIO_ALTFN,
240 .Otype = GPIO_PUSH_PULL,
241 .Speed = GPIO_VERY_HIGH_SPEED,
242 .PullType = GPIO_NO_PULL,
243 .AlternateFunction = GPIO_AF8};
244 U4_Pins.PinNum = GPIO_PIN0;
245 GPIO_enumPinInit(&U4_Pins); /* UART4 TX (PA0) */
246 U4_Pins.PinNum = GPIO_PIN1;
247 GPIO_enumPinInit(&U4_Pins); /* UART4 RX (PA1) */
248 USART_Init(&RPi_UART); /* RPi_UART.Channel == CHANNEL4 */
249
250 /* *
251 * init actuato *
252 * */
253 /* Initialize LEDs */
254 LED_Init(&FrontR_LED);
255 LED_Init(&FrontL_LED);
256 LED_Init(&BackR_LED);
257 LED_Init(&BackL_LED);
258 LED_Init(&Interior_LED);
259 /* Initialize Buzzer */
260 BUZ_Init(&V2X_Buzzer);
261
262 /* ── Magnetometer hard-iron calibration (boot-time rotation) ──
263 * One short beep + interior LED ON = "rotate the car ~360° NOW".
264 * Keep turning until the LED goes off (~12s). Then:
265 * two short beeps = calibration OK
266 * one long beep = not enough rotation → power-cycle and retry.
267 * NOTE: runs BEFORE the IWDG is started (in main), so the long window
268 * cannot trip the watchdog. */
269 BUZ_On(&V2X_Buzzer);
270 TIM_vDelayMs(TIM_TIMER6, 400);
271 BUZ_Off(&V2X_Buzzer);
272 LED_TurnOn(&Interior_LED);
273
274 if (MPU9250_enumCalibrateMag() == OK)
275 {
276 LED_TurnOff(&Interior_LED);
277 BUZ_On(&V2X_Buzzer);
278 TIM_vDelayMs(TIM_TIMER6, 150);
279 BUZ_Off(&V2X_Buzzer);
280 TIM_vDelayMs(TIM_TIMER6, 120);
281 BUZ_On(&V2X_Buzzer);
282 TIM_vDelayMs(TIM_TIMER6, 150);
283 BUZ_Off(&V2X_Buzzer);
284 }
285 else
286 {
287 LED_TurnOff(&Interior_LED);
288 BUZ_On(&V2X_Buzzer);
289 TIM_vDelayMs(TIM_TIMER6, 1000);
290 BUZ_Off(&V2X_Buzzer);
291 }
292
293 // DSRC init
294 DSRC_Init();
295}
