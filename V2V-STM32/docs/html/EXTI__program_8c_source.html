27/*
28 * @brief Initialize EXTI line with configuration
29 * @details Configures EXTI line based on provided configuration structure
30 * - Validates configuration parameters
31 * - Sets trigger source
32 * - Configures interrupt enable/disable state
33 * - Stores callback function
34 * @param Copy_pEXTI_LineConfig - Pointer to EXTI configuration structure
35 * @return ErrorState_t - Error state of the operation
36 * OK - Operation successful
37 * NULL_POINTER - Invalid configuration pointer or callback
38 * NOK - Invalid EXTI line number
39 * @note Must be called after SYSCFG configuration
40 */
41ErrorState_t EXTI_vLineInit(const EXTI_LineConfig_t *Copy_pEXTI_LineConfig)
42{
43 ErrorState_t Local_u8ErrorState = OK;
44 if (Copy_pEXTI_LineConfig == NULL && Copy_pEXTI_LineConfig->Copy_pvCallBack == NULL)
45 {
46 Local_u8ErrorState = NULL_POINTER;
47 }
48 else if (Copy_pEXTI_LineConfig->Line > EXTI_LINE15)
49 {
50 Local_u8ErrorState = NOK;
51 }
52 else
53 {
54 switch (Copy_pEXTI_LineConfig->TrigSrc)
55 {
56 case EXTI_RISING_EDGE:
57 MEXTI->RTSR |= (1 << Copy_pEXTI_LineConfig->Line);
58 MEXTI->FTSR &= ~(1 << Copy_pEXTI_LineConfig->Line);
59 break;
60 case EXTI_FALLING_EDGE:
61 MEXTI->FTSR |= (1 << Copy_pEXTI_LineConfig->Line);
62 MEXTI->RTSR &= ~(1 << Copy_pEXTI_LineConfig->Line);
63 break;
64 case EXTI_RISING_FALLING_EDGE:
65 MEXTI->RTSR |= (1 << Copy_pEXTI_LineConfig->Line);
66 MEXTI->FTSR |= (1 << Copy_pEXTI_LineConfig->Line);
67 break;
68 case EXTI_NO_TRIGGER:
69 MEXTI->RTSR &= ~(1 << Copy_pEXTI_LineConfig->Line);
70 MEXTI->FTSR &= ~(1 << Copy_pEXTI_LineConfig->Line);
71 break;
72 }
73
74 switch (Copy_pEXTI_LineConfig->Enable)
75 {
76 case EXTI_EN:
77 MEXTI->IMR |= (1 << Copy_pEXTI_LineConfig->Line);
78 break;
79 case EXTI_DIS:
80 MEXTI->IMR &= ~(1 << Copy_pEXTI_LineConfig->Line);
81 break;
82 }
83
84 EXTI_CallBack[Copy_pEXTI_LineConfig->Line] = Copy_pEXTI_LineConfig->Copy_pvCallBack;
85 }
86 return Local_u8ErrorState;
87}

112/*
113 * @brief Set EXTI trigger source
114 * @details Configures how the EXTI line is triggered by setting/clearing RTSR and FTSR bits
115 * @param Copy_pEXTI_LineConfig - Pointer to EXTI configuration structure
116 * Copy_u8Trigger - New trigger source configuration
117 * @return ErrorState_t - Error state of the operation
118 * OK - Operation successful
119 * NULL_POINTER - Invalid configuration pointer
120 * OUT_OF_RANGE - Invalid trigger source
121 */
122ErrorState_t EXTI_vSetTrigSrc(const EXTI_LineConfig_t *Copy_pEXTI_LineConfig, EXTI_TriggerSrc_t Copy_u8Trigger)
123{
124 ErrorState_t Local_u8ErrorState = OK;
125 if (Copy_pEXTI_LineConfig == NULL)
126 {
127 Local_u8ErrorState = NULL_POINTER;
128 }
129 else
130 {
131 switch (Copy_u8Trigger)
132 {
133 case EXTI_NO_TRIGGER:
134 MEXTI->RTSR &= ~(1 << Copy_pEXTI_LineConfig->Line);
135 MEXTI->FTSR &= ~(1 << Copy_pEXTI_LineConfig->Line);
136 break;
137 case EXTI_RISING_EDGE:
138 MEXTI->RTSR |= (1 << Copy_pEXTI_LineConfig->Line);
139 MEXTI->FTSR &= ~(1 << Copy_pEXTI_LineConfig->Line);
140 break;
141 case EXTI_FALLING_EDGE:
142 MEXTI->RTSR &= ~(1 << Copy_pEXTI_LineConfig->Line);
143 MEXTI->FTSR |= (1 << Copy_pEXTI_LineConfig->Line);
144 break;
145 case EXTI_RISING_FALLING_EDGE:
146 MEXTI->RTSR |= (1 << Copy_pEXTI_LineConfig->Line);
147 MEXTI->FTSR |= (1 << Copy_pEXTI_LineConfig->Line);
148 break;
149 default:
150 break;
151 }
152 return Local_u8ErrorState;
153 }
154}

214/*=================================================================================================================*/
215/*
216 * @brief Read EXTI pending flag status
217 * @details Reads the status of the pending flag for specified EXTI line from PR register
218 * @param Copy_u8Line - EXTI line number
219 * @return uint8_t - Status of the pending flag (0: not pending, 1: pending)
220 */
221uint8_t EXTI_u8ReadPendFlag(EXTI_Line_t Copy_u8Line)
222{
223 return ((MEXTI->PR & (1 << Copy_u8Line)) >> Copy_u8Line);
