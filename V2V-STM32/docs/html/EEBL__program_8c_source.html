37/*
38 * @brief Start a new cycle: set the sudden-braking gate and reset the result.
39 * While not braking, ProcessNeighbor skips every neighbor.
40 */
41void EEBL_voidBeginCycle(void)
42{
43 /* Sudden braking = a drop in speed since the last cycle. */
44 float decel = Host_Speed - EEBL_PrevSpeed;
45 EEBL_BrakingDetected = (decel <= EEBL_DECEL_THRESHOLD) ? 1U : 0U;
46
47 EEBL_PrevSpeed = Host_Speed;
48 EEBL_WorstLevel = RISK_SAFE;
49}

51/*
52 * @brief Evaluate one same-direction neighbor against the rear gap.
53 *
54 * Risk depends on the host's own speed (the cycle safe/critical gaps), not on
55 * relative speed: a sudden brake with a vehicle close behind raises a warning.
56 * Keeps the worst severity across the cycle.
57 *
58 * @param rear_distance Rear ultrasonic distance (cm)
59 */
60void EEBL_voidProcessNeighbor(float rear_distance)
61{
62 if (!EEBL_BrakingDetected)
63 {
64 return;
65 }
66
67 /* Nothing behind, or beyond the safe gap (which is RISK_SAFE anyway). */
68 if (rear_distance <= 0.0f || rear_distance >= SafetyEngine_SafeDist)
69 {
70 return;
71 }
72
73 RiskLevel_t level = (rear_distance < SafetyEngine_CriticalDist) ? RISK_CRITICAL : RISK_WARNING;
74
75 if (level > EEBL_WorstLevel)
76 {
77 EEBL_WorstLevel = level;
78 }
79}
