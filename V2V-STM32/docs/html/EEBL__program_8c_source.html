42void EEBL_voidBeginCycle(void)

61void EEBL_voidProcessNeighbor(float rear_distance)
62{

74 RiskLevel_t level = (rear_distance < SafetyEngine_CriticalDist) ? RISK_CRITICAL : RISK_WARNING;
