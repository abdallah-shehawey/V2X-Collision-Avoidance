40ErrorState_t TIM_vInit(const TIM_Config_t *pxConfig)
41{
42 ErrorState_t Local_ErrorState = OK;
43
44 if (pxConfig == NULL)
45 {
46 Local_ErrorState = NULL_POINTER;
47 }
48 else if (pxConfig->Timer >= TIM_TIMER_COUNT)
49 {
50 Local_ErrorState = NOK;
51 }
52 else
53 {
54 TIM_TypeDef *TIMx = TIM_Instances[pxConfig->Timer];
55
56 /* Disable Timer during configuration */
57 CLR_BIT(TIMx->CR1, TIM_CR1_CEN);
58
59 /* Set Prescaler */
60 TIMx->PSC = pxConfig->Prescaler;
61
62 /* Set Auto-Reload Value */
63 TIMx->ARR = pxConfig->AutoReloadValue;
64
65 /* Set Counter Mode */
66 if (pxConfig->Mode == TIM_COUNTERMODE_UP)
67 {
68 CLR_BIT(TIMx->CR1, TIM_CR1_DIR);
69 }
70 else
71 {
72 SET_BIT(TIMx->CR1, TIM_CR1_DIR);
73 }
74
75 /* Generate Update Event to load PSC and ARR immediately */
76 SET_BIT(TIMx->EGR, TIM_EGR_UG);
77 /* Clear Update Interrupt Flag to avoid immediate interrupt */
78 CLR_BIT(TIMx->SR, TIM_SR_UIF);
79 }
80
81 return Local_ErrorState;
82}

128ErrorState_t TIM_vPWM_Init(const TIM_PWMConfig_t *pxPWMConfig)
129{
130 ErrorState_t Local_ErrorState = OK;
131
132 if (pxPWMConfig == NULL)
133 {
134 Local_ErrorState = NULL_POINTER;
135 }
136 else if (pxPWMConfig->Timer >= TIM_TIMER_COUNT)
137 {
138 Local_ErrorState = NOK;
139 }
140 else
141 {
142 TIM_TypeDef *TIMx = TIM_Instances[pxPWMConfig->Timer];
143
144 /* Disable Timer */
145 CLR_BIT(TIMx->CR1, TIM_CR1_CEN);
146
147 /* Set Prescaler and Period */
148 TIMx->PSC = pxPWMConfig->Prescaler;
149 TIMx->ARR = pxPWMConfig->Period;
150
151 /* Configure Channel */
152 switch (pxPWMConfig->Channel)
153 {
154 case TIM_CHANNEL1:
155 /* CC1S = 00 (Output) */
156 TIMx->CCMR1 &= ~(0x3 << TIM_CCMR1_CC1S);
157 /* OC1M = Mode (PWM1 or PWM2) */
158 TIMx->CCMR1 &= ~(0x7 << TIM_CCMR1_OC1M);
159 TIMx->CCMR1 |= (pxPWMConfig->Mode << TIM_CCMR1_OC1M);
160 /* Enable Preload */
161 SET_BIT(TIMx->CCMR1, TIM_CCMR1_OC1PE);
162 /* Set Compare Value */
163 TIMx->CCR1 = pxPWMConfig->DutyCycle;
164 /* Polarity */
165 if (pxPWMConfig->Polarity == TIM_POLARITY_LOW)
166 SET_BIT(TIMx->CCER, TIM_CCER_CC1P);
167 else
168 CLR_BIT(TIMx->CCER, TIM_CCER_CC1P);
169 /* Enable Output */
170 SET_BIT(TIMx->CCER, TIM_CCER_CC1E);
171 break;
172
173 case TIM_CHANNEL2:
174 TIMx->CCMR1 &= ~(0x3 << TIM_CCMR1_CC2S);
175 TIMx->CCMR1 &= ~(0x7 << TIM_CCMR1_OC2M);
176 TIMx->CCMR1 |= (pxPWMConfig->Mode << TIM_CCMR1_OC2M);
177 SET_BIT(TIMx->CCMR1, TIM_CCMR1_OC2PE);
178 TIMx->CCR2 = pxPWMConfig->DutyCycle;
179 if (pxPWMConfig->Polarity == TIM_POLARITY_LOW)
180 SET_BIT(TIMx->CCER, TIM_CCER_CC2P);
181 else
182 CLR_BIT(TIMx->CCER, TIM_CCER_CC2P);
183 SET_BIT(TIMx->CCER, TIM_CCER_CC2E);
184 break;
185
186 case TIM_CHANNEL3:
187 TIMx->CCMR2 &= ~(0x3 << TIM_CCMR2_CC3S);
188 TIMx->CCMR2 &= ~(0x7 << TIM_CCMR2_OC3M);
189 TIMx->CCMR2 |= (pxPWMConfig->Mode << TIM_CCMR2_OC3M);
190 SET_BIT(TIMx->CCMR2, TIM_CCMR2_OC3PE);
191 TIMx->CCR3 = pxPWMConfig->DutyCycle;
192 if (pxPWMConfig->Polarity == TIM_POLARITY_LOW)
193 SET_BIT(TIMx->CCER, TIM_CCER_CC3P);
194 else
195 CLR_BIT(TIMx->CCER, TIM_CCER_CC3P);
196 SET_BIT(TIMx->CCER, TIM_CCER_CC3E);
197 break;
198
199 case TIM_CHANNEL4:
200 TIMx->CCMR2 &= ~(0x3 << TIM_CCMR2_CC4S);
201 TIMx->CCMR2 &= ~(0x7 << TIM_CCMR2_OC4M);
202 TIMx->CCMR2 |= (pxPWMConfig->Mode << TIM_CCMR2_OC4M);
203 SET_BIT(TIMx->CCMR2, TIM_CCMR2_OC4PE);
204 TIMx->CCR4 = pxPWMConfig->DutyCycle;
205 if (pxPWMConfig->Polarity == TIM_POLARITY_LOW)
206 SET_BIT(TIMx->CCER, TIM_CCER_CC4P);
207 else
208 CLR_BIT(TIMx->CCER, TIM_CCER_CC4P);
209 SET_BIT(TIMx->CCER, TIM_CCER_CC4E);
210 break;
211 }
212
213 /* Auto-reload Preload Enable */
214 SET_BIT(TIMx->CR1, TIM_CR1_ARPE);
215
216 /* Generate Update Event */
217 SET_BIT(TIMx->EGR, TIM_EGR_UG);
218 }
219
220 return Local_ErrorState;
221}

240 {

242
243 /* Disable Timer */
244 CLR_BIT(TIMx->CR1, TIM_CR1_CEN);
245
246 switch (pxICConfig->Channel)
247 {
248 case TIM_CHANNEL1:
249 /* CC1S = Selection (Direct/Indirect/TRC) */
250 TIMx->CCMR1 &= ~(0x3 << TIM_CCMR1_CC1S);
251 TIMx->CCMR1 |= (pxICConfig->Selection << TIM_CCMR1_CC1S);
252 /* IC1F = Filter */
253 TIMx->CCMR1 &= ~(0xF << 4); /* Filter bits at offset 4 */
254 TIMx->CCMR1 |= (pxICConfig->Filter << 4);
255 /* IC1PSC = Prescaler */
256 TIMx->CCMR1 &= ~(0x3 << 2); /* Prescaler bits at offset 2 */
257 TIMx->CCMR1 |= (pxICConfig->Prescaler << 2);
258 /* CC1P/CC1NP = Polarity */
259 CLR_BIT(TIMx->CCER, TIM_CCER_CC1P);
260 CLR_BIT(TIMx->CCER, TIM_CCER_CC1NP);
261 if (pxICConfig->Polarity == TIM_POLARITY_LOW)
262 {
263 SET_BIT(TIMx->CCER, TIM_CCER_CC1P);
264 }
265 /* Enable Capture */
266 SET_BIT(TIMx->CCER, TIM_CCER_CC1E);
267 break;
268
269 case TIM_CHANNEL2:
270 TIMx->CCMR1 &= ~(0x3 << TIM_CCMR1_CC2S);
271 TIMx->CCMR1 |= (pxICConfig->Selection << TIM_CCMR1_CC2S);
272 TIMx->CCMR1 &= ~(0xF << 12);
273 TIMx->CCMR1 |= (pxICConfig->Filter << 12);
274 TIMx->CCMR1 &= ~(0x3 << 10);
275 TIMx->CCMR1 |= (pxICConfig->Prescaler << 10);
276 CLR_BIT(TIMx->CCER, TIM_CCER_CC2P);
277 CLR_BIT(TIMx->CCER, TIM_CCER_CC2NP);
278 if (pxICConfig->Polarity == TIM_POLARITY_LOW)
279 {
280 SET_BIT(TIMx->CCER, TIM_CCER_CC2P);
281 }
282 SET_BIT(TIMx->CCER, TIM_CCER_CC2E);
283 break;
284
285 case TIM_CHANNEL3:
286 TIMx->CCMR2 &= ~(0x3 << TIM_CCMR2_CC3S);
287 TIMx->CCMR2 |= (pxICConfig->Selection << TIM_CCMR2_CC3S);
288 TIMx->CCMR2 &= ~(0xF << 4);
289 TIMx->CCMR2 |= (pxICConfig->Filter << 4);
290 TIMx->CCMR2 &= ~(0x3 << 2);
291 TIMx->CCMR2 |= (pxICConfig->Prescaler << 2);
292 CLR_BIT(TIMx->CCER, TIM_CCER_CC3P);
293 CLR_BIT(TIMx->CCER, TIM_CCER_CC3NP);
294 if (pxICConfig->Polarity == TIM_POLARITY_LOW)
295 {
296 SET_BIT(TIMx->CCER, TIM_CCER_CC3P);
297 }
298 SET_BIT(TIMx->CCER, TIM_CCER_CC3E);
299 break;
300
301 case TIM_CHANNEL4:
302 TIMx->CCMR2 &= ~(0x3 << TIM_CCMR2_CC4S);
303 TIMx->CCMR2 |= (pxICConfig->Selection << TIM_CCMR2_CC4S);
304 TIMx->CCMR2 &= ~(0xF << 12);
305 TIMx->CCMR2 |= (pxICConfig->Filter << 12);
306 TIMx->CCMR2 &= ~(0x3 << 10);
307 TIMx->CCMR2 |= (pxICConfig->Prescaler << 10);
308 CLR_BIT(TIMx->CCER, TIM_CCER_CC4P);
309 CLR_BIT(TIMx->CCER, TIM_CCER_CC4NP);
310 if (pxICConfig->Polarity == TIM_POLARITY_LOW)
311 {
312 SET_BIT(TIMx->CCER, TIM_CCER_CC4P);
313 }
314 SET_BIT(TIMx->CCER, TIM_CCER_CC4E);
315 break;
316 }
317 }
318 return Local_ErrorState;

434ErrorState_t TIM_vDelayMs(TIM_Num_t Copy_eTimer, uint32_t Copy_u32Ms)
435{
436 ErrorState_t Local_ErrorState = OK;
437
438 if (Copy_eTimer >= TIM_TIMER_COUNT)
439 {
440 Local_ErrorState = NOK;
441 }
442 else
443 {
444 TIM_TypeDef *TIMx = TIM_Instances[Copy_eTimer];
445 uint32_t Local_u32Prescaler = (TIM_CLOCK_FREQ / 1000) - 1; /* 1KHz Timer Frequency */
446
447 /* Configure Timer for 1ms tick */
448 TIMx->PSC = Local_u32Prescaler;
449 TIMx->ARR = Copy_u32Ms; /* Set Auto-Reload to Delay Ms */
450 TIMx->CNT = 0;
451
452 /* Clear Update Flag and Disable Interrupts for this timer during blocking delay */
453 CLR_BIT(TIMx->SR, TIM_SR_UIF);
454 CLR_BIT(TIMx->DIER, TIM_DIER_UIE);
455
456 /* Enable Timer */
457 SET_BIT(TIMx->CR1, TIM_CR1_CEN);
458
459 /* Wait for Update Flag */
460 while (READ_BIT(TIMx->SR, TIM_SR_UIF) == 0);
461
462 /* Disable Timer */
463 CLR_BIT(TIMx->CR1, TIM_CR1_CEN);
464 CLR_BIT(TIMx->SR, TIM_SR_UIF);
465 }
466 return Local_ErrorState;
467}

552/* Reading CCRx clears the corresponding CCxIF flag automatically (RM0390). */
553static void TIM_voidHandleCC(TIM_TypeDef *TIMx, TIM_Num_t Copy_eTimer)
554{
555 void (*cb)(TIM_Num_t, uint8_t, uint32_t) = TIM_CC_Callbacks[Copy_eTimer];
556 uint32_t sr = TIMx->SR;
557 uint32_t dier = TIMx->DIER;
558
559 if ((sr & (1u << TIM_SR_CC1IF)) && (dier & (1u << TIM_DIER_CC1IE)))
560 {
561 uint32_t cap = TIMx->CCR1;
562 if (cb) cb(Copy_eTimer, 0u, cap);
563 }
564 if ((sr & (1u << TIM_SR_CC2IF)) && (dier & (1u << TIM_DIER_CC2IE)))
565 {
566 uint32_t cap = TIMx->CCR2;
567 if (cb) cb(Copy_eTimer, 1u, cap);
568 }
569 if ((sr & (1u << TIM_SR_CC3IF)) && (dier & (1u << TIM_DIER_CC3IE)))
570 {
571 uint32_t cap = TIMx->CCR3;
572 if (cb) cb(Copy_eTimer, 2u, cap);
573 }
574 if ((sr & (1u << TIM_SR_CC4IF)) && (dier & (1u << TIM_DIER_CC4IE)))
575 {
576 uint32_t cap = TIMx->CCR4;
577 if (cb) cb(Copy_eTimer, 3u, cap);
578 }
579}
