53/*
54 * @brief Initialize the Timer with the specified configuration.
55 */
56ErrorState_t TIM_vInit(const TIM_Config_t *pxConfig)
57{
58 ErrorState_t Local_ErrorState = OK;
59
60 if (pxConfig == NULL)
61 {
62 Local_ErrorState = NULL_POINTER;
63 }
64 else if (pxConfig->Timer >= TIM_TIMER_COUNT)
65 {
66 Local_ErrorState = NOK;
67 }
68 else
69 {
70 TIM_TypeDef *TIMx = TIM_Instances[pxConfig->Timer];
71
72 /* Disable Timer during configuration */
73 CLR_BIT(TIMx->CR1, TIM_CR1_CEN);
74
75 /* Set Prescaler */
76 TIMx->PSC = pxConfig->Prescaler;
77
78 /* Set Auto-Reload Value */
79 TIMx->ARR = pxConfig->AutoReloadValue;
80
81 /* Set Counter Mode */
82 if (pxConfig->Mode == TIM_COUNTERMODE_UP)
83 {
84 CLR_BIT(TIMx->CR1, TIM_CR1_DIR);
85 }
86 else
87 {
88 SET_BIT(TIMx->CR1, TIM_CR1_DIR);
89 }
90
91 /* Generate Update Event to load PSC and ARR immediately */
92 SET_BIT(TIMx->EGR, TIM_EGR_UG);
93 /* Clear Update Interrupt Flag to avoid immediate interrupt */
94 CLR_BIT(TIMx->SR, TIM_SR_UIF);
95 }
96
97 return Local_ErrorState;
98}

138/*
139 * @brief Initialize a PWM channel.
140 */
141ErrorState_t TIM_vPWM_Init(const TIM_PWMConfig_t *pxPWMConfig)
142{
143 ErrorState_t Local_ErrorState = OK;
144
145 if (pxPWMConfig == NULL)
146 {
147 Local_ErrorState = NULL_POINTER;
148 }
149 else if (pxPWMConfig->Timer >= TIM_TIMER_COUNT)
150 {
151 Local_ErrorState = NOK;
152 }
153 else
154 {
155 TIM_TypeDef *TIMx = TIM_Instances[pxPWMConfig->Timer];
156
157 /* Disable Timer */
158 CLR_BIT(TIMx->CR1, TIM_CR1_CEN);
159
160 /* Set Prescaler and Period */
161 TIMx->PSC = pxPWMConfig->Prescaler;
162 TIMx->ARR = pxPWMConfig->Period;
163
164 /* Configure Channel */
165 switch (pxPWMConfig->Channel)
166 {
167 case TIM_CHANNEL1:
168 /* CC1S = 00 (Output) */
169 TIMx->CCMR1 &= ~(0x3 << TIM_CCMR1_CC1S);
170 /* OC1M = Mode (PWM1 or PWM2) */
171 TIMx->CCMR1 &= ~(0x7 << TIM_CCMR1_OC1M);
172 TIMx->CCMR1 |= (pxPWMConfig->Mode << TIM_CCMR1_OC1M);
173 /* Enable Preload */
174 SET_BIT(TIMx->CCMR1, TIM_CCMR1_OC1PE);
175 /* Set Compare Value */
176 TIMx->CCR1 = pxPWMConfig->DutyCycle;
177 /* Polarity */
178 if (pxPWMConfig->Polarity == TIM_POLARITY_LOW)
179 SET_BIT(TIMx->CCER, TIM_CCER_CC1P);
180 else
181 CLR_BIT(TIMx->CCER, TIM_CCER_CC1P);
182 /* Enable Output */
183 SET_BIT(TIMx->CCER, TIM_CCER_CC1E);
184 break;
185
186 case TIM_CHANNEL2:
187 TIMx->CCMR1 &= ~(0x3 << TIM_CCMR1_CC2S);
188 TIMx->CCMR1 &= ~(0x7 << TIM_CCMR1_OC2M);
189 TIMx->CCMR1 |= (pxPWMConfig->Mode << TIM_CCMR1_OC2M);
190 SET_BIT(TIMx->CCMR1, TIM_CCMR1_OC2PE);
191 TIMx->CCR2 = pxPWMConfig->DutyCycle;
192 if (pxPWMConfig->Polarity == TIM_POLARITY_LOW)
193 SET_BIT(TIMx->CCER, TIM_CCER_CC2P);
194 else
195 CLR_BIT(TIMx->CCER, TIM_CCER_CC2P);
196 SET_BIT(TIMx->CCER, TIM_CCER_CC2E);
197 break;
198
199 case TIM_CHANNEL3:
200 TIMx->CCMR2 &= ~(0x3 << TIM_CCMR2_CC3S);
201 TIMx->CCMR2 &= ~(0x7 << TIM_CCMR2_OC3M);
202 TIMx->CCMR2 |= (pxPWMConfig->Mode << TIM_CCMR2_OC3M);
203 SET_BIT(TIMx->CCMR2, TIM_CCMR2_OC3PE);
204 TIMx->CCR3 = pxPWMConfig->DutyCycle;
205 if (pxPWMConfig->Polarity == TIM_POLARITY_LOW)
206 SET_BIT(TIMx->CCER, TIM_CCER_CC3P);
207 else
208 CLR_BIT(TIMx->CCER, TIM_CCER_CC3P);
209 SET_BIT(TIMx->CCER, TIM_CCER_CC3E);
210 break;
211
212 case TIM_CHANNEL4:
213 TIMx->CCMR2 &= ~(0x3 << TIM_CCMR2_CC4S);
214 TIMx->CCMR2 &= ~(0x7 << TIM_CCMR2_OC4M);
215 TIMx->CCMR2 |= (pxPWMConfig->Mode << TIM_CCMR2_OC4M);
216 SET_BIT(TIMx->CCMR2, TIM_CCMR2_OC4PE);
217 TIMx->CCR4 = pxPWMConfig->DutyCycle;
218 if (pxPWMConfig->Polarity == TIM_POLARITY_LOW)
219 SET_BIT(TIMx->CCER, TIM_CCER_CC4P);
220 else
221 CLR_BIT(TIMx->CCER, TIM_CCER_CC4P);
222 SET_BIT(TIMx->CCER, TIM_CCER_CC4E);
223 break;
224 }
225
226 /* Auto-reload Preload Enable */
227 SET_BIT(TIMx->CR1, TIM_CR1_ARPE);
228
229 /* Generate Update Event */
230 SET_BIT(TIMx->EGR, TIM_EGR_UG);
231 }
232
233 return Local_ErrorState;
234}

242

244 {
245 Local_ErrorState = NULL_POINTER;
246 }

248 {
249 Local_ErrorState = NOK;
250 }
251 else
252 {
253 TIM_TypeDef *TIMx = TIM_Instances[pxICConfig->Timer];
254
255 /* Disable Timer */
256 CLR_BIT(TIMx->CR1, TIM_CR1_CEN);
257
258 switch (pxICConfig->Channel)
259 {
260 case TIM_CHANNEL1:
261 /* CC1S = Selection (Direct/Indirect/TRC) */
262 TIMx->CCMR1 &= ~(0x3 << TIM_CCMR1_CC1S);
263 TIMx->CCMR1 |= (pxICConfig->Selection << TIM_CCMR1_CC1S);
264 /* IC1F = Filter */
265 TIMx->CCMR1 &= ~(0xF << 4); /* Filter bits at offset 4 */
266 TIMx->CCMR1 |= (pxICConfig->Filter << 4);
267 /* IC1PSC = Prescaler */
268 TIMx->CCMR1 &= ~(0x3 << 2); /* Prescaler bits at offset 2 */
269 TIMx->CCMR1 |= (pxICConfig->Prescaler << 2);
270 /* CC1P/CC1NP = Polarity */
271 CLR_BIT(TIMx->CCER, TIM_CCER_CC1P);
272 CLR_BIT(TIMx->CCER, TIM_CCER_CC1NP);
273 if (pxICConfig->Polarity == TIM_POLARITY_LOW)
274 {
275 SET_BIT(TIMx->CCER, TIM_CCER_CC1P);
276 }
277 /* Enable Capture */
278 SET_BIT(TIMx->CCER, TIM_CCER_CC1E);
279 break;
280
281 case TIM_CHANNEL2:
282 TIMx->CCMR1 &= ~(0x3 << TIM_CCMR1_CC2S);
283 TIMx->CCMR1 |= (pxICConfig->Selection << TIM_CCMR1_CC2S);
284 TIMx->CCMR1 &= ~(0xF << 12);
285 TIMx->CCMR1 |= (pxICConfig->Filter << 12);
286 TIMx->CCMR1 &= ~(0x3 << 10);
287 TIMx->CCMR1 |= (pxICConfig->Prescaler << 10);
288 CLR_BIT(TIMx->CCER, TIM_CCER_CC2P);
289 CLR_BIT(TIMx->CCER, TIM_CCER_CC2NP);
290 if (pxICConfig->Polarity == TIM_POLARITY_LOW)
291 {
292 SET_BIT(TIMx->CCER, TIM_CCER_CC2P);
293 }
294 SET_BIT(TIMx->CCER, TIM_CCER_CC2E);
295 break;
296
297 case TIM_CHANNEL3:
298 TIMx->CCMR2 &= ~(0x3 << TIM_CCMR2_CC3S);
299 TIMx->CCMR2 |= (pxICConfig->Selection << TIM_CCMR2_CC3S);
300 TIMx->CCMR2 &= ~(0xF << 4);
301 TIMx->CCMR2 |= (pxICConfig->Filter << 4);
302 TIMx->CCMR2 &= ~(0x3 << 2);
303 TIMx->CCMR2 |= (pxICConfig->Prescaler << 2);
304 CLR_BIT(TIMx->CCER, TIM_CCER_CC3P);
305 CLR_BIT(TIMx->CCER, TIM_CCER_CC3NP);

311 break;

314 TIMx->CCMR2 &= ~(0x3 << TIM_CCMR2_CC4S);
315 TIMx->CCMR2 |= (pxICConfig->Selection << TIM_CCMR2_CC4S);
316 TIMx->CCMR2 &= ~(0xF << 12);
317 TIMx->CCMR2 |= (pxICConfig->Filter << 12);
318 TIMx->CCMR2 &= ~(0x3 << 10);
319 TIMx->CCMR2 |= (pxICConfig->Prescaler << 10);
320 CLR_BIT(TIMx->CCER, TIM_CCER_CC4P);

323 {
324 SET_BIT(TIMx->CCER, TIM_CCER_CC4P);
325 }
326 SET_BIT(TIMx->CCER, TIM_CCER_CC4E);
327 break;
328 }
329 }
330 return Local_ErrorState;

438/*
439 * @brief Blocking Delay in ms.
440 * @note Assumes 16MHz clock if not configured.
441 */
442ErrorState_t TIM_vDelayMs(TIM_Num_t Copy_eTimer, uint32_t Copy_u32Ms)
443{
444 ErrorState_t Local_ErrorState = OK;
445
446 if (Copy_eTimer >= TIM_TIMER_COUNT)
447 {
448 Local_ErrorState = NOK;
449 }
450 else
451 {
452 TIM_TypeDef *TIMx = TIM_Instances[Copy_eTimer];
453 uint32_t Local_u32Prescaler = (TIM_CLOCK_FREQ / 1000) - 1; /* 1KHz Timer Frequency */
454
455 /* Configure Timer for 1ms tick */
456 TIMx->PSC = Local_u32Prescaler;
457 TIMx->ARR = Copy_u32Ms; /* Set Auto-Reload to Delay Ms */
458 TIMx->CNT = 0;
459
460 /* Clear Update Flag and Disable Interrupts for this timer during blocking delay */
461 CLR_BIT(TIMx->SR, TIM_SR_UIF);
462 CLR_BIT(TIMx->DIER, TIM_DIER_UIE);
463
464 /* Enable Timer */
465 SET_BIT(TIMx->CR1, TIM_CR1_CEN);
466
467 /* Wait for Update Flag */
468 while (READ_BIT(TIMx->SR, TIM_SR_UIF) == 0);
469
470 /* Disable Timer */
471 CLR_BIT(TIMx->CR1, TIM_CR1_CEN);
472 CLR_BIT(TIMx->SR, TIM_SR_UIF);
473 }
474 return Local_ErrorState;
475}

567static void TIM_voidHandleCC(TIM_TypeDef *TIMx, TIM_Num_t Copy_eTimer)
568{
569 void (*cb)(TIM_Num_t, uint8_t, uint32_t) = TIM_CC_Callbacks[Copy_eTimer];
570 uint32_t sr = TIMx->SR;
571 uint32_t dier = TIMx->DIER;
572
573 if ((sr & (1u << TIM_SR_CC1IF)) && (dier & (1u << TIM_DIER_CC1IE)))
574 {
575 uint32_t cap = TIMx->CCR1;
576 if (cb) cb(Copy_eTimer, 0u, cap);
577 }
578 if ((sr & (1u << TIM_SR_CC2IF)) && (dier & (1u << TIM_DIER_CC2IE)))
579 {
580 uint32_t cap = TIMx->CCR2;
581 if (cb) cb(Copy_eTimer, 1u, cap);
582 }
583 if ((sr & (1u << TIM_SR_CC3IF)) && (dier & (1u << TIM_DIER_CC3IE)))
584 {
585 uint32_t cap = TIMx->CCR3;
586 if (cb) cb(Copy_eTimer, 2u, cap);
587 }
588 if ((sr & (1u << TIM_SR_CC4IF)) && (dier & (1u << TIM_DIER_CC4IE)))
589 {
590 uint32_t cap = TIMx->CCR4;
591 if (cb) cb(Copy_eTimer, 3u, cap);
592 }
593}
