41/* ============ Single-Pass Update ============ */
42/*
43 * One pass over the neighbor table:
44 * 1. Latch host state + this cycle's safe/critical gaps
45 * 2. BeginCycle → each module resets its per-cycle state
46 * 3. For each neighbor → dispatch to the modules its direction affects
47 * 4. Aggregate every module's result into the G_u16SystemFlags status word
48 */
49void SafetyEngine_voidUpdate(void)
50{
51 Neighbor *table = DSRC_GetTable();
52 uint8_t count = DSRC_GetCount();
53
54 /* Latch host state. Speed is stored as cm/s in G_stHostVehicleState; the
55 * distance-based model and module thresholds work in m/s, so convert once. */
56 Host_Speed = G_stHostVehicleState.Speed * 0.01f; /* cm/s -> m/s */
57 Host_Heading = G_stHostVehicleState.Heading;
58
59 float front_dist = G_stHostVehicleState.FrontCenterUS;
60 float rear_dist = G_stHostVehicleState.BackCenterUS;
61 float front_left = G_stHostVehicleState.FrontLeftUS;
62 float front_right = G_stHostVehicleState.FrontRightUS;
63 float rear_left = G_stHostVehicleState.BackLeftUS;
64 float rear_right = G_stHostVehicleState.BackRightUS;
65
66 /* Speed-dependent safe/critical gaps for this cycle, floored at the minimum. */
67 SafetyEngine_SafeDist = Host_Speed * SAFE_DIST_PER_MS;
68 if (SafetyEngine_SafeDist < MIN_SAFE_DISTANCE)
69 {
70 SafetyEngine_SafeDist = MIN_SAFE_DISTANCE;
71 }
72 SafetyEngine_CriticalDist = SafetyEngine_SafeDist * CRITICAL_RATIO;
73
74 /* 1. Begin cycle */
75 FCW_DNPW_voidBeginCycle(front_dist, front_left);
76 EEBL_voidBeginCycle();
77 BSW_voidBeginCycle(front_left, front_right, rear_left, rear_right);
78 IMA_voidBeginCycle();
79
80 /* 2. Dispatch each neighbor only to the modules its direction can affect. */
81 for (uint8_t i = 0; i < count; i++)
82 {
83 Direction_t dir = SafetyEngine_DetectDirection(Host_Heading, table[i].heading);
84
85 if (dir == DIR_SAME)
86 {
87 /* Same lane ahead/behind: forward collision, rear brake, blind spot. */
88 FCW_DNPW_voidProcessSameDirection();
89 EEBL_voidProcessNeighbor(rear_dist);
90 BSW_voidProcessNeighbor(&table[i]);
91 }
92 else if (dir == DIR_OPPOSITE)
93 {
94 /* Oncoming traffic: head-on FCW vs do-not-pass. */
95 FCW_DNPW_voidProcessOppositeDirection(&table[i]);
96 }
97 else if (dir == DIR_CROSSING)
98 {
99 /* Crossing traffic at an intersection. */
100 IMA_voidProcessNeighbor(&table[i]);
101 }
102 }
103
104 /* 3. Aggregate the modules' results into the 16-bit status word.
105 * 2 bits per module (00 safe / 01 warning / 10 critical); RiskLevel maps
106 * directly. FCW = worst of the local front collision and a confirmed
107 * head-on; BSW is distance-graded (WARNING < 30cm, CRITICAL < 20cm). */
108 uint8_t fcw_front = FCW_GetFrontFlag(); /* 0/1/2 */
109 uint8_t fcw_headon = FCW_GetHeadonConfirmed(); /* 0/1/2 */
110 uint8_t fcw = (fcw_headon > fcw_front) ? fcw_headon : fcw_front;
111 uint8_t dnpw = DNPW_GetFlag(); /* 0=safe/1=warning/2=critical (front-right escalates) */
112 uint8_t bsw = BSW_u8GetSeverity(); /* 0=safe/1=warning/2=critical */
113
114 uint16_t flags = 0;
115 flags |= ((uint16_t)(fcw & SYS_MASK)) << SYS_FCW_POS;
116 flags |= ((uint16_t)(EEBL_u8GetFlag() & SYS_MASK)) << SYS_EEBL_POS;
117 flags |= ((uint16_t)(bsw & SYS_MASK)) << SYS_BSW_POS;
118 flags |= ((uint16_t)(dnpw & SYS_MASK)) << SYS_DNPW_POS;
119 flags |= ((uint16_t)(IMA_u8GetFlag() & SYS_MASK)) << SYS_IMA_POS;
120
121 /* Publish the full 16-bit word (must be uint16: IMA lives at bit 8). */
122 G_u16SystemFlags = flags;
123}

130static float CalcHeadingDiff(float h1, float h2)
131{
132 float diff = h1 - h2;
133
134 if (diff > 180.0f)
135 {
136 diff -= 360.0f;
137 }
138 if (diff < -180.0f)
139 {
140 diff += 360.0f;
141 }
142
143 return (diff < 0.0f) ? -diff : diff;
144}
145
146Direction_t SafetyEngine_DetectDirection(float my_heading, float other_heading)
147{
148 float diff = CalcHeadingDiff(my_heading, other_heading);
149
150 if (diff <= HEADING_SAME_THRESHOLD)
151 {
152 return DIR_SAME;
153 }
154
155 if (diff >= (180.0f - HEADING_OPPOSITE_THRESHOLD))
156 {
157 return DIR_OPPOSITE;
158 }
159
160 if (diff >= (90.0f - HEADING_CROSS_THRESHOLD) && diff <= (90.0f + HEADING_CROSS_THRESHOLD))
161 {
162 return DIR_CROSSING;
163 }
164
165 return DIR_UNKNOWN;
166}

168/* ============ Shared Threshold Risk Evaluation ============ */
169/*
170 * Generic "lower value = higher risk" evaluator, used by IMA for time-gap and
171 * delay thresholds.
172 */
173RiskLevel_t SafetyEngine_EvaluateRisk(float value, float warning_thr, float critical_thr)
174{
175 if (value < 0.0f)
176 {
177 return RISK_SAFE;
178 }
179
180 if (value <= critical_thr)
181 {
182 return RISK_CRITICAL;
183 }
184
185 if (value <= warning_thr)
186 {
187 return RISK_WARNING;
188 }
189
190 return RISK_SAFE;
191}

193/* ============ Shared Distance-Based Risk Assessment ============ */
194
195RiskLevel_t SafetyEngine_AssessDistanceRisk(float host_speed, float distance,
196 float dist_per_ms, float min_dist,
197 float crit_ratio)
198{
199 /* No valid distance reading → nothing in range → safe */
200 if (distance <= 0.0f)
201 {
202 return RISK_SAFE;
203 }
204
205 /* Speed-dependent safe gap, floored so a near-stopped car still has a gap */
206 float safe_dist = host_speed * dist_per_ms;
207 if (safe_dist < min_dist)
208 {
209 safe_dist = min_dist;
210 }
211
212 float critical_dist = safe_dist * crit_ratio;
213
214 if (distance < critical_dist)
215 {
216 return RISK_CRITICAL;
217 }
218
219 if (distance < safe_dist)
220 {
221 return RISK_WARNING;
222 }
223
224 return RISK_SAFE;
225}
