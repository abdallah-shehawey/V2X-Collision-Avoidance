42ErrorState_t USART_Init(USART_Config_t *ChannelConfig)
43{
44 ErrorState_t Local_u8ErrorState = OK;
45
46 if (ChannelConfig == NULL)
47 {
48 Local_u8ErrorState = NULL_POINTER;
49 }
50 else
51 {
52 if(ChannelConfig->Channel > USART_CHANNEL_COUNT)
53 {
54 Local_u8ErrorState = NOK;
55 }
56 else
57 {
58 /* Configure Over Sampling */
59 USART_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->OverSampling & 0X1) << CR1_OVER;
60
61 /* Configure Baud Rate */
62 if (ChannelConfig->OverSampling == USART_OVERSAMPLING_16)
63 {
64 /* BAUD_RATE = (PCK / (8 * (2 - Over8) * USART_DIV) */
65 /* USART_DIV = (PCK / (8 * (2 - Over8) * BAUD_RATE) */
66 /* USART_Fraction = Fraction Part * 16 (in oversampling 16)*/
67 uint32_t Local_u32BaudRate = (double)(MAXIMUM_CLOCK / (8.0 * (2.0 - ChannelConfig->OverSampling) * ChannelConfig->BaudRate) * 1000);
68 uint32_t Local_u32USARTDiv_Mantissa = (uint32_t)(Local_u32BaudRate / 1000);
69 /* +500 To get Round */
70 uint32_t Local_u32USARTDiv_Fraction = (uint32_t)((((Local_u32BaudRate % 1000) * 16) + 500) / 1000);
71
72 USART_Channel[ChannelConfig->Channel]->BRR &= ~BRR_MASKING;
73 USART_Channel[ChannelConfig->Channel]->BRR |= ((Local_u32USARTDiv_Mantissa & 0XFFF) << BRR_DIV_MANTISSA_SHIFTING);
74 USART_Channel[ChannelConfig->Channel]->BRR |= (Local_u32USARTDiv_Fraction & 0XF);
75 }
76 else if (ChannelConfig->OverSampling == USART_OVERSAMPLING_8)
77 {
78 /* BAUD_RATE = (fCK / (8 * (2 - Over8) * USART_DIV) */
79 /* USART_DIV = (fCK / (8 * (2 - Over8) * BAUD_RATE) */
80 /* USART_Fraction = Fraction Part * 8 (in oversampling 8) */
81 uint32_t Local_u32BaudRate = (double)(MAXIMUM_CLOCK / (8.0 * (2.0 - ChannelConfig->OverSampling) * ChannelConfig->BaudRate) * 1000);
82 uint32_t Local_u32USARTDiv_Mantissa = (uint32_t)(Local_u32BaudRate / 1000);
83 /* +500 To get Round */
84 uint32_t Local_u32USARTDiv_Fraction = (uint32_t)((((Local_u32BaudRate % 1000) * 8) + 500) / 1000);
85
86 USART_Channel[ChannelConfig->Channel]->BRR &= ~BRR_MASKING;
87 USART_Channel[ChannelConfig->Channel]->BRR |= ((Local_u32USARTDiv_Mantissa & 0XFFF) << BRR_DIV_MANTISSA_SHIFTING);
88 USART_Channel[ChannelConfig->Channel]->BRR |= (Local_u32USARTDiv_Fraction & 0X7);
89 }
90
91 /* Configure Word Length */
92 USART_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->WordLength & 0X1) << CR1_WL ;
93 /* Configure Stop Bits */
94 USART_Channel[ChannelConfig->Channel]->CR2 |= (ChannelConfig->StopBits & 0X3) << CR2_SB;
95 /* Configure Parity */
96 if (ChannelConfig->Parity != USART_PARITY_NONE)
97 {
98 /* Enable Parity */
99 USART_Channel[ChannelConfig->Channel]->CR1 |= (1 << CR1_PCE);
100 /* Configure Parity */
101 USART_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->Parity & 0X1) << CR1_PS;
102 }
103
104 /* Configure Mode */
105 switch (ChannelConfig->Mode)
106 {
107 case USART_MODE_TX_RX: /* Enable both transmitter and receiver */
108 USART_Channel[ChannelConfig->Channel]->CR1 |= (1 << CR1_TE) | (1 << CR1_RE);
109 break;
110 case USART_MODE_TX: /* Enable transmitter only */
111 USART_Channel[ChannelConfig->Channel]->CR1 |= (1 << CR1_TE);
112 break;
113 case USART_MODE_RX: /* Enable receiver only */
114 USART_Channel[ChannelConfig->Channel]->CR1 |= (1 << CR1_RE);
115 break;
116 default:
117 Local_u8ErrorState = NOK;
118 break;
119 }
120
121 USART_Channel[ChannelConfig->Channel]->CR1 |= (1 << 13);
122 }
123 }
124
125 return Local_u8ErrorState;

250ErrorState_t USART_enumTransmit(USART_Config_t *ChannelConfig, uint8_t TX_Data)
251{
252 ErrorState_t Local_u8ErrorState = OK;
253 ErrorState_t Local_u32TimeoutCounter = 0;
254
255 if (ChannelConfig->Channel > USART_CHANNEL_COUNT)
256 {
257 Local_u8ErrorState = NULL_POINTER;
258 }
259 else
260 {
261 /* Wait until TXE (data register empty), write the byte, then wait until TC
262 * (transmission complete) before returning.
263 *
264 * Why wait for TC and not just TXE: with a TXE-only blocking write, back-to-
265 * back bytes (and especially long 0x00 runs in the DSRC packet — speed=0,
266 * last_update=0, dist=0) were dropped on the wire. The ESP saw zero-runs one
267 * byte short (a 4-zero field arriving as 3), truncating every frame so the
268 * END/checksum always failed. (The RPi path already worked around the SAME
269 * link by sending ASCII with no 0x00 bytes.) Waiting for TC guarantees each
270 * byte — including its stop bit — fully shifts out before the next is loaded,
271 * which makes the binary stream survive intact.
272 *
273 * The timeout is a stuck-hardware escape ONLY; USART_u32TIMEOUT is sized so
274 * it cannot expire during a legitimate byte transfer. */
275 while ((((USART_Channel[ChannelConfig->Channel]->SR & (1 << SR_TXE)) >> SR_TXE) == 0) && (Local_u32TimeoutCounter < USART_u32TIMEOUT))
276 {
277 Local_u32TimeoutCounter++;
278 }
279
280 if (Local_u32TimeoutCounter >= USART_u32TIMEOUT)
281 {
282 Local_u8ErrorState = TIMEOUT_STATE;
283 }
284 else
285 {
286 /* Store Data */
287 USART_Channel[ChannelConfig->Channel]->DR = TX_Data;
288
289 /* Wait for Transmission Complete so the byte fully leaves the shift
290 * register (with its stop bit) before the next DR write can clobber it. */
291 Local_u32TimeoutCounter = 0;
292 while ((((USART_Channel[ChannelConfig->Channel]->SR & (1 << SR_TC)) >> SR_TC) == 0) && (Local_u32TimeoutCounter < USART_u32TIMEOUT))
293 {
294 Local_u32TimeoutCounter++;
295 }
296 }
297 }
298 return Local_u8ErrorState;
299}

356ErrorState_t USART_enumReceive(USART_Config_t *ChannelConfig, uint8_t *RX_Data)
357{
358
359 ErrorState_t Local_u8ErrorState = OK;
360 ErrorState_t Local_u32TimeoutCounter = 0;
361
362 if (ChannelConfig->Channel > USART_CHANNEL_COUNT || RX_Data == NULL)
363 {
364 Local_u8ErrorState = NULL_POINTER;
365 }
366 else
367 {
368 if (USART_u8State[ChannelConfig->Channel] == IDLE)
369 {
370 USART_u8State[ChannelConfig->Channel] = BUSY;
371
372 while((USART_Channel[ChannelConfig->Channel]->SR & (1 << SR_RXNE)) >> SR_RXNE == 0 && Local_u32TimeoutCounter != USART_u32TIMEOUT)
373 {
374 Local_u32TimeoutCounter++;
375 }
376
377 if (Local_u32TimeoutCounter == USART_u32TIMEOUT)
378 {
379 Local_u8ErrorState = TIMEOUT_STATE;
380 }
381 else
382 {
383 /* Store Data */
384 *RX_Data = USART_Channel[ChannelConfig->Channel]->DR;
385 }
386
387 USART_u8State[ChannelConfig->Channel] = IDLE;
388 }
389 else
390 {
391 Local_u8ErrorState = BUSY_STATE;
392 }
393 }
394 return Local_u8ErrorState;

397uint8_t USART_ReceiveByteDirect(USART_Channel_t Channel)
398{
399 /* Read SR THEN DR: this exact sequence also clears the ORE (overrun) flag.
400 * Reading DR alone would leave a set ORE asserting the RXNE interrupt line
401 * forever → an ISR storm that starves every task. Reading SR first makes
402 * the overrun self-clearing on the next received byte. */
403 volatile uint32_t Local_u32Status = USART_Channel[Channel]->SR;
404 (void)Local_u32Status;
405 return (uint8_t)(USART_Channel[Channel]->DR);
406}
