25// ====== Struct ======
26// Wire format shared with the ESP firmware (esp32/master, esp32/slave). It is
27// transmitted as raw bytes over UART/ESP-NOW, so it MUST be packed (no implicit
28// padding) and IDENTICAL on every side. Packed → 21 bytes exactly; any field
29// reorder/resize must be mirrored in the ESP .ino structs.
30typedef struct __attribute__((packed))
31{
32 uint8_t vehicle_id;
33 float speed;
34 float heading; /* 0 to 360 degrees */
35 uint32_t last_update;
36 uint8_t fcw_headon_flag; /* cooperative head-on candidate (front vehicle + opposite neighbor): 0/1 */
37 uint8_t bsw_flag; /* sender's front side(s): bit0=LEFT, bit1=RIGHT (0=none,1,2,3=both) */
38 float distance_to_intersection; /* distance to nearest intersection (cm), 0 = not near */
39 uint8_t ima_flag; /* 0=Safe, 1=Warning, 2=Critical */
40} Neighbor;
