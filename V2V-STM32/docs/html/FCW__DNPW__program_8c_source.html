10/*
11 * Cooperative Forward Collision Warning (FCW) and Do-Not-Pass Warning (DNPW).
12 * Both are derived from the same per-cycle signals, so they share one module.
13 *
14 * The SafetyEngine feeds the module by direction during the neighbor pass; the
15 * three flags are then derived on demand by the getters:
16 *
17 * fcw_front_flag : forward collision with a vehicle ahead in the same lane.
18 * Local, severity WARNING/CRITICAL from the front distance.
19 *
20 * fcw_headon_flag : head-on candidate — a vehicle ahead and an oncoming
21 * vehicle both present. Broadcast over DSRC so the oncoming
22 * car can confirm. Boolean.
23 *
24 * dnpw_flag : the oncoming car has no head-on candidate of its own, so
25 * it is in another lane — an overtaking risk, not head-on.
26 *
27 * Severity comes from the front distance against the cycle safe/critical gaps
28 * (SafetyEngine_SafeDist / SafetyEngine_CriticalDist).
29 */

86/*
87 * @brief Start a new cycle: latch the front distances and reset the signals.
88 * @param front_distance Front-center ultrasonic distance (cm)
89 * @param front_left_distance Front-left ultrasonic distance (cm)
90 */
91void FCW_DNPW_voidBeginCycle(float front_distance, float front_left_distance)
92{
93 /* Two front gates (not yet known to be a vehicle — a neighbor confirms later):
94 * - FCW gate (wider, 40 cm) feeds the same-lane forward collision: warn earlier.
95 * - DNPW/head-on gate (nearer, 20 cm) feeds the oncoming/overtaking case. */
96 FCW_DNPW_FcwObject = (front_distance > 0.0f && front_distance < FCW_FRONT_THRESHOLD) ? 1U : 0U;
97 FCW_DNPW_FrontObject = (front_distance > 0.0f && front_distance < DNPW_FRONT_THRESHOLD) ? 1U : 0U;
98
99 /* The oncoming car alongside on the LEFT (overtaking side) escalates a DNPW to
100 * CRITICAL: to pass we pull out left, so the car we must not pass is on our left. */
101 FCW_DNPW_FrontLeftNear = (front_left_distance > 0.0f && front_left_distance < DNPW_FRONT_LEFT_CRITICAL) ? 1U : 0U;
102
103 /* Latch the distance-based severity once. Two severities: a same-direction
104 * collision closes at the speed difference, but a head-on closes at the sum of
105 * both speeds (~double), so the head-on gaps are doubled to warn at twice the
106 * distance. The same-direction severity uses the wider FCW gate; the head-on
107 * severity uses the nearer DNPW gate. */
108 FCW_DNPW_FrontSeverity = RISK_SAFE;
109 FCW_DNPW_HeadonSeverity = RISK_SAFE;
110
111 /* Same-direction (FCW gate): closing at the speed difference, plain gaps. */
112 if (FCW_DNPW_FcwObject && front_distance < SafetyEngine_SafeDist)
113 {
114 FCW_DNPW_FrontSeverity = (front_distance < SafetyEngine_CriticalDist) ? RISK_CRITICAL : RISK_WARNING;
115 }
116
117 /* Head-on (DNPW gate): closing at the sum of both speeds (~double), doubled gaps. */
118 if (FCW_DNPW_FrontObject && front_distance < SafetyEngine_SafeDist * 2.0f)
119 {
120 FCW_DNPW_HeadonSeverity = (front_distance < SafetyEngine_CriticalDist * 2.0f) ? RISK_CRITICAL : RISK_WARNING;
121 }
122
123 /* Reset the results for the new cycle. */
124 FCW_DNPW_FrontFlag = RISK_SAFE;
125 FCW_DNPW_HeadonFlag = 0;
126 FCW_DNPW_HeadonConfirmed = 0;
127 FCW_DNPW_DnpwFlag = RISK_SAFE;
128}

141/*
142 * @brief An oncoming neighbor exists. With an object ahead, that makes a head-on
143 * candidate (broadcast for confirmation). Whether the oncoming car raised
144 * its own head-on flag splits the case:
145 * - it did → both face the same obstacle: confirmed head-on (severity).
146 * - it did not → it is in another lane: a do-not-pass / overtaking risk.
147 * All three results are settled here, ready for the getters to return.
148 * @param n Pointer to neighbor data
149 */
150void FCW_DNPW_voidProcessOppositeDirection(const Neighbor *n)
151{
152 if (!FCW_DNPW_FrontObject)
153 {
154 return; /* no object ahead — oncoming car alone is not our hazard */
155 }
156
157 FCW_DNPW_HeadonFlag = 1; /* candidate: object ahead + oncoming present */
158
159 if (n->fcw_headon_flag > 0)
160 {
161 /* Same obstacle: a real head-on overrides any do-not-pass seen this cycle.
162 * Use the head-on severity (doubled gaps) for the higher closing speed. */
163 FCW_DNPW_HeadonConfirmed = (uint8_t)FCW_DNPW_HeadonSeverity;
164 FCW_DNPW_DnpwFlag = RISK_SAFE;
165 }
166 else if (!FCW_DNPW_HeadonConfirmed)
167 {
168 /* Oncoming is in another lane: overtaking risk. WARNING by default, CRITICAL
169 * when the front-left sensor reads a near object (the oncoming car alongside
170 * on the overtaking side). */
171 FCW_DNPW_DnpwFlag = FCW_DNPW_FrontLeftNear ? RISK_CRITICAL : RISK_WARNING;
172 }
173}
