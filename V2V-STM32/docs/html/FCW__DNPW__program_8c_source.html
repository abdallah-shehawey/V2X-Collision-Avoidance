14/*
15 * Cooperative Forward Collision Warning (FCW) and Do-Not-Pass Warning (DNPW).
16 * Both are derived from the same per-cycle signals, so they share one module.
17 *
18 * The SafetyEngine feeds the module by direction during the neighbor pass; the
19 * three flags are then derived on demand by the getters:
20 *
21 * fcw_front_flag : forward collision with a vehicle ahead in the same lane.
22 * Local, severity WARNING/CRITICAL from the front distance.
23 *
24 * fcw_headon_flag : head-on candidate — a vehicle ahead and an oncoming
25 * vehicle both present. Broadcast over DSRC so the oncoming
26 * car can confirm. Boolean.
27 *
28 * dnpw_flag : the oncoming car has no head-on candidate of its own, so
29 * it is in another lane — an overtaking risk, not head-on.
30 *
31 * Severity comes from the front distance against the cycle safe/critical gaps
32 * (SafetyEngine_SafeDist / SafetyEngine_CriticalDist).
33 */

81void FCW_DNPW_voidBeginCycle(float front_distance, float front_left_distance)
82{
83 /* Two front gates (not yet known to be a vehicle — a neighbor confirms later):
84 * - FCW gate (wider, 40 cm) feeds the same-lane forward collision: warn earlier.
85 * - DNPW/head-on gate (nearer, 20 cm) feeds the oncoming/overtaking case. */
86 FCW_DNPW_FcwObject = (front_distance > 0.0f && front_distance < FCW_FRONT_THRESHOLD) ? 1U : 0U;
87 FCW_DNPW_FrontObject = (front_distance > 0.0f && front_distance < DNPW_FRONT_THRESHOLD) ? 1U : 0U;
88
89 /* The oncoming car alongside on the LEFT (overtaking side) escalates a DNPW to
90 * CRITICAL: to pass we pull out left, so the car we must not pass is on our left. */
91 FCW_DNPW_FrontLeftNear = (front_left_distance > 0.0f && front_left_distance < DNPW_FRONT_LEFT_CRITICAL) ? 1U : 0U;
92
93 /* Latch the distance-based severity once. Two severities: a same-direction
94 * collision closes at the speed difference, but a head-on closes at the sum of
95 * both speeds (~double), so the head-on gaps are doubled to warn at twice the
96 * distance. The same-direction severity uses the wider FCW gate; the head-on
97 * severity uses the nearer DNPW gate. */
98 FCW_DNPW_FrontSeverity = RISK_SAFE;
99 FCW_DNPW_HeadonSeverity = RISK_SAFE;
100
101 /* Same-direction (FCW gate): closing at the speed difference, plain gaps. */
102 if (FCW_DNPW_FcwObject && front_distance < SafetyEngine_SafeDist)
103 {
104 FCW_DNPW_FrontSeverity = (front_distance < SafetyEngine_CriticalDist) ? RISK_CRITICAL : RISK_WARNING;
105 }
106
107 /* Head-on (DNPW gate): closing at the sum of both speeds (~double), doubled gaps. */
108 if (FCW_DNPW_FrontObject && front_distance < SafetyEngine_SafeDist * 2.0f)
109 {
110 FCW_DNPW_HeadonSeverity = (front_distance < SafetyEngine_CriticalDist * 2.0f) ? RISK_CRITICAL : RISK_WARNING;
111 }
112
113 /* Reset the results for the new cycle. */
114 FCW_DNPW_FrontFlag = RISK_SAFE;
115 FCW_DNPW_HeadonFlag = 0;
116 FCW_DNPW_HeadonConfirmed = 0;
117 FCW_DNPW_DnpwFlag = RISK_SAFE;
118}

140void FCW_DNPW_voidProcessOppositeDirection(const Neighbor *n)
141{
142 if (!FCW_DNPW_FrontObject)
143 {
144 return; /* no object ahead — oncoming car alone is not our hazard */
145 }
146
147 FCW_DNPW_HeadonFlag = 1; /* candidate: object ahead + oncoming present */
148
149 if (n->fcw_headon_flag > 0)
150 {
151 /* Same obstacle: a real head-on overrides any do-not-pass seen this cycle.
152 * Use the head-on severity (doubled gaps) for the higher closing speed. */
153 FCW_DNPW_HeadonConfirmed = (uint8_t)FCW_DNPW_HeadonSeverity;
154 FCW_DNPW_DnpwFlag = RISK_SAFE;
155 }
156 else if (!FCW_DNPW_HeadonConfirmed)
157 {
158 /* Oncoming is in another lane: overtaking risk. WARNING by default, CRITICAL
159 * when the front-left sensor reads a near object (the oncoming car alongside
160 * on the overtaking side). */
161 FCW_DNPW_DnpwFlag = FCW_DNPW_FrontLeftNear ? RISK_CRITICAL : RISK_WARNING;
162 }
163}
