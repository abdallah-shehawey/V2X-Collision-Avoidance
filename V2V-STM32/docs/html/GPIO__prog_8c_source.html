41ErrorState_t GPIO_enumPinInit(const GPIO_PinConfig_t *PinConfig)
42{
43 ErrorState_t Local_u8ErrorState = OK;
44
45 if (PinConfig != NULL)
46 {
47 /* Check if port and pin numbers are valid */
48 if ((PinConfig->Port <= GPIO_PORTH ) &&
49 (PinConfig->PinNum <= GPIO_PIN15 ) &&
50 (PinConfig->Mode <= GPIO_ANALOG ) &&
51 (PinConfig->Otype <= GPIO_OPEN_DRAIN ) &&
52 (PinConfig->Speed <= GPIO_VERY_HIGH_SPEED) &&
53 (PinConfig->PullType <= GPIO_PULL_DOWN ) &&
54 (PinConfig->AlternateFunction <= GPIO_AF15 ))
55 {
56 /* Configure pin mode (Input/Output/Alternate Function/Analog) */
57 (GPIO_Port[PinConfig->Port]->MODER) &= ~(MODER_MASK << ((PinConfig->PinNum) * MODER_PIN_ACCESS));
58 (GPIO_Port[PinConfig->Port]->MODER) |= (PinConfig->Mode << ((PinConfig->PinNum) * MODER_PIN_ACCESS));
59 //_______________________________________________________________________________________________________________//
60 /* Set Pull up or Pull down or No pull */
61 // if the pin mode is analog, HW will ignore the pull type
62 (GPIO_Port[PinConfig->Port]->PUPDR) &= ~(PUPDR_MASK << ((PinConfig->PinNum) * PUPDR_PIN_ACCESS));
63 (GPIO_Port[PinConfig->Port]->PUPDR) |= (PinConfig->PullType << ((PinConfig->PinNum) * PUPDR_PIN_ACCESS));
64 //_______________________________________________________________________________________________________________//
65 /* select output type and output speed in case of general purpose output or alternate function */
66 if ((PinConfig->Mode == GPIO_OUTPUT) || (PinConfig->Mode == GPIO_ALTFN))
67 {
68 /* Set output type : push pull or open drain */
69 GPIO_Port[PinConfig->Port]->OTYPER &= ~(OTYPER_MASK << PinConfig->PinNum);
70 GPIO_Port[PinConfig->Port]->OTYPER |= (PinConfig->Otype << PinConfig->PinNum);
71 //_______________________________________________________________________________________________________________//
72 /* Configure output speed : low, medium, high, fast */
73 GPIO_Port[PinConfig->Port]->OSPEEDR &= ~(OSPEEDR_MASK << (PinConfig->PinNum * OSPEEDR_PIN_ACCESS));
74 GPIO_Port[PinConfig->Port]->OSPEEDR |= (PinConfig->Speed << (PinConfig->PinNum * OSPEEDR_PIN_ACCESS));
75 //_______________________________________________________________________________________________________________//
76 /* select the pin alternate function */
77 if (PinConfig->Mode == GPIO_ALTFN)
78 {
79 GPIO_Port[PinConfig->Port]->AFR[(PinConfig->PinNum) / AFR_PIN_SHIFT] &= ~(AFR_MASK << (((PinConfig->PinNum) % 8u) * AFR_PIN_ACCESS));
80 GPIO_Port[PinConfig->Port]->AFR[(PinConfig->PinNum) / AFR_PIN_SHIFT] |= (PinConfig->AlternateFunction << (((PinConfig->PinNum) % 8u) * AFR_PIN_ACCESS));
81 }
82 }
83 }
84 else
85 {
86 Local_u8ErrorState = NOK;
87 }
88 }
89 else
90 {
91 Local_u8ErrorState = NULL_POINTER;
92 }
93 return Local_u8ErrorState;
94}

486{
487 ErrorState_t Local_u8ErrorState = OK;
488
489 /* Check if port and pin numbers are valid */
490 if ((Port <= GPIO_PORTH) && (PinNum <= GPIO_PIN15) && (PinVal == GPIO_PIN_HIGH || PinVal == GPIO_PIN_LOW))
491 {
492 /* Write the value to the pin using BSRR register for atomic access */
493 switch (PinVal)
494 {
495 case GPIO_PIN_HIGH:
496 GPIO_Port[Port]->BSRR = (1u << PinNum); /* Set bits are in the low half of BSRR */
497 /*GPIO_Port[Port]->ODR |= (1u << PinNum);*/
498 break;
499 case GPIO_PIN_LOW:
500 GPIO_Port[Port]->BSRR = (1u << (PinNum + 16u)); /* Reset bits are in the high half of BSRR */
501 /*GPIO_Port[Port]->ODR &= ~(1u << PinNum);*/
502 break;
503 default:
504 Local_u8ErrorState = NOK;
505 break;
506 }
507 }
508 else
509 {
510 Local_u8ErrorState = NOK;
511 }
512 return Local_u8ErrorState;
