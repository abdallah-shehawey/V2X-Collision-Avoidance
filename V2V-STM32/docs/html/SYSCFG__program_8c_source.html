20/*
21 * @brief Configure EXTI line mapping to GPIO port
22 * @details Maps a specific EXTI line to a GPIO port in the SYSCFG peripheral
23 * @param Copy_u8EXTI - EXTI line number (0-15)
24 * Copy_u8Port - GPIO port identifier (A-H)
25 * @return void
26 * @note
27 * 1. Each EXTI line is mapped to a 4-bit field in one of four EXTICR registers
28 * 2. The mapping is done in groups of 4 EXTI lines per register
29 * 3. Must be called before configuring EXTI line
30 * @par Example:
31 *
32 * SYSCFG_vSetEXTIConfig(SYSCFG_EXTI0, SYSCFG_PORTA); // Map EXTI0 to Port A
33 * @internal:
34 * - Local_u8RegNum: Calculates which EXTICR register to use (0-3)
35 * - Local_u8BitNum: Calculates the bit position within the register (0-12)
36 * - 0x0F mask: Clears the current port mapping (4 bits)
37 */
38void SYSCFG_vSetEXTIConfig(SYSCFG_EXTI_t Copy_u8EXTI, SYSCFG_Port_t Copy_u8Port)
39{
40 /* Calculate register number (0-3) based on EXTI line number */
41 uint8_t Local_u8RegNum = Copy_u8EXTI / EXTI_CTRL_REG_LINEBITS;
42
43 /* Calculate bit position within register (0-12) */
44 uint8_t Local_u8BitNum = (Copy_u8EXTI % EXTI_CTRL_REG_LINEBITS) * EXTI_CTRL_REG_LINEBITS;
45
46 /* Clear current port mapping for this EXTI line */
47 MSYSCFG->EXTICR[Local_u8RegNum] &= ~(EXTI_CTRL_REG_MASK << Local_u8BitNum);
48
49 /* Set new port mapping for this EXTI line */
50 MSYSCFG->EXTICR[Local_u8RegNum] |= (Copy_u8Port << Local_u8BitNum);
51}
