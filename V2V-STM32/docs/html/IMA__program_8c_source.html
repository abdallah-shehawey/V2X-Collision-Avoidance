44/*
45 * @brief Process one crossing-direction DSRC neighbor for IMA.
46 *
47 * Any crossing vehicle raises an alert so the driver knows an IMA case is in
48 * play; the priority rule (higher speed = right of way) only sets its severity:
49 * - host faster or equal → it passes first → WARNING.
50 * - host slower → it must yield → CRITICAL.
51 * The SafetyEngine only dispatches crossing-direction neighbors here.
52 *
53 * @param n Pointer to neighbor data
54 */
55void IMA_voidProcessNeighbor(const Neighbor *n)
56{
57 /* A crossing vehicle is detected */
58 IMA_CrossingDetected = 1;
59
60 /* Severity from the priority rule: the slower vehicle yields (CRITICAL), the
61 * faster/equal one has right of way and passes first (WARNING). Either way an
62 * alert fires so both drivers know an IMA case is in play. */
63 RiskLevel_t risk = (Host_Speed < n->speed) ? RISK_CRITICAL /* host slower → yield */
64 : RISK_WARNING; /* host faster/equal → right of way */
65
66 if (risk > IMA_WorstRisk)
67 {
68 IMA_WorstRisk = risk;
69 }
70}

74/*
75 * @brief Current IMA flag. Any crossing vehicle raises an alert; the severity is
76 * WARNING when the host has right of way, CRITICAL when it must yield.
77 * @return 0=Safe, 1=Warning, 2=Critical
78 */
79uint8_t IMA_u8GetFlag(void)
80{
81 if (IMA_CrossingDetected && IMA_WorstRisk > RISK_SAFE)
82 {
83 return (uint8_t)IMA_WorstRisk;
84 }
85 return 0;
