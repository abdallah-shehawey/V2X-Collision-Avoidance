26/*
27 * @brief Initialize the SPI peripheral with the provided configuration.
28 *
29 * This function configures all SPI parameters including clock phase, polarity,
30 * master/slave mode, baud rate, data frame format, and other settings.
31 *
32 * @param ChannelConfig Pointer to the SPI configuration structure
33 * @return ErrorState_t OK if initialization successful, error code otherwise
34 *
35 * @warning NULL pointer check is performed on the input parameter.
36 */
37ErrorState_t SPI_enumInit(SPI_Config_t *ChannelConfig)
38{
39 ErrorState_t Local_u8ErrorState = OK;
40
41 // Check for NULL pointer input
42 if (ChannelConfig == NULL)
43 {
44 Local_u8ErrorState = NULL_POINTER;
45 }
46 else
47 {
48 // Validate channel number
49 if(ChannelConfig->Channel > SPI_CHANNEL_COUNT)
50 {
51 Local_u8ErrorState = NOK;
52 }
53 else
54 {
55 // Configure CLOCK PHASE (CPHA)
56 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->CPHA & 0X1) << CR1_CPHA;
57 // Configure Clock Polarity (CPOL)
58 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->CPOL & 0X1) << CR1_CPOL;
59 // Configure Master/Slave Mode
60 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->Mode & 0X1) << CR1_MSTR;
61 // Configure Baud Rate Prescaler
62 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->BaudRatePrescaler & 0X7) << CR1_BR;
63 // Configure Bit Order (MSB/LSB first)
64 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->BFIRST & 0X1) << CR1_LSBFIRST;
65 // Configure NSS Management (Hardware/Software)
66 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->NSS_MAN & 0X1) << CR1_SSM;
67 // If using Software NSS, configure NSS Mode
68 if (ChannelConfig->NSS_MAN == SPI_NSS_SOFTWARE)
69 {
70 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->NSSI_MODE & 0X1) << CR1_SSI;
71 }
72 // Configure Receive Only Mode
73 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->RXONLY & 0X1) << CR1_RXONLY;
74 // Configure Data Frame Format (8/16 bit)
75 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->DFF & 0X1) << CR1_DFF;
76 // Configure CRC Mode
77 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->CRC_MODE & 0X1) << CR1_CRCEN;
78 // If CRC is enabled, configure CRC Next
79 if (ChannelConfig->CRC_MODE == SPI_CRCEN)
80 {
81 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->CRCNEXT & 0X1) << CR1_CRCNEXT;
82 }
83 // Configure Bidirectional Mode
84 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->DIMODE & 0X1) << CR1_BIDIMODE;
85 // If Bidirectional mode is enabled, configure Output Enable
86 if (ChannelConfig->DIMODE == SPI_BIDIMODE)
87 {
88 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->BIDIOE & 0X1) << CR1_BIDIOE;
89 }
90 // Enable SPI peripheral
91 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->SPE & 0X1) << CR1_SPE;
92 }
93 }
94 return Local_u8ErrorState;
95}

97/*
98 * @brief Perform full-duplex SPI transaction (transmit and receive)
99 *
100 * This function sends data and simultaneously receives data through the SPI interface.
101 * It implements a blocking mechanism with timeout protection.
102 *
103 * @param ChannelConfig Pointer to the SPI configuration structure
104 * @param TX_Data Data to be transmitted
105 * @param RX_Data Pointer to store received data
106 * @return ErrorState_t OK if successful, error code otherwise
107 *
108 * @note This is a blocking function with timeout protection
109 * @warning NULL pointer check is performed on input parameters
110 */
111ErrorState_t SPI_enumTrancieve(SPI_Config_t *ChannelConfig, uint16_t TX_Data, uint16_t *RX_Data)
112{
113 ErrorState_t Local_u8ErrorState = OK;
114 uint32_t Local_u32TimeoutCounter = 0;
115
116 // Check for NULL pointers
117 if (ChannelConfig == NULL || RX_Data == NULL)
118 {
119 Local_u8ErrorState = NULL_POINTER;
120 }
121 else
122 {
123 // Validate channel number
124 if (ChannelConfig->Channel > SPI_CHANNEL_COUNT)
125 {
126 Local_u8ErrorState = NOK;
127 }
128 else
129 {
130 // Check if SPI is not busy
131 if (SPI_u8State == IDLE)
132 {
133 SPI_u8State = BUSY;
134 while((SPI_Channel[ChannelConfig->Channel]->SR & (1 << SR_TXE)) >> SR_TXE == 0 && (Local_u32TimeoutCounter != SPI_u32TIMEOUT))
135 {
136 Local_u32TimeoutCounter++;
137 }
138
139 // Check if timeout occurred
140 if (Local_u32TimeoutCounter == SPI_u32TIMEOUT)
141 {
142 Local_u8ErrorState = TIMEOUT_STATE;
143 }
144 else
145 {
146 // Transmit data
147 SPI_Channel[ChannelConfig->Channel]->DR = TX_Data;
148 }
149
150 // Reset timeout counter
151 Local_u32TimeoutCounter = 0;
152 while((SPI_Channel[ChannelConfig->Channel]->SR & (1 << SR_RXNE)) >> SR_RXNE == 0 && (Local_u32TimeoutCounter != SPI_u32TIMEOUT))
153 {
154 Local_u32TimeoutCounter++;
155 }
156
157 // Check if timeout occurred
158 if (Local_u32TimeoutCounter == SPI_u32TIMEOUT)
159 {
160 Local_u8ErrorState = TIMEOUT_STATE;
161 }
162 else
163 {
164 // Read received data
165 *RX_Data = SPI_Channel[ChannelConfig->Channel]->DR;
166 }
167
168 // Mark SPI as idle
169 SPI_u8State = IDLE;
170 }
171 }
172 }
173 return Local_u8ErrorState;
174}
