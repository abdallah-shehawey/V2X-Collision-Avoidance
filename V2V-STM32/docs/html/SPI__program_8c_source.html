38ErrorState_t SPI_enumInit(SPI_Config_t *ChannelConfig)
39{
40 ErrorState_t Local_u8ErrorState = OK;
41
42 // Check for NULL pointer input
43 if (ChannelConfig == NULL)
44 {
45 Local_u8ErrorState = NULL_POINTER;
46 }
47 else
48 {
49 // Validate channel number
50 if(ChannelConfig->Channel > SPI_CHANNEL_COUNT)
51 {
52 Local_u8ErrorState = NOK;
53 }
54 else
55 {
56 // Configure CLOCK PHASE (CPHA)
57 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->CPHA & 0X1) << CR1_CPHA;
58 // Configure Clock Polarity (CPOL)
59 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->CPOL & 0X1) << CR1_CPOL;
60 // Configure Master/Slave Mode
61 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->Mode & 0X1) << CR1_MSTR;
62 // Configure Baud Rate Prescaler
63 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->BaudRatePrescaler & 0X7) << CR1_BR;
64 // Configure Bit Order (MSB/LSB first)
65 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->BFIRST & 0X1) << CR1_LSBFIRST;
66 // Configure NSS Management (Hardware/Software)
67 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->NSS_MAN & 0X1) << CR1_SSM;
68 // If using Software NSS, configure NSS Mode
69 if (ChannelConfig->NSS_MAN == SPI_NSS_SOFTWARE)
70 {
71 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->NSSI_MODE & 0X1) << CR1_SSI;
72 }
73 // Configure Receive Only Mode
74 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->RXONLY & 0X1) << CR1_RXONLY;
75 // Configure Data Frame Format (8/16 bit)
76 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->DFF & 0X1) << CR1_DFF;
77 // Configure CRC Mode
78 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->CRC_MODE & 0X1) << CR1_CRCEN;
79 // If CRC is enabled, configure CRC Next
80 if (ChannelConfig->CRC_MODE == SPI_CRCEN)
81 {
82 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->CRCNEXT & 0X1) << CR1_CRCNEXT;
83 }
84 // Configure Bidirectional Mode
85 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->DIMODE & 0X1) << CR1_BIDIMODE;
86 // If Bidirectional mode is enabled, configure Output Enable
87 if (ChannelConfig->DIMODE == SPI_BIDIMODE)
88 {
89 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->BIDIOE & 0X1) << CR1_BIDIOE;
90 }
91 // Enable SPI peripheral
92 SPI_Channel[ChannelConfig->Channel]->CR1 |= (ChannelConfig->SPE & 0X1) << CR1_SPE;
93 }
94 }
95 return Local_u8ErrorState;
96}

112ErrorState_t SPI_enumTrancieve(SPI_Config_t *ChannelConfig, uint16_t TX_Data, uint16_t *RX_Data)
113{
114 ErrorState_t Local_u8ErrorState = OK;
115 uint32_t Local_u32TimeoutCounter = 0;
116
117 // Check for NULL pointers
118 if (ChannelConfig == NULL || RX_Data == NULL)
119 {
120 Local_u8ErrorState = NULL_POINTER;
121 }
122 else
123 {
124 // Validate channel number
125 if (ChannelConfig->Channel > SPI_CHANNEL_COUNT)
126 {
127 Local_u8ErrorState = NOK;
128 }
129 else
130 {
131 // Check if SPI is not busy
132 if (SPI_u8State == IDLE)
133 {
134 SPI_u8State = BUSY;
135 while((SPI_Channel[ChannelConfig->Channel]->SR & (1 << SR_TXE)) >> SR_TXE == 0 && (Local_u32TimeoutCounter != SPI_u32TIMEOUT))
136 {
137 Local_u32TimeoutCounter++;
138 }
139
140 // Check if timeout occurred
141 if (Local_u32TimeoutCounter == SPI_u32TIMEOUT)
142 {
143 Local_u8ErrorState = TIMEOUT_STATE;
144 }
145 else
146 {
147 // Transmit data
148 SPI_Channel[ChannelConfig->Channel]->DR = TX_Data;
149 }
150
151 // Reset timeout counter
152 Local_u32TimeoutCounter = 0;
153 while((SPI_Channel[ChannelConfig->Channel]->SR & (1 << SR_RXNE)) >> SR_RXNE == 0 && (Local_u32TimeoutCounter != SPI_u32TIMEOUT))
154 {
155 Local_u32TimeoutCounter++;
156 }
157
158 // Check if timeout occurred
159 if (Local_u32TimeoutCounter == SPI_u32TIMEOUT)
160 {
161 Local_u8ErrorState = TIMEOUT_STATE;
162 }
163 else
164 {
165 // Read received data
166 *RX_Data = SPI_Channel[ChannelConfig->Channel]->DR;
167 }
168
169 // Mark SPI as idle
170 SPI_u8State = IDLE;
171 }
172 }
173 }
174 return Local_u8ErrorState;
175}
