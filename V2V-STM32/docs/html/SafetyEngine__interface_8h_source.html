29/* ====== Shared Safe-Distance Model ======
30 * Speed-dependent gap shared by the distance-based modules (Local FCW + EEBL):
31 *
32 * safe_cm = Host_Speed(m/s) * SAFE_DIST_PER_MS (floored at MIN_SAFE_DISTANCE)
33 * crit_cm = safe_cm * CRITICAL_RATIO
34 *
35 * Prototype scale (small car, top speed ~0.5 m/s, corridor distances ~10-20 cm):
36 * @ 0.5 m/s -> safe = max(0.5*20, 15) = 15 cm [warning zone: 10-15 cm]
37 * crit = safe * 0.667 = 10 cm (the system minimum distance)
38 * Dashboard thresholds: SAFE=15cm WARNING=10cm CRITICAL=10cm
39 * -> Use MIN_SAFE_DISTANCE as the dominant floor. */
40#define SAFE_DIST_PER_MS (40.0f) /* cm of safe gap per 1 m/s (prototype) */
41#define MIN_SAFE_DISTANCE (30.0f) /* floor: SAFE zone >= 30 cm */
42#define CRITICAL_RATIO (0.667f) /* CRITICAL below ~67% -> crit = 20 cm */
